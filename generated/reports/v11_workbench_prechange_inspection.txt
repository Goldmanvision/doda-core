 489: def validate_metric_tiers(
 490:     tiers: list[dict[str, str]],
 491:     metrics_by_id: dict[str, dict[str, str]],
 492:     flavor_text_ids: set[str],
 493:     result: ValidationResult,
 494: ) -> None:
 495:     tiers_by_metric: dict[str, list[dict[str, str]]] = defaultdict(list)
 496: 
 497:     for row in tiers:
 498:         metric_id = row["metric_id"]
 499: 
 500:         if metric_id not in metrics_by_id:
 501:             result.error(
 502:                 f"[metric_tiers] Line {row['_line_number']}: metric_id '{metric_id}' "
 503:                 "does not exist in metrics.csv"
 504:             )
 505: 
 506:         flavor_text_id = row["flavor_text_id"]
 507:         if flavor_text_id not in flavor_text_ids:
 508:             result.error(
 509:                 f"[metric_tiers] Line {row['_line_number']}: flavor_text_id "
 510:                 f"'{flavor_text_id}' does not exist in flavor_text.csv"
 511:             )
 512: 
 513:         minimum = parse_int("metric_tiers", row, "min_inclusive", result)
 514:         maximum = parse_int("metric_tiers", row, "max_inclusive", result)
 515: 
 516:         if minimum is not None and maximum is not None and minimum > maximum:
 517:             result.error(
 518:                 f"[metric_tiers] Line {row['_line_number']}: min_inclusive "
 519:                 f"({minimum}) cannot exceed max_inclusive ({maximum})"
 520:             )
 521: 
 522:         tiers_by_metric[metric_id].append(row)
 523: 
 524:     validate_sort_order("metric_tiers", tiers, result)
 525: 
 526:     for metric_id, metric_row in metrics_by_id.items():
 527:         metric_tiers = tiers_by_metric.get(metric_id, [])
 528: 
 529:         if not metric_tiers:
 530:             result.error(f"[metric_tiers] No tiers exist for metric '{metric_id}'")
 531:             continue
 532: 
 533:         parsed_tiers: list[tuple[int, int, int, dict[str, str]]] = []
 534: 
 535:         for row in metric_tiers:
 536:             minimum = parse_int("metric_tiers", row, "min_inclusive", result)
 537:             maximum = parse_int("metric_tiers", row, "max_inclusive", result)
 538:             sort_order = parse_int("metric_tiers", row, "sort_order", result)
 539: 
 540:             if minimum is None or maximum is None or sort_order is None:
 541:                 continue
 542: 
 543:             parsed_tiers.append((minimum, maximum, sort_order, row))
 544: 
 545:         parsed_tiers.sort(key=lambda item: (item[0], item[1], item[2]))
 546: 
 547:         metric_minimum = parse_int("metrics", metric_row, "internal_min", result)
 548:         metric_maximum = parse_int("metrics", metric_row, "internal_max", result)
 549: 
 550:         if metric_minimum is None or metric_maximum is None or not parsed_tiers:
 551:             continue
 552: 
 553:         first_minimum = parsed_tiers[0][0]
 554:         if first_minimum != metric_minimum:
 555:             result.error(
 556:                 f"[metric_tiers] Metric '{metric_id}' begins at {first_minimum}, "
 557:                 f"but metrics.csv defines internal_min {metric_minimum}"
 558:             )
 559: 
 560:         expected_next_minimum = metric_minimum
 561: 
 562:         for minimum, maximum, _, row in parsed_tiers:
 563:             if minimum < expected_next_minimum:
 564:                 result.error(
 565:                     f"[metric_tiers] Line {row['_line_number']}: tier range "
 566:                     f"[{minimum}, {maximum}] overlaps or starts before the "
 567:                     f"expected next value {expected_next_minimum} for {metric_id}"
 568:                 )
 569:             elif minimum > expected_next_minimum:
 570:                 result.error(
 571:                     f"[metric_tiers] Line {row['_line_number']}: tier range "
 572:                     f"[{minimum}, {maximum}] leaves a gap after "
 573:                     f"{expected_next_minimum - 1} for {metric_id}"
 574:                 )
 575: 
 576:             expected_next_minimum = max(expected_next_minimum, maximum + 1)
 577: 
 578:         final_maximum = parsed_tiers[-1][1]
 579:         if final_maximum != metric_maximum:
 580:             result.error(
 581:                 f"[metric_tiers] Metric '{metric_id}' ends at {final_maximum}, "
 582:                 f"but metrics.csv defines internal_max {metric_maximum}"
 583:             )
