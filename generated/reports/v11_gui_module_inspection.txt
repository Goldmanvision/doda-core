 272:     def update(self) -> None:
 273:         selected = self.editing_id
 274: 
 275:         self.category_combo.configure(values=self.categories())
 276: 
 277:         for item in self.tree.get_children():
 278:             self.tree.delete(item)
 279: 
 280:         rows = sorted(
 281:             self.rows(),
 282:             key=lambda row: (
 283:                 row.get("category", ""),
 284:                 row.get("flavor_text_id", ""),
 285:             ),
 286:         )
 287: 
 288:         for row in rows:
 289:             flavor_id = row.get("flavor_text_id", "")
 290:             title = " ".join(row.get("title", "").split())
 291: 
 292:             if len(title) > 70:
 293:                 title = title[:67] + "..."
 294: 
 295:             self.tree.insert(
 296:                 "",
 297:                 tk.END,
 298:                 iid=flavor_id,
 299:                 values=(
 300:                     flavor_id,
 301:                     row.get("category", ""),
 302:                     title,
 303:                     row.get("active", ""),
 304:                 ),
 305:             )
 306: 
 307:         if selected and self.tree.exists(selected):
 308:             self.tree.selection_set(selected)
 309:             self.tree.focus(selected)

 484:     def save(self) -> None:
 485:         row = self.record()
 486: 
 487:         if row is None:
 488:             return
 489: 
 490:         rows = self.rows()
 491: 
 492:         if self.editing_id:
 493:             if row["flavor_text_id"] != self.editing_id:
 494:                 messagebox.showerror(
 495:                     "DoDA Content Workbench",
 496:                     "Flavor Text ID cannot change after creation.",
 497:                 )
 498:                 return
 499: 
 500:             rows = [
 501:                 row
 502:                 if item.get("flavor_text_id") == self.editing_id
 503:                 else item
 504:                 for item in rows
 505:             ]
 506: 
 507:             action = "updated"
 508: 
 509:         else:
 510:             if any(
 511:                 item.get("flavor_text_id") == row["flavor_text_id"]
 512:                 for item in rows
 513:             ):
 514:                 messagebox.showerror(
 515:                     "DoDA Content Workbench",
 516:                     "Flavor Text ID already exists.",
 517:                 )
 518:                 return
 519: 
 520:             rows.append(row)
 521:             action = "added"
 522: 
 523:         temporary = self.app.source_directory / (
 524:             f".flavor-text-{uuid4().hex}.tmp.csv"
 525:         )
 526: 
 527:         self.app.config(cursor="watch")
 528:         self.app.update_idletasks()
 529: 
 530:         try:
 531:             self.write(temporary, rows)
 532: 
 533:             valid, output = self.candidate_valid(temporary)
 534: 
 535:             self.app.log("Flavor Text candidate validation output:")
 536: 
 537:             for line in output.splitlines():
 538:                 self.app.log(" " + line)
 539: 
 540:             if not valid:
 541:                 messagebox.showerror(
 542:                     "DoDA Content Workbench",
 543:                     (
 544:                         "Flavor Text candidate validation failed. Source CSV "
 545:                         "was unchanged.\n\nReview Activity Log for details."
 546:                     ),
 547:                 )
 548:                 return
 549: 
 550:             folder = self.backup_directory()
 551:             folder.mkdir(parents=True, exist_ok=True)
 552: 
 553:             backup = folder / (

 555:                 + datetime.now().strftime("%Y%m%d_%H%M%S_%f")
 556:                 + ".csv"
 557:             )
 558: 
 559:             shutil.copy2(self.source_path(), backup)
 560:             os.replace(temporary, self.source_path())
 561: 
 562:             self.editing_id = row["flavor_text_id"]
 563: 
 564:             self.app.log(f"Created Flavor Text backup: {backup}")
 565:             self.app.log(
 566:                 f"Flavor Text {action}: {row['flavor_text_id']}"
 567:             )

 569:             self.app.refresh_all(False)
 570: 
 571:             valid, output = self.post_validate()
 572: 
 573:             for line in output.splitlines():

 575: 
 576:             self.app.refresh_all(False)
 577: 
 578:             if valid:
 579:                 messagebox.showinfo(
 580:                     "DoDA Content Workbench",

 584:                         "Project validation passed."
 585:                     ),
 586:                 )
 587:             else:
 588:                 messagebox.showwarning(
