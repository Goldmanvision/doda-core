 774:     def save(self) -> None:
 775:         row = self.record()
 776: 
 777:         if row is None:
 778:             return
 779: 
 780:         rows = self.rows()
 781: 
 782:         if self.editing_key:
 783:             if (
 784:                 row["metric_id"] != self.editing_key[0]
 785:                 or row["tier_id"] != self.editing_key[1]
 786:             ):
 787:                 messagebox.showerror(
 788:                     "DoDA Content Workbench",
 789:                     "Metric ID and Tier ID cannot change after creation.",
 790:                 )
 791:                 return

 793:             rows = [

 796:                     item.get("metric_id") == self.editing_key[0]

 798:                 )
 799:                 else item
 800:                 for item in rows
 801:             ]
 802: 
 803:             action = "updated"
 804: 
 805:         else:
 806:             if any(
 807:                 item.get("metric_id") == row["metric_id"]
 808:                 and item.get("tier_id") == row["tier_id"]
 809:                 for item in rows
 810:             ):
 811:                 messagebox.showerror(
 812:                     "DoDA Content Workbench",
 813:                     "This Metric ID and Tier ID combination already exists.",
 814:                 )
 815:                 return

 817:             rows.append(row)
 818:             action = "added"
 819: 
 820:         temporary = self.app.source_directory / (
 821:             f".metric-tiers-{uuid4().hex}.tmp.csv"
 822:         )
 823: 
 824:         self.app.config(cursor="watch")
 825:         self.app.update_idletasks()
 826: 
 827:         try:
 828:             self.write(temporary, rows)

 830:             valid, output = self.candidate_valid(temporary)

 833:

 840:                     (
