#!/usr/bin/env python3
r"""
DoDA Content Workbench - content exporter and PK3 builder.

This script performs a safe, data-only export pipeline:

1. Runs the CSV validator.
2. Loads validated authoring CSV files.
3. Creates normalized JSON content data.
4. Creates a future UZDoom LANGUAGE fragment.
5. Creates a source/output hash manifest.
6. Builds a standalone PK3 overlay archive.

This version deliberately does not edit MAPINFO, GAMEINFO, LANGUAGE, zscript.txt,
maps, native C++, ZScript, or the doda-core package.

Usage:
    python tools/doda_content_workbench/doda_content_export.py
    python tools/doda_content_workbench/doda_content_export.py --root C:/Dev/uzdoom_src/doda-core
"""

def find_project_root(start: Path) -> Path:
    current = start.resolve()

    for candidate in (current, *current.parents):
        if (
            (candidate / "data_authoring").is_dir()
            and (candidate / "tools").is_dir()
            and (candidate / "DoDA").is_dir()
        ):
            return candidate

    raise FileNotFoundError(
        "Unable to locate the DoDA project root. Use --root to provide it explicitly."
    )


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()

    with path.open("rb") as source_file:
        for chunk in iter(lambda: source_file.read(65536), b""):
            digest.update(chunk)

    return digest.hexdigest()


def run_validator(project_root: Path) -> None:
    validator_path = (
        project_root
        / "tools"
        / "doda_content_workbench"
        / "doda_content_workbench.py"
    )

    if not validator_path.is_file():
        raise FileNotFoundError(
            f"Validator script was not found: {validator_path}"
        )

    command = [
        sys.executable,
        str(validator_path),
        "--root",
        str(project_root),
        "validate",
    ]

    print("Running source-data validation...")
    completed = subprocess.run(command, cwd=project_root)

    if completed.returncode != 0:
        raise RuntimeError(
            "Source-data validation failed. Export was not attempted. "
            "Correct CSV source data first, then run export again."
        )


def normalize_value(table_name: str, column_name: str, value: str) -> Any:
    normalized = value.strip()

    if column_name in INTEGER_COLUMNS.get(table_name, set()):
        return int(normalized)

    if column_name in BOOLEAN_COLUMNS.get(table_name, set()):
        return normalized.upper() == "TRUE"

    return normalized


def load_table(
    project_root: Path,
    table_name: str,
    filename: str,
) -> list[dict[str, Any]]:
    path = project_root / "data_authoring" / "source" / filename

    with path.open("r", encoding="utf-8-sig", newline="") as source_file:
        reader = csv.DictReader(source_file)
        rows: list[dict[str, Any]] = []

        for raw_row in reader:
            if raw_row is None:
                continue

            if not any((value or "").strip() for value in raw_row.values()):
                continue

            row: dict[str, Any] = {}

            for column_name, raw_value in raw_row.items():
                if column_name is None:
                    continue

                row[column_name.strip()] = normalize_value(
                    table_name,
                    column_name.strip(),
                    raw_value or "",
                )

            rows.append(row)

        return rows


def escape_language_text(value: str) -> str:
    return (
        value.replace("\\", "\\\\")
        .replace('"', '\\"')
        .replace("\r\n", "\\n")
        .replace("\n", "\\n")
        .replace("\r", "\\n")
    )


def language_key_from_flavor_id(flavor_text_id: str) -> str:
    safe_key = re.sub(r"[^A-Za-z0-9_]", "_", flavor_text_id.upper())
    return f"DODA_{safe_key}"


def build_language_fragment(flavor_rows: list[dict[str, Any]]) -> str:
    lines = [
        "// GENERATED FILE - DO NOT EDIT BY HAND",
        "// Source: data_authoring/source/flavor_text.csv",
        "// Generated by DoDA Content Workbench.",
        "",
        "[default]",
        "",
    ]

    for row in sorted(flavor_rows, key=lambda item: item["flavor_text_id"]):
        base_key = language_key_from_flavor_id(str(row["flavor_text_id"]))
        title = escape_language_text(str(row["title"]))
        body = escape_language_text(str(row["body"]))

        lines.append(f'{base_key}_TITLE = "{title}";')
        lines.append(f'{base_key}_BODY = "{body}";')
        lines.append("")

    return "\n".join(lines).rstrip() + "\n"


def write_json(path: Path, payload: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps(payload, indent=2, ensure_ascii=False) + "\n",
        encoding="utf-8",
    )


def write_text(path: Path, content: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(content, encoding="utf-8")


def build_pk3(
    pk3_path: Path,
    department_json_path: Path,
    language_path: Path,
    manifest_path: Path,
) -> list[str]:
    pk3_path.parent.mkdir(parents=True, exist_ok=True)

    archive_entries = {
        "data/department_content.json": department_json_path,
        "language/DODA_GENERATED_LANGUAGE.txt": language_path,
        "manifest/content_manifest.json": manifest_path,
    }

    package_readme = (
        "DoDA Generated Content Overlay\n"
        "==============================\n"
        "\n"
        "This PK3 was generated by DoDA Content Workbench.\n"
        "\n"
        "It is currently a data-packaging proof. The archive contains generated\n"
        "department data, a future LANGUAGE fragment, and a build manifest.\n"
        "It does not yet add active runtime behavior because doda-core has not\n"
        "yet been given a native data loader or static include seam for these\n"
        "generated resources.\n"
    )

    temporary_path = pk3_path.with_suffix(".pk3.tmp")

    if temporary_path.exists():
        temporary_path.unlink()

    with zipfile.ZipFile(
        temporary_path,
        mode="w",
        compression=zipfile.ZIP_DEFLATED,
        compresslevel=9,
    ) as archive:
        archive.writestr("README.txt", package_readme)

        for archive_name, source_path in archive_entries.items():
            archive.write(source_path, archive_name)

    if pk3_path.exists():
        pk3_path.unlink()

    temporary_path.replace(pk3_path)

    with zipfile.ZipFile(pk3_path, mode="r") as archive:
        return sorted(archive.namelist())


def export_content(project_root: Path) -> tuple[dict[str, Path], list[str]]:
    run_validator(project_root)

    print()
    print("Loading validated authoring CSV files...")

    tables: dict[str, list[dict[str, Any]]] = {}

    for table_name, filename in TABLE_FILES.items():
        table_rows = load_table(project_root, table_name, filename)
        tables[table_name] = table_rows
        print(f"  {table_name}: {len(table_rows)} row(s)")

    generated_at = datetime.now(timezone.utc).isoformat()

    department_payload = {
        "schemaVersion": SCHEMA_VERSION,
        "toolVersion": TOOL_VERSION,
        "generatedAtUtc": generated_at,
        "tables": tables,
    }

    generated_directory = project_root / "generated"
    department_json_path = generated_directory / "data" / "department_content.json"
    language_path = (
        generated_directory
        / "language"
        / "DODA_GENERATED_LANGUAGE.txt"
    )
    manifest_path = (
        generated_directory
        / "manifest"
        / "content_manifest.json"
    )
    pk3_path = project_root / "build" / "doda-generated-content.pk3"

    write_json(department_json_path, department_payload)

    language_fragment = build_language_fragment(tables["flavor_text"])
    write_text(language_path, language_fragment)

    source_hashes = {
        filename: sha256_file(
            project_root / "data_authoring" / "source" / filename
        )
        for filename in TABLE_FILES.values()
    }

    output_hashes = {
        "generated/data/department_content.json": sha256_file(department_json_path),
        "generated/language/DODA_GENERATED_LANGUAGE.txt": sha256_file(language_path),
    }

    manifest_payload = {
        "schemaVersion": SCHEMA_VERSION,
        "toolVersion": TOOL_VERSION,
        "generatedAtUtc": generated_at,
        "projectRoot": str(project_root),
        "sourceFiles": source_hashes,
        "outputFiles": output_hashes,
        "tableRowCounts": {
            table_name: len(rows)
            for table_name, rows in tables.items()
        },
        "runtimeStatus": {
            "nativeLoaderAvailable": False,
            "mapinfoIntegrationEnabled": False,
            "languageIntegrationEnabled": False,
            "zscriptIntegrationEnabled": False,
            "notes": [
                "This is a generated content overlay proof.",
                "The PK3 is not yet active runtime data.",
                "Do not manually edit generated output files.",
            ],
        },
    }

    write_json(manifest_path, manifest_payload)

    archive_entries = build_pk3(
        pk3_path,
        department_json_path,
        language_path,
        manifest_path,
    )

    outputs = {
        "department_json": department_json_path,
        "language_fragment": language_path,
        "manifest": manifest_path,
        "pk3_overlay": pk3_path,
    }

    return outputs, archive_entries


def print_export_result(
    project_root: Path,
    outputs: dict[str, Path],
    archive_entries: list[str],
) -> None:
    print()
    print("=" * 72)
    print("DoDA Content Workbench Export")
    print("=" * 72)
    print(f"Project root: {project_root}")
    print()

    print("Generated outputs:")
    for output_name, output_path in outputs.items():
        print(f"  {output_name}: {output_path}")

    print()
    print("PK3 archive entries:")
    for entry in archive_entries:
        print(f"  {entry}")

    print()
    print(
        "Export completed successfully. This PK3 is a safe data overlay proof "
        "and does not yet alter live UZDoom behavior."
    )
    print("=" * 72)


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="doda_content_export",
        description="Export validated DoDA authoring content and build a PK3 overlay.",
    )

    parser.add_argument(
        "--root",
        type=Path,
        default=None,
        help="Path to doda-core. Defaults to automatic project-root discovery.",
    )

    return parser


def main() -> int:
    parser = build_parser()
    arguments = parser.parse_args()

    try:
        project_root = (
            arguments.root.resolve()
            if arguments.root is not None
            else find_project_root(Path.cwd())
        )

        outputs, archive_entries = export_content(project_root)
        print_export_result(project_root, outputs, archive_entries)
        return 0

    except (
        FileNotFoundError,
        RuntimeError,
        ValueError,
        OSError,
        zipfile.BadZipFile,
    ) as error:
        print()
        print(f"EXPORT FAILED: {error}", file=sys.stderr)
        return 1


if __name__ == "__main__":
    sys.exit(main())

def enable_windows_dark_title_bar(window: tk.Tk) -> None:
    """Request a dark native Windows title bar for the top-level app frame."""

    if sys.platform != "win32":
        return

    try:
        import ctypes

        window.update_idletasks()

        user32 = ctypes.windll.user32
        dwm_api = ctypes.windll.dwmapi.DwmSetWindowAttribute

        user32.GetAncestor.argtypes = [
            ctypes.c_void_p,
            ctypes.c_uint,
        ]
        user32.GetAncestor.restype = ctypes.c_void_p

        dwm_api.argtypes = [
            ctypes.c_void_p,
            ctypes.c_uint,
            ctypes.c_void_p,
            ctypes.c_uint,
        ]
        dwm_api.restype = ctypes.c_long

        # Tk exposes a client-area handle. DWM title-bar settings must target
        # the top-level frame window instead.
        GA_ROOT = 2
        client_handle = window.winfo_id()
        top_level_handle = user32.GetAncestor(
            ctypes.c_void_p(client_handle),
            GA_ROOT,
        )

        hwnd = top_level_handle or client_handle
        enabled = ctypes.c_int(1)

        # Attribute 20 is used by current Windows 10/11 releases.
        result = dwm_api(
            ctypes.c_void_p(hwnd),
            20,
            ctypes.byref(enabled),
            ctypes.sizeof(enabled),
        )

        if result != 0:
            # Attribute 19 is the Windows 10 1809â€“1909 compatibility value.
            dwm_api(
                ctypes.c_void_p(hwnd),
                19,
                ctypes.byref(enabled),
                ctypes.sizeof(enabled),
            )

    except (AttributeError, OSError, tk.TclError):
        # Unsupported Windows versions retain their standard system title bar.
        pass

def read_csv(path: Path) -> list[dict[str, str]]:
    if not path.is_file():
        return []

    with path.open("r", encoding="utf-8-sig", newline="") as source_file:
        reader = csv.DictReader(source_file)
        rows: list[dict[str, str]] = []

        for raw_row in reader:
            if raw_row is None:
                continue

            row = {
                (key or "").strip(): (value or "").strip()
                for key, value in raw_row.items()
                if key is not None
            }

            if any(row.values()):
                rows.append(row)

        return rows


def truncate_text(value: str, maximum_length: int = 120) -> str:
    normalized = " ".join((value or "").split())

    if len(normalized) <= maximum_length:
        return normalized

    return normalized[: maximum_length - 3] + "..."

    def open_foims_preview(self) -> None:
        """Open the singleton FOIMS-X preview monitor or raise it if already open."""

        if (
            self.foims_preview is not None and
            self.foims_preview.is_open()
        ):
            self.foims_preview.show()
            self.foims_preview.refresh_preview()
            return

        self.foims_preview = FoimsXPreview(self)
        self.log("Opened FOIMS-X Preview Monitor.")

    def refresh_foims_preview(self) -> None:
        """Refresh an open FOIMS-X monitor without creating a new window."""

        if (
            self.foims_preview is None or
            not self.foims_preview.is_open()
        ):
            self.open_foims_preview()
            return

        self.foims_preview.refresh_preview()
        self.foims_preview.show()
        self.log("Refreshed FOIMS-X Preview Monitor.")

    def toggle_fullscreen(
        self,
        _event: Any | None = None,
    ) -> str:
        """Toggle the main Content Workbench window between normal and full screen."""
        is_fullscreen = bool(self.attributes("-fullscreen"))
        self.attributes("-fullscreen", not is_fullscreen)
        return "break"

    def open_foims_preview_from_shortcut(
        self,
        _event: Any | None = None,
    ) -> str:
        """Open FOIMS-X from Ctrl+O while consuming the Tk key event."""
        self.open_foims_preview()
        return "break"
    def _show_client_menu(
        self,
        event: tk.Event[Any],
        menu: tk.Menu,
    ) -> str:
        """Post a client-area menu directly below the clicked menu button."""

        try:
            menu.tk_popup(
                event.x_root,
                event.y_root + event.widget.winfo_height(),
            )
        finally:
            menu.grab_release()

        return "break"

    def _build_client_menu_bar(self, parent: tk.Widget) -> None:
        """Render a dark, fully functional menu row inside the client area."""

        menu_row = tk.Frame(
            parent,
            background="#161b22",
            highlightthickness=0,
            borderwidth=0,
        )
        menu_row.pack(fill=tk.X, pady=(0, 10))

        menu_items = (
            ("File", self.file_menu),
            ("Actions", self.actions_menu),
            ("Preview", self.preview_menu),
            ("View", self.view_menu),
            ("Folders", self.folders_menu),
            ("Help", self.help_menu),
        )

        for label, menu in menu_items:
            button = tk.Menubutton(
                menu_row,
                text=label,
                menu=menu,
                background="#161b22",
                foreground="#e6edf3",
                activebackground="#264f78",
                activeforeground="#ffffff",
                disabledforeground="#6e7681",
                borderwidth=0,
                relief=tk.FLAT,
                highlightthickness=0,
                padx=10,
                pady=5,
                anchor=tk.W,
                cursor="hand2",
                font=("Segoe UI", 9),
                takefocus=True,
            )

            button.pack(side=tk.LEFT)

            button.bind(
                "<Button-1>",
                lambda event, target_menu=menu: self._show_client_menu(
                    event,
                    target_menu,
                ),
            )

            button.bind(
                "<Return>",
                lambda event, target_menu=menu: self._show_client_menu(
                    event,
                    target_menu,
                ),
            )

            button.bind(
                "<space>",
                lambda event, target_menu=menu: self._show_client_menu(
                    event,
                    target_menu,
                ),
            )

            button.bind(
                "<Enter>",
                lambda event, target=button: target.configure(
                    background="#264f78",
                    foreground="#ffffff",
                ),
            )

            button.bind(
                "<Leave>",
                lambda event, target=button: target.configure(
                    background="#161b22",
                    foreground="#e6edf3",
                ),
            )
    def select_relative_notebook_tab(self, offset: int) -> str:
        tab_ids = self.notebook.tabs()

        if not tab_ids:
            return "break"

        try:
            current_index = self.notebook.index(self.notebook.select())
        except tk.TclError:
            current_index = 0

        target_index = (current_index + offset) % len(tab_ids)

        self.notebook.select(tab_ids[target_index])
        self.notebook.focus_set()

        return "break"

    def select_previous_notebook_tab(
        self,
        _event: Any = None,
    ) -> str:
        return self.select_relative_notebook_tab(-1)

    def select_next_notebook_tab(
        self,
        _event: Any = None,
    ) -> str:
        return self.select_relative_notebook_tab(1)

    def _build_browser(
        self,
        parent: ttk.Frame,
        title: str,
        columns: tuple[tuple[str, str, int], ...],
        select_handler: Any,
    ) -> tuple[ttk.Treeview, tk.Text]:
        parent.columnconfigure(0, weight=3)
        parent.columnconfigure(1, weight=2)
        parent.rowconfigure(0, weight=1)

        list_frame = ttk.LabelFrame(parent, text=title, padding=8)
        detail_frame = ttk.LabelFrame(parent, text="Record Detail", padding=8)

        list_frame.grid(row=0, column=0, sticky=tk.NSEW, padx=(0, 6))
        detail_frame.grid(row=0, column=1, sticky=tk.NSEW)

        list_frame.columnconfigure(0, weight=1)
        list_frame.rowconfigure(0, weight=1)

        column_ids = tuple(column[0] for column in columns)

        tree = ttk.Treeview(
            list_frame,
            columns=column_ids,
            show="headings",
            selectmode="browse",
        )

        for column_id, heading, width in columns:
            tree.heading(column_id, text=heading)
            tree.column(column_id, width=width, anchor=tk.W)

        scrollbar = ttk.Scrollbar(
            list_frame,
            orient=tk.VERTICAL,
            command=tree.yview,
        )
        tree.configure(yscrollcommand=scrollbar.set)

        tree.grid(row=0, column=0, sticky=tk.NSEW)
        scrollbar.grid(row=0, column=1, sticky=tk.NS)

        detail = tk.Text(
            detail_frame,
            wrap=tk.WORD,
            borderwidth=0,
            highlightthickness=0,
            background="#0d1117",
            foreground="#e6edf3",
            insertbackground="#e6edf3",
            font=("Consolas", 10),
        )

        detail_scrollbar = ttk.Scrollbar(
            detail_frame,
            orient=tk.VERTICAL,
            command=detail.yview,
        )
        detail.configure(yscrollcommand=detail_scrollbar.set)

        detail.pack(side=tk.LEFT, fill=tk.BOTH, expand=True)
        detail_scrollbar.pack(side=tk.RIGHT, fill=tk.Y)

        detail.configure(state=tk.DISABLED)

        tree.bind("<<TreeviewSelect>>", select_handler)

        return tree, detail

    def run_network_rules(self, mode: str, announce: bool) -> bool:

        announce=False only raises a dialog on failure (used after the main

        """
        dirty = network_editors_dirty(self)
        if dirty:
            self.log("Network rules note: unsaved drafts are not included: " + ", ".join(dirty))

                    result.errors += condition_result.errors
                    result.warnings += condition_result.warnings
                    if condition_path is not None:
                        self.log(f"Personnel conditions catalog exported: {condition_path}")
                    attribute_result, attribute_path = doda_agent_attribute_rules.export(self.project_root)
                    result.errors += attribute_result.errors
                    result.warnings += attribute_result.warnings
                    if attribute_path is not None:

                    pcare_result, pcare_path = doda_personal_care_rules.export(self.project_root)

                    if rc_path is not None:

                    sobriety_result, sobriety_path = doda_sobriety_rules.export(self.project_root)
                    result.errors += sobriety_result.errors
                    result.warnings += sobriety_result.warnings
                    if sobriety_path is not None:
                        self.log(f"Sobriety catalog exported: {sobriety_path}")
                    territory_result, territory_path = doda_territory_rules.export(self.project_root)
                    result.errors += territory_result.errors
                    result.warnings += territory_result.warnings

    def run_command(
        self,
        label: str,
        command: list[str],
        success_message: str,
        announce_success: bool = True,
    ) -> bool:
        self.config(cursor="watch")
        self.update_idletasks()

        self.log(f"Starting {label.lower()}: {' '.join(command)}")

        try:
            completed = subprocess.run(
                command,
                cwd=self.project_root,
                text=True,
                encoding="utf-8",
                errors="replace",
                capture_output=True,
            )

            combined_output = (
                completed.stdout.rstrip()
                + ("\n" if completed.stdout and completed.stderr else "")
                + completed.stderr.rstrip()
            ).strip()

            if combined_output:
                self.log(f"{label} output:")
                for line in combined_output.splitlines():
                    self.log(f"  {line}")

            self.log(f"{label} process exit code: {completed.returncode}")

            self.refresh_all(show_message=False)

            if completed.returncode == 0:
                self.status_text.set(success_message)
                if announce_success:
                    messagebox.showinfo(APP_TITLE, success_message)
                return True
            else:
                self.status_text.set(f"{label} failed.")
                messagebox.showerror(
                    APP_TITLE,
                    f"{label} failed with exit code {completed.returncode}.\n\n"
                    "Review the Activity Log tab for complete output.",
                )
                return False

        except OSError as error:
            self.status_text.set(f"{label} could not start.")
            self.log(f"{label} launch error: {error}")
            messagebox.showerror(
                APP_TITLE,
                f"Could not start {label.lower()}.\n\n{error}",
            )
            return False
        finally:
            self.config(cursor="")
            self.update_idletasks()

@dataclass(frozen=True)
class FieldSpec:
    name: str
    label: str
    kind: str = "text"            # text | int | bool | choice | ref
    required: bool = True
    choices: tuple[str, ...] = ()
    ref_table: str = ""           # for kind == "ref": table whose key supplies choices
    multiline: bool = False


@dataclass(frozen=True)
class TableSpec:
    table: str
    filename: str
    key: str
    fields: tuple[FieldSpec, ...] = field(default_factory=tuple)

    @property
    def columns(self) -> list[str]:
        return [f.name for f in self.fields]

    @property
    def key_columns(self) -> tuple[str, ...]:
        """Composite keys are written as 'a+b'."""
        return tuple(self.key.split("+"))

def page_for_tab_text(text: str) -> PageSpec | None:
    """Longest matching prefix wins, so 'Facility Capabilities' beats 'Facilities'."""
    best: PageSpec | None = None
    for page in PAGES:
        if text.startswith(page.tab_text_prefix):
            if best is None or len(page.tab_text_prefix) > len(best.tab_text_prefix):
                best = page
    return best

"""
DoDA Content Workbench - grouped, resizable, collapsible navigation sidebar.

Builds a grouped tree from the notebook's actual tabs (so no editor is
reinstalled or rebuilt) using doda_page_registry for grouping and labels.
Tabs not in the registry appear under "Other". The notebook tab strip is
hidden while the sidebar is shown; View > Navigation can restore it.
Switching pages never destroys page widgets, so unsaved forms are preserved.
"""

from __future__ import annotations

import tkinter as tk
from tkinter import ttk
from typing import Any

from doda_page_registry import GROUP_ORDER, page_for_tab_text

MIN_WIDTH, MAX_WIDTH, DEFAULT_WIDTH = 160, 480, 240
TAB_BG, TAB_FG = "#1c2430", "#e6edf3"
TAB_HOVER = "#2a3544"
TAB_SEL_BG, TAB_SEL_FG = "#0d1117", "#58a6ff"


class GroupedNavigation:
    def __init__(self, app: Any) -> None:
        self.app = app
        self.nb: ttk.Notebook = app.notebook
        self.width = DEFAULT_WIDTH
        self.visible = True
        self.tab_strip_hidden = False
        self.item_to_tab: dict[str, str] = {}
        self.tab_to_item: dict[str, str] = {}
        self.groups: dict[str, list[tuple[str, str]]] = {}
        self.subgroup_of: dict[str, str] = {}

        style = ttk.Style(app)
        self._orig_layout = style.layout("TNotebook.Tab")
        style.layout("Grouped.TNotebook.Tab", [])
        style.configure("Grouped.TNotebook", tabmargins=0)

        self.container = ttk.Frame(self.nb.master, width=self.width)
        self.container.pack_propagate(False)
        self.container.pack(side=tk.LEFT, fill=tk.Y, before=self.nb, padx=(0, 0))
        self.handle = tk.Frame(self.nb.master, width=5, cursor="sb_h_double_arrow", background="#30363d")
        self.handle.pack(side=tk.LEFT, fill=tk.Y, before=self.nb, padx=(2, 8))
        self.handle.bind("<B1-Motion>", self._drag)
        self.handle.bind("<Enter>", lambda _e: self.handle.configure(background="#58a6ff"))
        self.handle.bind("<Leave>", lambda _e: self.handle.configure(background="#30363d"))

        bar = ttk.Frame(self.container)

            return

        total = max(1, self.tab_inner.winfo_reqwidth())
        view = self.tab_canvas.winfo_width()
        left = self.tab_canvas.canvasx(0)

            self.tab_canvas.xview_moveto(max(0.0, (x0 - 8) / total))
        elif x1 > left + view:
            self.tab_canvas.xview_moveto(min(1.0, (x1 + 8 - view) / total))

    def _tab_wheel(self, event: Any) -> str:
        self.tab_canvas.xview_scroll(-3 if event.delta > 0 else 3, "units")
        return "break"

        self.nb.select(tab_id)

                         selectcolor=opts["foreground"], **opts)

    def candidate_valid(self, candidate: Path) -> tuple[bool, str]:
        with tempfile.TemporaryDirectory(
            prefix="doda-qualification-"
        ) as temporary:
            root = Path(temporary)
            destination = root / "data_authoring" / "source"
            destination.mkdir(parents=True)

            for filename in FILES:
                source = (
                    candidate
                    if filename == "qualification_catalog.csv"
                    else self.app.source_directory / filename

                shutil.copy2(source, destination / filename)

            command = [
                sys.executable,
                str(self.app.validator_path),
                "--root",
                str(root),
                "validate",
            ]

            completed = subprocess.run(
                command,
                cwd=root,
                text=True,
                encoding="utf-8",
                errors="replace",
                capture_output=True,
            )

            output = (
                completed.stdout
                + ("\n" if completed.stdout and completed.stderr else "")
                + completed.stderr
            ).strip()

            return completed.returncode == 0, output

    def post_validate(self) -> tuple[bool, str]:
        command = [
            sys.executable,
            str(self.app.validator_path),
            "--root",
            str(self.app.project_root),
            "validate",
        ]

        completed = subprocess.run(
            command,
            cwd=self.app.project_root,
            text=True,
            encoding="utf-8",
            errors="replace",
            capture_output=True,
        )

        output = (
            completed.stdout
            + ("\n" if completed.stdout and completed.stderr else "")
            + completed.stderr
        ).strip()

        return completed.returncode == 0, output

    def save(self) -> None:
        row = self.record()

        if row is None:
            return

        rows = self.rows()

        if self.editing_id:
            if row["qualification_id"] != self.editing_id:
                messagebox.showerror(
                    "DoDA Content Workbench",
                    "Qualification ID cannot change after creation.",
                )
                return

            rows = [
                row
                if item.get("qualification_id") == self.editing_id
                else item
                for item in rows
            ]
            action = "updated"
        else:
            if any(
                item.get("qualification_id") == row["qualification_id"]
                for item in rows
            ):
                messagebox.showerror(
                    "DoDA Content Workbench",
                    "Qualification ID already exists.",
                )
                return

            rows.append(row)
            action = "added"

        rows.sort(
            key=lambda item: (
                int(item.get("sort_order", "999999")),
                item.get("qualification_id", ""),
            )
        )

        temporary = self.app.source_directory / (
            f".qualification-catalog-{uuid4().hex}.tmp.csv"
        )

        self.app.config(cursor="watch")
        self.app.update_idletasks()

        try:
            self.write(temporary, rows)

            valid, output = self.candidate_valid(temporary)

            self.app.log("Qualification candidate validation output:")

            for line in output.splitlines():
                self.app.log(" " + line)

            if not valid:
                messagebox.showerror(
                    "DoDA Content Workbench",
                    (
                        "Qualification candidate validation failed. Source CSV "
                        "was unchanged.\n\nReview Activity Log for details."
                    ),
                )
                return

            folder = self.backup_directory()
            folder.mkdir(parents=True, exist_ok=True)

            backup = folder / (
                "qualification_catalog_"
                + datetime.now().strftime("%Y%m%d_%H%M%S_%f")
                + ".csv"
            )

            shutil.copy2(self.source_path(), backup)
            os.replace(temporary, self.source_path())

            self.editing_id = row["qualification_id"]

            self.app.log(f"Created qualification backup: {backup}")
            self.app.log(
                f"Qualification {action}: {row['qualification_id']}"
            )

            self.app.refresh_all(False)

            valid, output = self.post_validate()

            for line in output.splitlines():
                self.app.log(" " + line)

            self.app.refresh_all(False)

            if valid:
                messagebox.showinfo(
                    "DoDA Content Workbench",
                    (
                        f"Qualification '{row['qualification_id']}' {action}.\n\n"
                        f"Backup created:\n{backup}\n\n"
                        "Project validation passed."
                    ),
                )
            else:
                messagebox.showwarning(
                    "DoDA Content Workbench",
                    (
                        f"Qualification {action}, but post-save validation failed.\n\n"
                        f"Backup:\n{backup}"
                    ),
                )

        except (OSError, ValueError) as error:
            self.app.log(f"QUALIFICATION SAVE ERROR: {error}")
            messagebox.showerror(
                "DoDA Content Workbench",
                f"Unable to save qualification.\n\n{error}",
            )
        finally:
            if temporary.exists():
                temporary.unlink()

            self.app.config(cursor="")
            self.app.update_idletasks()

    def remove(self) -> None:
        if not self.editing_id:
            messagebox.showwarning(
                "DoDA Content Workbench",
                "Select an existing qualification before using Remove.",
            )
            return

        qualification_id = self.editing_id

        confirmed = messagebox.askyesno(
            "Remove Qualification?",
            (
                f"ARE YOU SURE you want to remove '{qualification_id}'?\n\n"
                "Removal will be blocked if the qualification is referenced "
                "by staffing rules or overrides."
            ),
            icon="warning",
            default=messagebox.NO,
        )

        if not confirmed:
            self.app.log(
                f"Remove Qualification cancelled for '{qualification_id}'."
            )
            return

        rows = [
            row
            for row in self.rows()
            if row.get("qualification_id") != qualification_id
        ]

        temporary = self.app.source_directory / (
            f".qualification-catalog-remove-{uuid4().hex}.tmp.csv"
        )

        self.app.config(cursor="watch")
        self.app.update_idletasks()

        try:
            self.write(temporary, rows)

            valid, output = self.candidate_valid(temporary)

            self.app.log(
                "Qualification removal candidate validation output "
                f"for '{qualification_id}':"
            )

            for line in output.splitlines():
                self.app.log(" " + line)

            if not valid:
                messagebox.showerror(
                    "DoDA Content Workbench",
                    (
                        "Qualification removal was blocked by validation.\n\n"
                        "Source CSV was unchanged."
                    ),
                )
                return

            folder = self.backup_directory()
            folder.mkdir(parents=True, exist_ok=True)

            backup = folder / (
                "qualification_catalog_before_remove_"
                + datetime.now().strftime("%Y%m%d_%H%M%S_%f")
                + ".csv"
            )

            shutil.copy2(self.source_path(), backup)
            os.replace(temporary, self.source_path())

            self.editing_id = None

            self.app.log(
                f"Created pre-removal qualification backup: {backup}"
            )
            self.app.log(f"Removed qualification: {qualification_id}")

            self.app.refresh_all(False)
            self.new()

            valid, output = self.post_validate()

            for line in output.splitlines():
                self.app.log(" " + line)

            self.app.refresh_all(False)

            if valid:
                messagebox.showinfo(
                    "DoDA Content Workbench",
                    (
                        f"Qualification '{qualification_id}' removed.\n\n"
                        f"Backup created:\n{backup}\n\n"
                        "Project validation passed."
                    ),
                )
            else:
                messagebox.showwarning(
                    "DoDA Content Workbench",
                    (
                        f"Qualification '{qualification_id}' removed, but "
                        f"post-save validation failed.\n\nBackup:\n{backup}"
                    ),
                )

        except (OSError, ValueError) as error:
            self.app.log(f"QUALIFICATION REMOVE ERROR: {error}")
            messagebox.showerror(
                "DoDA Content Workbench",
                f"Unable to remove qualification.\n\n{error}",
            )
        finally:
            if temporary.exists():
                temporary.unlink()

            self.app.config(cursor="")
            self.app.update_idletasks()

    def open_backups(self) -> None:
        self.backup_directory().mkdir(parents=True, exist_ok=True)
        self.app.open_folder(self.backup_directory())


def install_qualification_catalog_editor(app: Any) -> None:
    editor = QualificationCatalogEditor(app)
    app.qualification_catalog_editor = editor
