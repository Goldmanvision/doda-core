class TaskTemplatesEditor:
    def __init__(self, app):
        self.app = app
        self.editing_id = None
        self.snapshot_hash = None
        self.loaded_rows = []
        self.vars = {
            field: tk.StringVar(master=app)
            for field in FIELDS
            if field != "description"
        }
        self.widgets = {}
        self.build()

    def source_path(self):
        return self.app.source_directory / "task_templates.csv"

    def backup_directory(self):
        return (
            self.app.project_root / "data_authoring"
            / "backups" / "task_templates"
        )

    @staticmethod
    def digest(path):
        return hashlib.sha256(path.read_bytes()).hexdigest()

    def choices(self, field):
        if field == "capability_id":
            return [""] + sorted({r["capability_id"] for r in self.app.tables.get("capabilities", [])})
        if field == "priority":
            rows = [r for r in self.app.tables.get("enum_catalog", [])
                    if r.get("enum_group") == "TASK_PRIORITY" and r.get("active", "").upper() == "TRUE"]
            return [r["enum_value"] for r in sorted(rows, key=lambda r: int(r["sort_order"]))]
        if field == "target_type":
            return list(task_schema.TARGET_TYPES)
        if field == "reward_mode":
            return list(task_schema.REWARD_MODES)
        if field == "target_family_id":
            path = self.app.source_directory / "equipment_families.csv"
            if not path.is_file():
                return [""]
            with path.open(encoding="utf-8-sig", newline="") as handle:
                return [""] + sorted(r["family_id"] for r in csv.DictReader(handle)
                                     if r.get("ownership", "").upper() == "LOCATION")
        return []

    def build(self):
        tab = self.app.task_templates_tab
        for child in tab.winfo_children():
            child.destroy()
        tab.columnconfigure(0, weight=3)
        tab.columnconfigure(1, weight=2)
        tab.rowconfigure(0, weight=1)

        left = ttk.LabelFrame(tab, text="Task Templates", padding=8)
        right = ttk.LabelFrame(tab, text="Task Template Editor", padding=10)
        left.grid(row=0, column=0, sticky="nsew", padx=(0, 6))
        right.grid(row=0, column=1, sticky="nsew")
        left.columnconfigure(0, weight=1)
        left.rowconfigure(0, weight=1)
        right.columnconfigure(1, weight=1)

        columns = ("task_template_id", "display_name", "priority",
                   "target_type", "loop_task")
        self.tree = ttk.Treeview(
            left, columns=columns, show="headings", selectmode="extended"
        )
        for field, title in zip(
            columns, ("Template ID", "Name", "Priority", "Target", "Loop")
        ):
            self.tree.heading(field, text=title)
            self.tree.column(field, width=140, minwidth=60)
        self.tree.grid(row=0, column=0, sticky="nsew")
        vertical = ttk.Scrollbar(left, command=self.tree.yview)
        vertical.grid(row=0, column=1, sticky="ns")
        horizontal = ttk.Scrollbar(
            left, orient="horizontal", command=self.tree.xview
        )
        horizontal.grid(row=1, column=0, sticky="ew")
        self.tree.configure(
            yscrollcommand=vertical.set, xscrollcommand=horizontal.set
        )
        self.tree.bind("<<TreeviewSelect>>", self.on_select)

        labels = (
            "Template ID", "Display Name", "Description", "Capability",
            "Priority", "Target Type", "Station Family (optional draft)", "Loop Task",
            "Time To Complete (minutes; optional)", "Reward Mode", "Payout Count", "Sort Order", "Active",
        )
        for index, (field, label) in enumerate(zip(FIELDS, labels)):

        self.new()

    def update(self):
        path = self.source_path()
        before = self.digest(path)
        with path.open(encoding="utf-8-sig", newline="") as handle:
            rows = [
                dict(row) for row in csv.DictReader(handle)
                if any(row.values())
            ]
        if before != self.digest(path):
            raise ValueError("Task CSV changed during refresh. Refresh again.")
        self.loaded_rows = rows
        self.snapshot_hash = before
        for field in ("capability_id", "priority", "target_type", "target_family_id", "reward_mode"):
            self.widgets[field].configure(values=self.choices(field))
        for item in self.tree.get_children():
            self.tree.delete(item)
        for row in sorted(rows, key=lambda r: (
            int(r["sort_order"]), r["task_template_id"]
        )):
            self.tree.insert(
                "", "end", iid=row["task_template_id"],
                values=tuple(row[field] for field in self.tree["columns"])
            )
        if self.editing_id and self.tree.exists(self.editing_id):
            self.tree.selection_set(self.editing_id)

    def new(self):
        self._multi_selection = False
        set_form_state(self, False)
        self.tree.selection_remove(*self.tree.selection())
        self.editing_id = None
        self.widgets["task_template_id"].configure(state="normal")
        for variable in self.vars.values():
            variable.set("")
        self.description.delete("1.0", "end")
        self.vars["priority"].set("NORMAL" if "NORMAL" in self.choices("priority") else "")
        self.vars["target_type"].set("TERRITORY")
        self.vars["reward_mode"].set("CONTINUOUS")
        self.vars["loop_task"].set("FALSE")
        self.vars["payout_count"].set("0")
        self.vars["active"].set("FALSE")
        self.vars["sort_order"].set(str(max(
            (int(r["sort_order"]) for r in self.loaded_rows), default=0
        ) + 1))

    def on_select(self, _event=None):
        selected = self.tree.selection()
        if not selected:
            if getattr(self, "_multi_selection", False):
                self.new()
            return
        if len(selected) > 1:
            self._multi_selection = True
            self.editing_id = None
            selected_rows = [
                row for row in self.loaded_rows
                if row["task_template_id"] in selected
            ]
            set_form_state(self, False)
            for field, variable in self.vars.items():
                variable.set(selection_display(selected_rows, field))
            self.description.delete("1.0", "end")
            self.description.insert(
                "1.0", selection_display(selected_rows, "description")
            )
            set_form_state(self, True)
            return
        was_multiple = getattr(self, "_multi_selection", False)
        self._multi_selection = False
        set_form_state(self, False)
        if _event is not None and not was_multiple and selected[0] == self.editing_id:
            return
        row = next(
            (r for r in self.loaded_rows
             if r["task_template_id"] == selected[0]), None
        )
        if row is None:
            return
        self.editing_id = row["task_template_id"]
        for field, variable in self.vars.items():
            variable.set(row[field])
        self.description.delete("1.0", "end")
        self.description.insert("1.0", row["description"])
        self.widgets["task_template_id"].configure(state="disabled")

    def revert(self):
        if self.editing_id and self.tree.exists(self.editing_id):
            self.tree.selection_set(self.editing_id)
            self.on_select()
        else:
            self.new()

    def record(self):
        row = {field: var.get().strip() for field, var in self.vars.items()}
        row["description"] = self.description.get("1.0", "end").strip()
        if not re.fullmatch(r"[A-Z][A-Z0-9_]*", row["task_template_id"]):
            raise ValueError("Template ID must be uppercase letters/digits/underscores, starting with a letter.")
        problems = task_schema.errors(row, set(self.choices("priority")))
        if row["capability_id"] and row["capability_id"] not in self.choices("capability_id"):
            problems.append("unknown capability_id")
        if row["target_family_id"] not in self.choices("target_family_id"):
            problems.append("unknown LOCATION station family")
        if problems:
            raise ValueError("; ".join(problems))
        return row

    def candidate_valid(self, candidate):
        with tempfile.TemporaryDirectory(prefix="doda-task-") as temporary:
            root = Path(temporary)
            destination = root / "data_authoring" / "source"
            destination.mkdir(parents=True)
            for filename in FILES:
                shutil.copy2(
                    candidate if filename == "task_templates.csv"
                    else self.app.source_directory / filename,
                    destination / filename
                )
            done = subprocess.run(
                [sys.executable, str(self.app.validator_path),
                 "--root", str(root), "validate"],
                cwd=root, text=True, encoding="utf-8",
                errors="replace", capture_output=True
            )
            return done.returncode == 0, done.stdout + "\n" + done.stderr

    def commit(self, rows):
        path = self.source_path()
        if self.snapshot_hash is None or self.digest(path) != self.snapshot_hash:
            raise ValueError("Task CSV changed externally. Refresh before saving.")
        temporary = path.with_name(f".tasks-{uuid4().hex}.tmp.csv")
        try:
            with temporary.open("w", encoding="utf-8", newline="") as handle:
                writer = csv.DictWriter(
                    handle, fieldnames=FIELDS, lineterminator="\n"
                )
                writer.writeheader()
                writer.writerows(sorted(rows, key=lambda row: (
                    int(row["sort_order"]), row["task_template_id"]
                )))
            valid, output = self.candidate_valid(temporary)
            self.app.log(output)
            if not valid:
                raise ValueError("Candidate validation failed. CSV unchanged; see Activity Log.")
            if self.digest(path) != self.snapshot_hash:
                raise ValueError("Task CSV changed during validation. CSV unchanged.")
            folder = self.backup_directory()
            folder.mkdir(parents=True, exist_ok=True)
            backup = folder / (
                "task_templates_" + datetime.now().strftime("%Y%m%d_%H%M%S_%f")
                + ".csv"
            )
            shutil.copy2(path, backup)
            os.replace(temporary, path)
            self.app.log(f"Task Templates saved. Backup: {backup}")
            self.app.refresh_all(False)
            messagebox.showinfo(
                "Task Templates", f"Candidate validation passed.\nBackup: {backup}",
                parent=self.app
            )
        finally:
            if temporary.exists():
                temporary.unlink()

    def save(self):
        if not require_single(self, "Save Template", messagebox):
            return
        try:
            row = self.record()
            key = row["task_template_id"]
            if self.editing_id and key != self.editing_id:
                raise ValueError("Template ID cannot change after creation.")
            if not self.editing_id and any(
                r["task_template_id"] == key for r in self.loaded_rows
            ):
                raise ValueError("Template ID already exists.")
            rows = [
                dict(r) for r in self.loaded_rows
                if r["task_template_id"] != key
            ] + [row]
            self.commit(rows)
            self.editing_id = key
            self.tree.selection_set(key)
            self.on_select()
        except (OSError, ValueError) as error:
            messagebox.showerror("Task Templates", str(error), parent=self.app)

    def remove(self):
        if not require_single(self, "Remove Template", messagebox):
            return
        if not self.editing_id:
            messagebox.showwarning(
                "Task Templates", "Select a template first.", parent=self.app
            )
            return
        key = self.editing_id
        if not messagebox.askyesno(
            "Remove Template?", f"Remove '{key}'?\nA CSV backup will be created.",
            default=messagebox.NO, icon="warning", parent=self.app
        ):
            return
        try:
            self.commit([
                dict(row) for row in self.loaded_rows
                if row["task_template_id"] != key
            ])
            self.new()
        except (OSError, ValueError) as error:
            messagebox.showerror("Task Templates", str(error), parent=self.app)

    def open_backups(self):
        self.backup_directory().mkdir(parents=True, exist_ok=True)
        self.app.open_folder(self.backup_directory())


def install_task_templates_editor(app):
    app.task_templates_editor = TaskTemplatesEditor(app)

def enable_windows_dark_title_bar(window: tk.Tk) -> None:
    """Request a dark native Windows title bar for the top-level app frame."""

    if sys.platform != "win32":
        return

    try:
        import ctypes

        window.update_idletasks()

        user32 = ctypes.windll.user32
        dwm_api = ctypes.windll.dwmapi.DwmSetWindowAttribute

        user32.GetAncestor.argtypes = [
            ctypes.c_void_p,
            ctypes.c_uint,
        ]
        user32.GetAncestor.restype = ctypes.c_void_p

        dwm_api.argtypes = [
            ctypes.c_void_p,
            ctypes.c_uint,
            ctypes.c_void_p,
            ctypes.c_uint,
        ]
        dwm_api.restype = ctypes.c_long

        # Tk exposes a client-area handle. DWM title-bar settings must target
        # the top-level frame window instead.
        GA_ROOT = 2
        client_handle = window.winfo_id()
        top_level_handle = user32.GetAncestor(
            ctypes.c_void_p(client_handle),
            GA_ROOT,
        )

        hwnd = top_level_handle or client_handle
        enabled = ctypes.c_int(1)

        # Attribute 20 is used by current Windows 10/11 releases.
        result = dwm_api(
            ctypes.c_void_p(hwnd),
            20,
            ctypes.byref(enabled),
            ctypes.sizeof(enabled),
        )

        if result != 0:
            # Attribute 19 is the Windows 10 1809–1909 compatibility value.
            dwm_api(
                ctypes.c_void_p(hwnd),
                19,
                ctypes.byref(enabled),
                ctypes.sizeof(enabled),
            )

    except (AttributeError, OSError, tk.TclError):
        # Unsupported Windows versions retain their standard system title bar.
        pass
def find_project_root(start: Path) -> Path:
    current = start.resolve()

    for candidate in (current, *current.parents):
        if (
            (candidate / "data_authoring").is_dir()
            and (candidate / "tools").is_dir()
            and (candidate / "DoDA").is_dir()
        ):
            return candidate

    raise FileNotFoundError(
        "Could not locate a DoDA project root. Choose the doda-core folder manually."
    )


def read_csv(path: Path) -> list[dict[str, str]]:
    if not path.is_file():
        return []

    with path.open("r", encoding="utf-8-sig", newline="") as source_file:
        reader = csv.DictReader(source_file)
        rows: list[dict[str, str]] = []

        for raw_row in reader:
            if raw_row is None:
                continue

            row = {
                (key or "").strip(): (value or "").strip()
                for key, value in raw_row.items()
                if key is not None
            }

            if any(row.values()):
                rows.append(row)

        return rows


def truncate_text(value: str, maximum_length: int = 120) -> str:
    normalized = " ".join((value or "").split())

    if len(normalized) <= maximum_length:
        return normalized

    return normalized[: maximum_length - 3] + "..."

    def open_foims_preview(self) -> None:
        """Open the singleton FOIMS-X preview monitor or raise it if already open."""

        if (
            self.foims_preview is not None and
            self.foims_preview.is_open()
        ):
            self.foims_preview.show()
            self.foims_preview.refresh_preview()
            return

        self.foims_preview = FoimsXPreview(self)
        self.log("Opened FOIMS-X Preview Monitor.")

    def refresh_foims_preview(self) -> None:
        """Refresh an open FOIMS-X monitor without creating a new window."""

        if (
            self.foims_preview is None or
            not self.foims_preview.is_open()
        ):
            self.open_foims_preview()
            return

        self.foims_preview.refresh_preview()
        self.foims_preview.show()
        self.log("Refreshed FOIMS-X Preview Monitor.")

    def toggle_fullscreen(
        self,
        _event: Any | None = None,
    ) -> str:
        """Toggle the main Content Workbench window between normal and full screen."""
        is_fullscreen = bool(self.attributes("-fullscreen"))
        self.attributes("-fullscreen", not is_fullscreen)
        return "break"

    def open_foims_preview_from_shortcut(
        self,
        _event: Any | None = None,
    ) -> str:
        """Open FOIMS-X from Ctrl+O while consuming the Tk key event."""
        self.open_foims_preview()
        return "break"
    def _show_client_menu(
        self,
        event: tk.Event[Any],
        menu: tk.Menu,
    ) -> str:
        """Post a client-area menu directly below the clicked menu button."""

        try:
            menu.tk_popup(
                event.x_root,
                event.y_root + event.widget.winfo_height(),
            )
        finally:
            menu.grab_release()

        return "break"

    def _build_client_menu_bar(self, parent: tk.Widget) -> None:
        """Render a dark, fully functional menu row inside the client area."""

        menu_row = tk.Frame(
            parent,
            background="#161b22",
            highlightthickness=0,
            borderwidth=0,
        )
        menu_row.pack(fill=tk.X, pady=(0, 10))

        menu_items = (
            ("File", self.file_menu),
            ("Actions", self.actions_menu),
            ("Preview", self.preview_menu),
            ("View", self.view_menu),
            ("Folders", self.folders_menu),
            ("Help", self.help_menu),
        )

        for label, menu in menu_items:
            button = tk.Menubutton(
                menu_row,
                text=label,
                menu=menu,
                background="#161b22",
                foreground="#e6edf3",
                activebackground="#264f78",
                activeforeground="#ffffff",
                disabledforeground="#6e7681",
                borderwidth=0,
                relief=tk.FLAT,
                highlightthickness=0,
                padx=10,
                pady=5,
                anchor=tk.W,
                cursor="hand2",
                font=("Segoe UI", 9),
                takefocus=True,
            )

            button.pack(side=tk.LEFT)

            button.bind(
                "<Button-1>",
                lambda event, target_menu=menu: self._show_client_menu(
                    event,
                    target_menu,
                ),
            )

            button.bind(
                "<Return>",
                lambda event, target_menu=menu: self._show_client_menu(
                    event,
                    target_menu,
                ),
            )

            button.bind(
                "<space>",
                lambda event, target_menu=menu: self._show_client_menu(
                    event,
                    target_menu,
                ),
            )

            button.bind(
                "<Enter>",
                lambda event, target=button: target.configure(
                    background="#264f78",
                    foreground="#ffffff",
                ),
            )

            button.bind(
                "<Leave>",
                lambda event, target=button: target.configure(
                    background="#161b22",
                    foreground="#e6edf3",
                ),
            )
    def select_relative_notebook_tab(self, offset: int) -> str:
        tab_ids = self.notebook.tabs()

        if not tab_ids:
            return "break"

        try:
            current_index = self.notebook.index(self.notebook.select())
        except tk.TclError:
            current_index = 0

        target_index = (current_index + offset) % len(tab_ids)

        self.notebook.select(tab_ids[target_index])
        self.notebook.focus_set()

        return "break"

    def select_previous_notebook_tab(
        self,
        _event: Any = None,
    ) -> str:
        return self.select_relative_notebook_tab(-1)

    def select_next_notebook_tab(
        self,
        _event: Any = None,
    ) -> str:
        return self.select_relative_notebook_tab(1)

    def refresh_all(self, show_message: bool = False) -> None:
        try:
            self.tables = {
                table_name: read_csv(self.source_directory / filename)
                for table_name, filename in SOURCE_FILES.items()
            }
            if hasattr(self, "metrics_editor"):
                self.metrics_editor.update()

            if hasattr(self, "metric_tiers_editor"):
                self.metric_tiers_editor.update()

            if hasattr(self, "facilities_editor"):
                self.facilities_editor.update()

            if hasattr(self, "enum_catalog_editor"):
                self.enum_catalog_editor.update()

                self.capabilities_editor.update()

                self.facility_capabilities_editor.update()

            if hasattr(self, "capability_staffing_rules_editor"):
                self.capability_staffing_rules_editor.update()

            if hasattr(self, "facility_capability_staffing_overrides_editor"):
                self.facility_capability_staffing_overrides_editor.update()

            if hasattr(self, "task_templates_editor"):
                self.task_templates_editor.update()

            for network_editor in getattr(self, "network_editors", []):
                network_editor.update()

            if hasattr(self, "transfer_rules_view"):
                self.transfer_rules_view.update()

            self.load_validation_report()
            self.update_dashboard()
            self.update_tracks()
            self.update_metrics()
            self.update_facilities()
            self.update_flavor_text()
            self.update_status_labels()

            self.status_text.set("Source data refreshed successfully.")
            self.log("Source data refreshed.")

            if show_message:
                messagebox.showinfo(
                    APP_TITLE,
                    "Source data was refreshed successfully.",
                )

        except Exception as error:
            self.status_text.set("Refresh failed.")
            self.log(f"REFRESH ERROR: {error}")
            messagebox.showerror(
                APP_TITLE,
                f"Unable to refresh source data.\n\n{error}",
            )

    def load_validation_report(self) -> None:
        self.validation_report = {}

        if not self.validation_report_path.is_file():
            return

        try:
            self.validation_report = json.loads(
                self.validation_report_path.read_text(encoding="utf-8")
            )
        except (OSError, json.JSONDecodeError) as error:
            self.log(f"Could not read validation report: {error}")

    def update_status_labels(self) -> None:
        if self.validation_report:
            valid = bool(self.validation_report.get("valid", False))
            error_count = self.validation_report.get("errorCount", "?")
            warning_count = self.validation_report.get("warningCount", "?")
            generated_at = self.validation_report.get("generatedAtUtc", "unknown")

            if valid:
                self.validation_text.set(
                    f"PASS — {error_count} error(s), {warning_count} warning(s), "
                    f"report: {generated_at}"
                )
            else:
                self.validation_text.set(
                    f"FAIL — {error_count} error(s), {warning_count} warning(s), "
                    f"report: {generated_at}"
                )
        else:
            self.validation_text.set(
                "UNKNOWN — no validation report has been generated yet."
            )

        pk3_path = self.build_directory / "doda-generated-content.pk3"

        if pk3_path.is_file():
            modified = datetime.fromtimestamp(
                pk3_path.stat().st_mtime
            ).strftime("%Y-%m-%d %H:%M:%S")

            self.export_text.set(
                f"READY — doda-generated-content.pk3, "
                f"{pk3_path.stat().st_size:,} bytes, built {modified}"
            )
        else:
            self.export_text.set(
                "NOT BUILT — no generated PK3 overlay exists yet."
            )

    def update_dashboard(self) -> None:
        for item in self.dashboard_tree.get_children():
            self.dashboard_tree.delete(item)

        total_rows = 0
        source_files_present = 0

        for table_name, filename in SOURCE_FILES.items():
            path = self.source_directory / filename
            rows = self.tables.get(table_name, [])
            total_rows += len(rows)

            if path.is_file():
                source_files_present += 1
                modified = datetime.fromtimestamp(
                    path.stat().st_mtime
                ).strftime("%Y-%m-%d %H:%M:%S")
                status = "Available"
            else:
                modified = "—"
                status = "Missing"

            self.dashboard_tree.insert(
                "",
                tk.END,
                iid=table_name,
                values=(
                    table_name,
                    filename,
                    len(rows),
                    status,
                    modified,
                ),
            )

        validation_state = "unknown"

        if self.validation_report:
            validation_state = (
                "passed"
                if self.validation_report.get("valid", False)
                else "failed"
            )

        self.dashboard_summary.configure(
            text=(
                f"Authoring tables: {source_files_present}/{len(SOURCE_FILES)} "
                f"available\n"
                f"Total authored records: {total_rows}\n"
                f"Latest validation state: {validation_state}\n"
                f"Current mode: V1.1 authoring data is loaded; editor coverage is expanding."
            )
        )

    def find_row(
        self,
        table_name: str,
        key_name: str,
        key_value: str,
    ) -> dict[str, str] | None:
        for row in self.tables.get(table_name, []):
            if row.get(key_name) == key_value:
                return row

        return None

    def run_validation(self) -> None:
        if not self.validator_path.is_file():
            messagebox.showerror(
                APP_TITLE,
                f"Validator script was not found:\n{self.validator_path}",
            )
            return

        command = [
            sys.executable,
            str(self.validator_path),
            "--root",
            str(self.project_root),
            "validate",
        ]

        if not self.run_command(
            label="Validation",
            command=command,
            success_message="Validation completed successfully.",
            announce_success=False,
        ):
            return
        if self.run_network_rules("validate", announce=False):
            self.status_text.set("Validate All passed.")
            messagebox.showinfo(APP_TITLE, "Validate All passed. See the Activity Log for warnings.")

    def run_export(self) -> None:
        if not self.exporter_path.is_file():
            messagebox.showerror(
                APP_TITLE,
                f"Exporter script was not found:\n{self.exporter_path}",
            )
            return

        command = [
            sys.executable,
            str(self.exporter_path),
            "--root",
            str(self.project_root),
        ]

        if not self.run_command(
            label="Export",
            command=command,
            success_message="Export and PK3 build completed successfully.",
            announce_success=False,
        ):
            self.log("Export All stopped: content export failed; catalog was not regenerated.")
            return
        if not self.run_network_rules("export", announce=False):
            self.status_text.set("Export All incomplete: rules/catalog export failed.")
            self.log("Export All incomplete: content/PK3 may have updated; no full success.")
            return
        self.status_text.set("Export All completed successfully.")
        messagebox.showinfo(APP_TITLE, "Export All completed: content, PK3, rules, and ZScript catalog.")

    def run_network_rules(self, mode: str, announce: bool) -> bool:
        """Validate or export network/transfer rules in-process and log the result.

        announce=False only raises a dialog on failure (used after the main
        validation/export so the user does not get two success dialogs).
        """
        dirty = network_editors_dirty(self)
        if dirty:
            self.log("Network rules note: unsaved drafts are not included: " + ", ".join(dirty))
        try:
            if mode == "export":
                if announce:
                    self.log("Rules-only export reuses generated/data/department_content.json; "
                             "use Export All to include saved facility/capability changes.")
                result, out_path = doda_network_rules.export(self.project_root)
                resource_result, resource_path = doda_resource_rules.export(self.project_root)
                result.errors += resource_result.errors
                result.warnings += resource_result.warnings
                flow_result, flow_path = doda_task_flow_rules.export(self.project_root)
                result.errors += flow_result.errors
                result.warnings += flow_result.warnings
                staffing_result, staffing_path = doda_task_staffing_rules.export(self.project_root)
                result.errors += staffing_result.errors
                result.warnings += staffing_result.warnings
                collection_result, collection_path = doda_collection_rules.export(self.project_root)
                result.errors += collection_result.errors
                result.warnings += collection_result.warnings
                collection_models_result, collection_models_path = doda_collection_model_rules.export(self.project_root)
                result.errors += collection_models_result.errors
                result.warnings += collection_models_result.warnings
                evidence_result, evidence_path = doda_evidence_instance_rules.export(self.project_root)
                result.errors += evidence_result.errors
                result.warnings += evidence_result.warnings
                prerequisite_result, prerequisite_path = doda_task_prerequisite_rules.export(self.project_root)
                result.errors += prerequisite_result.errors
                result.warnings += prerequisite_result.warnings
                vehicle_result, vehicle_path = doda_vehicle_rules.export(self.project_root)
                result.errors += vehicle_result.errors
                result.warnings += vehicle_result.warnings
                if vehicle_path is not None:
                    self.log(f"Vehicle catalog exported: {vehicle_path}")
                if prerequisite_path is not None:
                    self.log(f"Task prerequisites exported: {prerequisite_path}")
                if evidence_path is not None:
                    self.log(f"Evidence type instances exported: {evidence_path}")
                if collection_models_path is not None:
                    self.log(f"Collection models exported: {collection_models_path}")
                if collection_path is not None:
                    self.log(f"Collection receptacles exported: {collection_path}")
                if staffing_path is not None:
                    self.log(f"Task staffing rules exported: {staffing_path}")
                if flow_path is not None:
                    self.log(f"Task resource flows exported: {flow_path}")
                if resource_path is not None:
                    self.log(f"Resource catalog exported: {resource_path}")
                if result.valid:
                    equipment_result, equipment_path = doda_equipment_rules.export(self.project_root)
                    result.errors += equipment_result.errors
                    result.warnings += equipment_result.warnings
                    if equipment_path is not None:
                        self.log(f"Equipment catalog exported: {equipment_path}")
                    care_result, care_path = doda_care_rules.export(self.project_root)
                    result.errors += care_result.errors
                    result.warnings += care_result.warnings
                    if care_path is not None:
                        self.log(f"Care services catalog exported: {care_path}")
                    condition_result, condition_path = doda_condition_rules.export(self.project_root)
                    result.errors += condition_result.errors
                    result.warnings += condition_result.warnings
                    if condition_path is not None:
                        self.log(f"Personnel conditions catalog exported: {condition_path}")
                    attribute_result, attribute_path = doda_agent_attribute_rules.export(self.project_root)
                    result.errors += attribute_result.errors
                    result.warnings += attribute_result.warnings
                    if attribute_path is not None:
                        self.log(f"Personnel traits/training catalog exported: {attribute_path}")
                    pcare_result, pcare_path = doda_personal_care_rules.export(self.project_root)
                    result.errors += pcare_result.errors
                    result.warnings += pcare_result.warnings
                    if pcare_path is not None:
                        self.log(f"Personal-care catalog exported: {pcare_path}")
                    time_result, time_path = doda_time_rules.export(self.project_root)
                    result.errors += time_result.errors
                    result.warnings += time_result.warnings
                    if time_path is not None:
                        self.log(f"Time settings catalog exported: {time_path}")
                    rc_result, rc_path = doda_recreation_content_rules.export(self.project_root)
                    result.errors += rc_result.errors
                    result.warnings += rc_result.warnings
                    if rc_path is not None:
                        self.log(f"Recreation content catalog exported: {rc_path}")
                    sobriety_result, sobriety_path = doda_sobriety_rules.export(self.project_root)
                    result.errors += sobriety_result.errors
                    result.warnings += sobriety_result.warnings
                    if sobriety_path is not None:
                        self.log(f"Sobriety catalog exported: {sobriety_path}")
                    territory_result, territory_path = doda_territory_rules.export(self.project_root)
                    result.errors += territory_result.errors
                    result.warnings += territory_result.warnings
                    if territory_path is not None:
                        self.log(f"Territory catalog exported: {territory_path}")
                    weapon_result, weapon_path = doda_weapon_rules.export(self.project_root)
                    result.errors += weapon_result.errors
                    result.warnings += weapon_result.warnings
                    if weapon_path is not None:
                        self.log(f"Weapon catalog exported: {weapon_path}")
                    hostile_result, hostile_path = doda_hostile_rules.export(self.project_root)
                    result.errors += hostile_result.errors
                    result.warnings += hostile_result.warnings
                    if hostile_path is not None:
                        self.log(f"Hostile catalog exported: {hostile_path}")
                    identity_result, identity_path = doda_progression_identity_rules.export(self.project_root)
                    result.errors += identity_result.errors
                    result.warnings += identity_result.warnings
                    if identity_path is not None:
                        self.log(f"Progression identity catalog exported: {identity_path}")
                    mission_result, mission_path = doda_authored_mission_rules.export(self.project_root)
                    result.errors += mission_result.errors
                    result.warnings += mission_result.warnings
                    if mission_path is not None:
                        self.log(f"Authored mission catalog exported: {mission_path}")
                    if (equipment_result.valid and care_result.valid and condition_result.valid
                            and attribute_result.valid and pcare_result.valid and time_result.valid and rc_result.valid):
                        catalog_path, catalog_sha, included = doda_zscript_catalog.generate(self.project_root)
                        self.log(f"ZScript catalog regenerated: {catalog_path} sha256 {catalog_sha[:16]}")
                        if not included:
                            result.warnings.append("zscript.txt does not include DoDA/Generated/WorkbenchCatalog.zs")
            else:
                result = doda_network_rules.Result()
                doda_network_rules.validate_contract(result)
                doda_network_rules.validate_network(self.project_root, result)
                doda_equipment_rules.validate_equipment(self.project_root, result)
                doda_care_rules.validate_care(self.project_root, result)
                doda_condition_rules.validate_conditions(self.project_root, result)
                doda_agent_attribute_rules.validate_attributes(self.project_root, result)
                doda_personal_care_rules.validate_personal_care(self.project_root, result)
                doda_time_rules.validate_time_settings(self.project_root, result)
                doda_recreation_content_rules.validate_recreation_content(self.project_root, result)
                doda_sobriety_rules.validate_sobriety(self.project_root, result)
                doda_territory_rules.validate_territory(self.project_root, result)
                doda_weapon_rules.validate_weapons(self.project_root, result)
                doda_hostile_rules.validate_hostiles(self.project_root, result)
                doda_progression_identity_rules.validate_identities(self.project_root, result)
                doda_authored_mission_rules.validate_authored_missions(self.project_root, result)
                doda_resource_rules.validate_resources(self.project_root, result)
                doda_task_flow_rules.validate_flows(self.project_root, result)
                doda_task_staffing_rules.validate_staffing(self.project_root, result)
                doda_collection_rules.validate_collection(self.project_root, result)
                doda_collection_model_rules.validate_models(self.project_root, result)
                doda_evidence_instance_rules.validate_instances(self.project_root, result)
                doda_task_prerequisite_rules.validate_prerequisites(self.project_root, result)
                doda_vehicle_rules.validate_vehicles(self.project_root, result)
                if result.valid and not doda_zscript_catalog.is_current(self.project_root):
                    result.warnings.append("ZScript catalog is stale; export network rules to regenerate it")
                out_path = None
        except Exception as error:
            self.log(f"Network rules {mode} error: {error}")
            messagebox.showerror(APP_TITLE, f"Network rules {mode} could not run.\n\n{error}")
            return False
        for line in result.warnings:
            self.log(f"  NETWORK WARNING: {line}")
        for line in result.errors:
            self.log(f"  NETWORK ERROR: {line}")
        if out_path is not None:
            self.log(f"Network rules exported: {out_path}")
        verdict = "PASS" if result.valid else "FAIL"
        self.log(f"Network rules {mode}: {verdict} ({len(result.errors)} errors, {len(result.warnings)} warnings)")
        if not result.valid:
            messagebox.showerror(APP_TITLE, f"Network rules {mode} failed. See the Activity Log.")
        elif announce:
            messagebox.showinfo(APP_TITLE, f"Network rules {mode} passed ({len(result.warnings)} warnings).")
        return result.valid

        handler = getattr(self, "_network_close_handler", None)
        if handler is not None:
            handler()
        else:
            self.destroy()

COLUMNS = ('task_template_id', 'display_name', 'description', 'capability_id', 'priority', 'target_type', 'target_family_id', 'loop_task', 'time_to_complete_minutes', 'reward_mode', 'payout_count', 'sort_order', 'active')
TARGET_TYPES = ("STATION", "TERRITORY", "MISSION_AREA")
REWARD_MODES = ("CONTINUOUS", "PERIODIC_LUMP_SUM")

def boolean(value):
    if type(value) is bool:
        return value
    if isinstance(value, str) and value.strip().upper() in ("TRUE", "FALSE"):
        return value.strip().upper() == "TRUE"
    raise ValueError("must be TRUE or FALSE")

def integer(value, optional=False):
    if optional and value in (None, ""):
        return None
    if type(value) is int:
        return value
    if isinstance(value, str):
        return int(value.strip())
    raise ValueError("must be a whole number")

def errors(row, priorities):
    problems = []
    for field in ("display_name", "description"):
        if not row.get(field):
            problems.append(field + " is required")
    if row.get("priority") not in priorities:
        problems.append("unknown priority")
    if row.get("target_type") not in TARGET_TYPES:
        problems.append("invalid target_type")
    if row.get("reward_mode") not in REWARD_MODES:
        problems.append("invalid reward_mode")
    values = {}
    for field in ("active", "loop_task"):
        try:
            values[field] = boolean(row.get(field))
        except ValueError:
            problems.append(field + " must be TRUE or FALSE")
    for field, optional, minimum in (
        ("time_to_complete_minutes", True, 1), ("payout_count", False, 0),
        ("sort_order", False, 1),
    ):
        try:
            value = integer(row.get(field), optional)
            if value is not None and value < minimum:
                raise ValueError()
            values[field] = value
        except (ValueError, TypeError):
            problems.append(field + " has an invalid whole-number value")
    active = values.get("active", False)
    duration = values.get("time_to_complete_minutes")
    payouts = values.get("payout_count", 0)
    if active and not row.get("capability_id"):
        problems.append("active template requires capability_id")
    if row.get("target_type") != "STATION" and row.get("target_family_id"):
        problems.append("target_family_id applies only to STATION")
    if active and row.get("target_type") == "STATION" and not row.get("target_family_id"):
        problems.append("active STATION template requires target_family_id")
    if active and values.get("loop_task") and duration is None:
        problems.append("active loop task requires a completion-cycle duration")
    if row.get("reward_mode") == "CONTINUOUS" and payouts:
        problems.append("CONTINUOUS templates use payout_count 0")
    if row.get("reward_mode") == "PERIODIC_LUMP_SUM":
        if active and (duration is None or not payouts):
            problems.append("active periodic task requires duration and payout_count")
        if duration is not None and payouts and duration % payouts:
            problems.append("duration must divide evenly into payout intervals")
    return problems

"""Inline multi-row editing; no bulk window or operation chooser."""
import csv

from doda_bulk_edit import BulkEditError, apply_to_draft
from doda_row_selection import MULTIPLE_VALUES, selection_display, set_form_state

DROPDOWN_KINDS = {"bool", "choice", "ref", "track_ref", "task_ref", "collection_type_ref", "evidence_ref"}
REF_KINDS = {"ref", "track_ref", "task_ref", "collection_type_ref", "evidence_ref"}


class InlineBulkEditor:
    def __init__(self, editor):
        self.editor = editor
        self.active = False
        self.loading = False
        self.selection = ()
        self.original = []
        self.changes = {}
        self.initial_values = {}
        self.traces = []
        self.fields = {f.name: f for f in editor.spec.fields}
        for name, field in self.fields.items():
            widget = editor.widgets[name]
            kind = getattr(field, "kind", "text")
            if kind in DROPDOWN_KINDS:
                self.initial_values[name] = widget.cget("values")
            trace = editor.vars[name].trace_add("write", lambda *_a, n=name: self.changed(n))
            self.traces.append((name, trace))
            if name in editor.spec.key_columns:
                continue
            if kind in DROPDOWN_KINDS:
                widget.bind("<<ComboboxSelected>>", lambda _e, n=name: self.changed(n), add="+")
            elif kind != "evidence_multi":
                widget.bind("<FocusIn>", lambda _e, n=name: self.focus(n), add="+")
                widget.bind("<ButtonRelease-1>", lambda _e, n=name: self.focus(n), add="+")

    def same_session(self, selection):
        return self.active and self.selection == tuple(selection) and self.original == self.editor.rows

    def begin(self, selection, force=False):
        if not force and self.same_session(selection):
            return
        self.active = False
        self.loading = True
        try:
            self.selection = tuple(selection)
            self.original = [dict(row) for row in self.editor.rows]
            selected = [self.original[int(item)] for item in self.selection]
            self.changes = {}
            set_form_state(self.editor, False)
            for name, field in self.fields.items():
                widget = self.editor.widgets[name]
                if getattr(field, "kind", "text") == "evidence_multi":
                    widget._bulk_mode = True
                self.editor.vars[name].set(selection_display(selected, name, show_common=True))
                if name in self.editor.spec.key_columns:
                    widget.configure(state="disabled")
            self.active = True
        finally:
            self.loading = False
        self.editor._set_status(
            f"{len(self.selection)} rows selected - edit Record fields, then Apply to List. "
            "Untouched fields are preserved."
        )

    def end(self):
        self.active = False
        self.changes = {}
        self.selection = ()
        self.original = []
        for name, field in self.fields.items():
            if getattr(field, "kind", "text") == "evidence_multi":
                self.editor.widgets[name]._bulk_mode = False
        for name, values in self.initial_values.items():
            self.editor.widgets[name].configure(values=values)

    def changed(self, name):
        if not self.active or self.loading or name in self.editor.spec.key_columns:
            return
        self.changes[name] = self.editor.vars[name].get()
        suffix = " UNSAVED CHANGES." if self.editor.dirty else ""
        self.editor._set_status(
            f"{len(self.selection)} rows selected; {len(self.changes)} edited field(s) ready for Apply to List." + suffix
        )

    def focus(self, name):
        if not self.active or name in self.changes or name in self.editor.spec.key_columns:
            return
        if self.editor.vars[name].get() != MULTIPLE_VALUES:
            return
        widget = self.editor.widgets[name]
        def select_placeholder():
            if self.active and name not in self.changes and self.editor.vars[name].get() == MULTIPLE_VALUES:
                widget.selection_range(0, "end")
                widget.icursor("end")
        widget.after_idle(select_placeholder)

                        kinds = {controller.original[int(item)].get("collection_kind", "") for item in controller.selection}
                        collection_kind = next(iter(kinds)) if len(kinds) == 1 else None
                    choices = reference_keys(editor.app.source_directory, field, collection_kind)

def apply_inline(editor, validate, decimal_check):
    controller = install_inline(editor)
    if not controller.active:
        editor._set_status("Batch Apply blocked: reselect the rows before editing.")
        return False
    changes = dict(controller.changes)
    try:
        applied = apply_to_draft(
            editor, controller.selection, controller.original, changes, validate, decimal_check,
        )
    except (ValueError, OSError, UnicodeError, csv.Error) as error:
        message = str(error)
        log = getattr(editor.app, "log", None)
        if callable(log):
            log(f"[{editor.spec.table}] Batch Apply blocked; draft retained: {message}")
        editor._set_status("Batch Apply blocked: " + message.splitlines()[0] + " See Activity Log.")
        return False
    if not applied:
        editor._set_status("No deliberately edited fields. Draft unchanged.")
        return False
    selection = controller.selection
    for selected in selection:
        index = int(selected)
        editor.tree.item(selected, values=[
            editor.rows[index].get(field.name, "") for field in editor.spec.fields if not field.multiline

    controller.begin(selection, force=True)
    editor._ref_choices()
    editor._set_status(
        f"Updated {len(selection)} draft rows; {len(changes)} field(s). UNSAVED CHANGES."
    )
    return True

"""Explicit, atomic draft-only batch editing for shared table pages."""
import csv

from doda_row_selection import MULTIPLE_VALUES, selection_display


class BulkEditError(ValueError):
    pass


def build_candidate(rows, indices, changes, spec, decimal_check=None):
    indices = tuple(indices)
    if len(indices) < 2 or len(set(indices)) != len(indices):
        raise BulkEditError("Select at least two distinct rows.")
    if any(not isinstance(i, int) or i < 0 or i >= len(rows) for i in indices):
        raise BulkEditError("Selection no longer matches the draft.")
    fields = {f.name: f for f in spec.fields}
    for name, value in changes.items():
        if name not in fields:
            raise BulkEditError("Unknown field: " + name)
        if name in spec.key_columns:
            raise BulkEditError("Identity fields cannot be batch-edited: " + name)
        if not isinstance(value, str):
            raise BulkEditError("Field values must be text: " + name)
        if value.strip() == MULTIPLE_VALUES or (
            fields[name].kind == "evidence_multi"
            and MULTIPLE_VALUES in [v.strip() for v in value.split(";")]
        ):
            raise BulkEditError("Placeholder text cannot be applied: " + name)
    candidate = [dict(row) for row in rows]
    for index in indices:
        candidate[index].update({name: value.strip() for name, value in changes.items()})
        row = candidate[index]
        for f in spec.fields:
            value = row.get(f.name, "")
            if f.required and not value.strip():
                raise BulkEditError(f"Row {index + 1}: {f.label} is required.")
            if value and f.kind == "int":
                try:
                    int(value)
                except ValueError:
                    raise BulkEditError(f"Row {index + 1}: {f.label} must be a whole number.")
            if value and f.kind == "decimal" and decimal_check is not None:
                try:
                    decimal_check(value)
                except ValueError as error:
                    raise BulkEditError(f"Row {index + 1}: {f.label}: {error}") from error
            if f.name in changes and value:
                if f.kind == "bool" and value.lower() not in ("true", "false"):
                    raise BulkEditError(f"{f.label} must be true or false.")
                if f.kind == "choice" and f.choices and value not in f.choices:
                    raise BulkEditError(f"{f.label}: select an available choice.")
    keys = [tuple(row.get(k, "") for k in spec.key_columns) for row in candidate]
    if any(any(not value.strip() for value in key) for key in keys):
        raise BulkEditError("Draft contains a blank identity key.")
    if len(keys) != len(set(keys)):
        raise BulkEditError("Draft contains duplicate identity keys.")
    return candidate


def apply_to_draft(editor, selection, original_rows, changes, validate, decimal_check=None):
    if tuple(editor.tree.selection()) != tuple(selection) or editor.rows != original_rows:
        raise BulkEditError("Draft or selection changed. Reselect the rows before applying.")
    if not changes:
        return False
    candidate = build_candidate(
        original_rows, [int(item) for item in selection], changes,
        editor.spec, decimal_check,
    )
    result = validate(editor.app.project_root, editor.spec.table, candidate)
    if not result.valid:
        raise BulkEditError("Batch validation failed; no rows changed.\n\n" + "\n".join(result.errors[:12]))
    if tuple(editor.tree.selection()) != tuple(selection) or editor.rows != original_rows:
        raise BulkEditError("Draft or selection changed during validation. No rows changed.")
    editor.rows = candidate
    editor.dirty = True
    return True

import unittest

class TaskSchemaTests(unittest.TestCase):
    def row(self):
        return dict(task_template_id="T", display_name="Task", description="Work.",
                    capability_id="C", priority="NORMAL", target_type="TERRITORY",
                    target_family_id="", loop_task=False, time_to_complete_minutes=None,

    def test_open_ended(self):
        self.assertEqual(schema.errors(self.row(), {"NORMAL"}), [])
    def test_active_loop_requires_duration(self):
        row = self.row(); row["loop_task"] = True
        self.assertTrue(schema.errors(row, {"NORMAL"}))
        row["time_to_complete_minutes"] = 10
        self.assertEqual(schema.errors(row, {"NORMAL"}), [])
    def test_drafts_can_be_unconfigured(self):

        self.assertTrue(schema.errors(row, {"NORMAL"}))
        row["target_family_id"] = "PAPER_FORENSICS"
        self.assertEqual(schema.errors(row, {"NORMAL"}), [])
        row["target_type"] = "TERRITORY"
        self.assertTrue(schema.errors(row, {"NORMAL"}))

        row["time_to_complete_minutes"]=10
        self.assertTrue(schema.errors(row, {"NORMAL"}))
