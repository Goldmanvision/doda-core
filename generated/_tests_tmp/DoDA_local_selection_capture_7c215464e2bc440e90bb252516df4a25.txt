MULTIPLE_VALUES = "Multiple values"


def selection_count(editor):
    selected = editor.tree.selection()
    return len(selected) if isinstance(selected, (tuple, list)) else 0


def selection_display(rows, field):
    return MULTIPLE_VALUES if any(row.get(field, "") for row in rows) else ""


def set_form_state(editor, readonly):
    states = getattr(editor, "_selection_widget_states", None)
    if states is None:
        states = {
            name: widget.cget("state")
            for name, widget in editor.widgets.items()
        }
        editor._selection_widget_states = states
    for name, widget in editor.widgets.items():
        widget.configure(state="disabled" if readonly else states[name])


def require_single(editor, action, dialogs):
    if selection_count(editor) > 1:
        dialogs.showwarning(
            editor.title if hasattr(editor, "title") else "Task Templates",
            action + " is a single-record action. Select one row. "
            "Bulk Apply is not enabled in this selection-only slice.",
        )
        return False
    return True

def _sha(path: Path) -> str | None:
    return hashlib.sha256(path.read_bytes()).hexdigest() if path.is_file() else None


def read_rows(path: Path, spec: TableSpec) -> list[dict[str, str]]:
    if not path.is_file():
        return []
    with path.open("r", encoding="utf-8-sig", newline="") as fh:
        reader = csv.DictReader(fh)
        rows = []
        for raw in reader:
            row = {c: (raw.get(c) or "").strip() for c in spec.columns}
            if any(row.values()):
                rows.append(row)
    return rows


def write_rows(path: Path, spec: TableSpec, rows: list[dict[str, str]]) -> None:
    tmp = None
    try:
        with tempfile.NamedTemporaryFile(
            mode="w", encoding="utf-8", newline="",
            dir=path.parent, prefix=f".{path.name}.", suffix=".tmp", delete=False,
        ) as fh:
            tmp = Path(fh.name)
            writer = csv.DictWriter(fh, fieldnames=spec.columns, lineterminator="\r\n")
            writer.writeheader()
            for row in rows:
                writer.writerow({column: row.get(column, "") for column in spec.columns})
        tmp.replace(path)
    finally:
        if tmp is not None:
            tmp.unlink(missing_ok=True)


def validate_candidate(project_root: Path, table: str, rows: list[dict[str, str]]) -> network_rules.Result:

    src = project_root / "data_authoring" / "source"

    tmp_src.mkdir(parents=True)
    (tmp_root / "tools").mkdir()
    try:
        for other in TABLE_METADATA.values():
            if other.table == table:
                write_rows(tmp_src / other.filename, other, rows)
            elif (src / other.filename).is_file():
                shutil.copy2(src / other.filename, tmp_src / other.filename)
        task_path = src / "task_templates.csv"
        if task_path.is_file():
            shutil.copy2(task_path, tmp_src / task_path.name)
        track_path = src / "department_tracks.csv"
        if track_path.is_file():
            shutil.copy2(track_path, tmp_src / track_path.name)
        res = network_rules.Result()
        network_rules.validate_contract(res)
        network_rules.validate_network(tmp_root, res)

        territory_rules.validate_territory(tmp_root, res)

        authored_mission_rules.validate_authored_missions(tmp_root, res)
        resource_rules.validate_resources(tmp_root, res)
        task_flow_rules.validate_flows(tmp_root, res)
        if (tmp_src / task_staffing_rules.STAFFING_FILE).is_file():
            task_staffing_rules.validate_staffing(tmp_root, res)
        if (tmp_src / collection_rules.RECEPTACLES_FILE).is_file():
            collection_rules.validate_collection(tmp_root, res)
        for filename in (collection_model_rules.DEVICES_FILE, collection_model_rules.MODELS_FILE):

def reference_keys(source_directory, field, collection_kind=None):
    if field.kind in ("track_ref", "task_ref"):
        filename, key = (
            ("department_tracks.csv", "track_id")
            if field.kind == "track_ref"
            else ("task_templates.csv", "task_template_id")
        )
        path = source_directory / filename
        if not path.is_file():
            return []
        with path.open(encoding="utf-8-sig", newline="") as handle:
            keys = [(r.get(key) or "").strip() for r in csv.DictReader(handle)]
        return list(dict.fromkeys(k for k in keys if k))
    if field.kind == "collection_type_ref":
        keys = []
        tables = (
            ("collection_receptacles", "collection_device_types")
            if collection_kind is None
            else {
                "RECEPTACLE": ("collection_receptacles",),
                "DEVICE": ("collection_device_types",),
            }.get(collection_kind, ())
        )
        for table in tables:
            spec = TABLE_METADATA[table]
            keys.extend(row[spec.key] for row in read_rows(
                source_directory / spec.filename, spec))
        return list(dict.fromkeys(keys))
    if field.kind == "evidence_ref":
        spec = TABLE_METADATA["resources"]
        return list(dict.fromkeys(
            row["resource_id"]
            for row in read_rows(source_directory / spec.filename, spec)
            if row["resource_id"] in evidence_instance_rules.EVIDENCE_RESOURCES
            and row["resource_kind"].upper() == "STOCKPILE"
        ))
    ref = TABLE_METADATA[field.ref_table]
    return [row[ref.key] for row in read_rows(source_directory / ref.filename, ref)]

    # ---- panel show / hide (View > Show / Hide Panel) ----------------
    def _pane_paths(self) -> list:
        return [str(p) for p in self.pane.panes()]

    def set_panel_visible(self, which: str, show: bool) -> None:
        frame = self._list_frame if which == "list" else self._form_frame
        present = str(frame) in self._pane_paths()
        if show == present:
            return
        try:
            if not show:
                if len(self._pane_paths()) < 2:
                    return
                self.pane.forget(frame)
            elif which == "list":
                paths = self._pane_paths()
                if paths:
                    self.pane.add(frame, before=paths[0], minsize=320, stretch="always")
                else:
                    self.pane.add(frame, minsize=320, stretch="always")
            else:
                self.pane.add(frame, minsize=400, stretch="never")
            paths = self._pane_paths()
            if len(paths) == 1:
                self.pane.paneconfigure(paths[0], stretch="always")
            else:
                self.pane.paneconfigure(self._list_frame, stretch="always")
                self.pane.paneconfigure(self._form_frame, stretch="never")
                self._sash_set = False
                self.pane.after(60, self._place_sash)
        except tk.TclError:
            pass

    # ---- CSV copy / paste --------------------------------------------
    def copy_csv(self) -> None:
        if not require_single(self, "Copy to CSV", messagebox):
            return
        import io
        names = [f.name for f in self.spec.fields]
        buf = io.StringIO()
        w = csv.writer(buf, lineterminator="\r\n")
        w.writerow(names)
        w.writerow([self.vars[n].get().strip() for n in names])
        self.tab.clipboard_clear()
        self.tab.clipboard_append(buf.getvalue())
        self.tab.update()
        self.app.log(f"[{self.spec.table}] Copied current record to clipboard as CSV.")
        self._set_status("Copied current record to clipboard as CSV.")

    def paste_csv(self) -> None:
        if not require_single(self, "Paste CSV", messagebox):
            return
        import io
        try:
            text = self.tab.clipboard_get()
        except tk.TclError:
            messagebox.showwarning(self.title, "The clipboard is empty or does not contain text.")
            return
        try:
            rows = [r for r in csv.reader(io.StringIO(text.lstrip("\ufeff")))
                    if any(c.strip() for c in r)]
        except csv.Error as exc:
            messagebox.showwarning(self.title, f"Clipboard text is not valid CSV: {exc}")
            return
        if len(rows) < 2:
            messagebox.showwarning(self.title, "Expected a header row followed by one data row.")
            return
        header = [h.strip() for h in rows[0]]
        names = {f.name for f in self.spec.fields}
        pasted = {h: v.strip() for h, v in zip(header, rows[1]) if h in names}
        if not pasted:
            messagebox.showwarning(self.title, "No column in the clipboard CSV matches this table.")
            return
        for f in self.spec.fields:
            blank = "true" if f.kind == "bool" else ""
            self.vars[f.name].set(pasted.get(f.name, blank))
        notes = []
        unknown = [h for h in header if h and h not in names]
        if unknown:
            notes.append("ignored columns: " + ", ".join(unknown))
        if len(rows) > 2:
            notes.append(f"used the first of {len(rows) - 1} data rows")
        self._set_status(f"Pasted {len(pasted)} field(s) into the form; not applied yet."
                         + (" (" + "; ".join(notes) + ")" if notes else ""))
        self._ref_choices()

        else:
            scopes = ()
            roles = ("",)
            values = ("",)
        self.widgets["scope"].configure(values=scopes)
        self.widgets["personnel_role"].configure(values=roles)
        self.widgets["requirement_value"].configure(values=values)

    def _prerequisite_kind_changed(self, _event: Any = None) -> None:
        kind = self.vars["requirement_kind"].get().strip()
        if kind in ("WEAPON_TYPE", "APPLICABLE_COLLECTION_RECEPTACLE"):
            self.vars["scope"].set("PER_ASSIGNED_AGENT")
            self.vars["personnel_role"].set("AGENT")
        elif kind == "MOTOR_VEHICLE":

            self.vars["personnel_role"].set("")
        if kind == "WEAPON_TYPE":
            current = self.vars["requirement_value"].get().strip()
            if current not in weapon_rules.WEAPON_TYPES:
                self.vars["requirement_value"].set("")
        elif kind in ("APPLICABLE_COLLECTION_RECEPTACLE", "MOTOR_VEHICLE"):
            self.vars["requirement_value"].set("")
        self._prerequisite_choices()

    def _sash_hover(self, event: Any) -> None:
        over = bool(self.pane.identify(event.x, event.y))
        self.pane.configure(background=SASH_ACTIVE if over else SASH_IDLE)

    def _place_sash(self) -> None:
        if getattr(self, "_sash_set", False):
            return
        try:
            width = self.pane.winfo_width()
            if width > 700:
                self.pane.sash_place(0, max(320, width - 450), 0)
                self._sash_set = True

    def _ref_choices(self) -> None:
        for f in self.spec.fields:
            if f.kind == "evidence_multi":
                self.widgets[f.name].set_options(evidence_choices(self.app.source_directory))
            elif f.kind in ("ref", "track_ref", "task_ref", "collection_type_ref", "evidence_ref"):
                if f.kind == "collection_type_ref":
                    kind = self.vars["collection_kind"].get().strip()
                    keys = reference_keys(self.app.source_directory, f, kind)
                else:
                    keys = reference_keys(self.app.source_directory, f)
                if not f.required:
                    keys = [""] + keys
                self.widgets[f.name].configure(values=keys)
        if self.spec.table == "task_prerequisites":
            self._prerequisite_choices()

        self._multi_selection = False
        set_form_state(self, False)
        self.rows = read_rows(self.path, self.spec)
        self.loaded_sha = _sha(self.path)
        self.dirty = False
        self.selected_index = None
        self._ref_choices()
        self._render()
        self._clear_form()

        """Called by the main refresh. Never discards unsaved drafts silently."""
        if self.dirty:
            self._ref_choices()
            self._set_status("Unsaved draft kept; source refresh skipped for this table.")
            return
        self.load()

    def _render(self) -> None:
        self.tree.delete(*self.tree.get_children())
        for i, r in enumerate(self.rows):
            self.tree.insert("", tk.END, iid=str(i),
                             values=[r.get(f.name, "") for f in self.spec.fields if not f.multiline])
        n = len(self.rows)
        self._set_status(f"{n} row{'s' if n != 1 else ''}" + (" - UNSAVED CHANGES" if self.dirty else ""))

    def _set_status(self, text: str) -> None:
        self.status.set(text)

    def _clear_form(self) -> None:
        for f in self.spec.fields:
            self.vars[f.name].set("true" if f.kind == "bool" else "")
        self._ref_choices()

    def _on_select(self, _e: Any = None) -> None:
        sel = self.tree.selection()
        if not sel:
            set_form_state(self, False)
            self.selected_index = None
            if getattr(self, "_multi_selection", False):
                self._multi_selection = False
                self._clear_form()
            return
        if len(sel) > 1:
            self._multi_selection = True
            self.selected_index = None
            selected_rows = [self.rows[int(item)] for item in sel]
            for field in self.spec.fields:
                self.vars[field.name].set(selection_display(selected_rows, field.name))
            set_form_state(self, True)
            self._set_status(
                str(len(sel)) + " rows selected - bulk Apply follows in the next slice."
            )
            return
        self._multi_selection = False
        set_form_state(self, False)
        self.selected_index = int(sel[0])
        row = self.rows[self.selected_index]
        for f in self.spec.fields:
            self.vars[f.name].set(row.get(f.name, ""))
        self._ref_choices()

    def new_record(self) -> None:
        self._multi_selection = False
        set_form_state(self, False)
        self.tree.selection_remove(self.tree.selection())
        self.selected_index = None
        self._clear_form()
        next_sort = max([int(r["sort_order"]) for r in self.rows if r.get("sort_order", "").isdigit()] or [0]) + 10
        self.vars["sort_order"].set(str(next_sort))

    def _form_row(self) -> dict[str, str] | None:
        row = {f.name: self.vars[f.name].get().strip() for f in self.spec.fields}
        missing = [f.label for f in self.spec.fields if f.required and not row[f.name]]
        if missing:
            messagebox.showwarning(self.title, "Required fields are blank:\n" + "\n".join(missing))
            return None
        for f in self.spec.fields:
            if f.kind == "int" and (f.required or row[f.name]):
                try:
                    int(row[f.name])
                except ValueError:
                    messagebox.showwarning(self.title, f"{f.label} must be a whole number.")
                    return None
        for f in self.spec.fields:
            if f.kind == "decimal" and row[f.name]:
                try:
                    collection_model_rules.decimal_value(row[f.name])
                except ValueError as error:
                    messagebox.showwarning(self.title, f"{f.label}: {error}")
                    return None
        key = tuple(row[k] for k in self.spec.key_columns)
        for i, r in enumerate(self.rows):
            if tuple(r[k] for k in self.spec.key_columns) == key and i != self.selected_index:
                messagebox.showwarning(self.title, f"{' / '.join(key)} already exists.")
                return None
        return row

    def apply_record(self) -> None:
        if not require_single(self, "Apply to List", messagebox):
            return
        row = self._form_row()
        if row is None:
            return
        if self.selected_index is None:
            self.rows.append(row)
            self.selected_index = len(self.rows) - 1
        else:
            self.rows[self.selected_index] = row
        self.dirty = True
        self._render()
        self.tree.selection_set(str(self.selected_index))

    def delete_record(self) -> None:
        if not require_single(self, "Delete", messagebox):
            return
        if self.selected_index is None:
            return
        key = " / ".join(self.rows[self.selected_index][k] for k in self.spec.key_columns)
        if not messagebox.askyesno(self.title, f"Remove {key} from the draft? (Not saved until Save to Source.)"):
            return
        del self.rows[self.selected_index]
        self.selected_index = None
        self.dirty = True
        self._render()
        self._clear_form()

    def revert(self) -> None:
        if self.dirty and not messagebox.askyesno(self.title, "Discard unsaved changes to this table?"):
            return
        self.load()

    def validate_draft(self) -> bool:
        res = validate_candidate(self.app.project_root, self.spec.table, self.rows)
        self._report(res, "Draft validation")
        return res.valid

    def _report(self, res: network_rules.Result, label: str) -> None:
        for line in res.errors:
            self.app.log(f"[{self.spec.table}] ERROR: {line}")
        for line in res.warnings:
            self.app.log(f"[{self.spec.table}] WARNING: {line}")
        if res.errors:
            messagebox.showerror(self.title, f"{label} failed:\n\n" + "\n".join(res.errors[:12]))
        elif res.warnings:
            messagebox.showwarning(self.title, f"{label} passed with warnings:\n\n" + "\n".join(res.warnings[:12]))
        else:
            self._set_status(f"{label} passed.")

    def save(self) -> None:
        try:
            if _sha(self.path) != self.loaded_sha:
                messagebox.showerror(self.title, f"{self.spec.filename} changed on disk since it was loaded.\n"
                                                 "Revert to reload it, then reapply your edits.")
                return
            res = validate_candidate(self.app.project_root, self.spec.table, self.rows)
            if not res.valid:
                self._report(res, "Save blocked; validation")
                return
            if _sha(self.path) != self.loaded_sha:
                messagebox.showerror(self.title, f"{self.spec.filename} changed during candidate validation.\n"
                                                 "Save blocked. Revert to reload it, then reapply your edits.")
                return
        except (OSError, UnicodeError, csv.Error) as error:
            self.app.log(f"[{self.spec.table}] Pre-save validation failed: {error}")
            messagebox.showerror(
                self.title,
                f"Pre-save validation could not complete for {self.spec.filename}: {error}\n\n"
                "Save was not attempted. The in-memory draft is retained. Check file access and the Activity Log before retrying.",
            )
            return
        try:
            if self.path.is_file():
                self.backup_dir.mkdir(parents=True, exist_ok=True)
                backup = self.backup_dir / f"{self.path.stem}_{datetime.now():%Y%m%d_%H%M%S_%f}.csv"
                shutil.copy2(self.path, backup)
                self.app.log(f"Backup written: {backup}")
            if _sha(self.path) != self.loaded_sha:
                messagebox.showerror(self.title, f"{self.spec.filename} changed while preparing the backup.\n"
                                                 "Save blocked. Revert to reload it, then reapply your edits.")
                return
            write_rows(self.path, self.spec, self.rows)
        except (OSError, UnicodeError, csv.Error) as error:
            self.app.log(f"[{self.spec.table}] Save failed: {error}")
            messagebox.showerror(
                self.title,
                f"Could not save {self.spec.filename}: {error}\n\n"
                "The in-memory draft is retained. Check file access and the Activity Log before retrying.",
            )
            return
        self.app.log(f"Saved {self.spec.filename} ({len(self.rows)} rows).")
        if res.warnings:
            self._report(res, "Saved")
        self.load()
        for other in getattr(self.app, "network_editors", []):
            if other is not self and not other.dirty:
                other.load()

    def update(self):
        path = self.source_path()
        before = self.digest(path)
        with path.open(encoding="utf-8-sig", newline="") as handle:
            rows = [
                dict(row) for row in csv.DictReader(handle)
                if any(row.values())
            ]
        if before != self.digest(path):
            raise ValueError("Task CSV changed during refresh. Refresh again.")
        self.loaded_rows = rows
        self.snapshot_hash = before
        for field in ("capability_id", "priority", "target_type", "target_family_id", "reward_mode"):
            self.widgets[field].configure(values=self.choices(field))
        for item in self.tree.get_children():
            self.tree.delete(item)
        for row in sorted(rows, key=lambda r: (
            int(r["sort_order"]), r["task_template_id"]
        )):
            self.tree.insert(
                "", "end", iid=row["task_template_id"],
                values=tuple(row[field] for field in self.tree["columns"])
            )
        if self.editing_id and self.tree.exists(self.editing_id):
            self.tree.selection_set(self.editing_id)

    def new(self):
        self._multi_selection = False
        set_form_state(self, False)
        self.tree.selection_remove(*self.tree.selection())
        self.editing_id = None
        self.widgets["task_template_id"].configure(state="normal")
        for variable in self.vars.values():
            variable.set("")
        self.description.delete("1.0", "end")
        self.vars["priority"].set("NORMAL" if "NORMAL" in self.choices("priority") else "")
        self.vars["target_type"].set("TERRITORY")
        self.vars["reward_mode"].set("CONTINUOUS")
        self.vars["loop_task"].set("FALSE")
        self.vars["payout_count"].set("0")
        self.vars["active"].set("FALSE")
        self.vars["sort_order"].set(str(max(
            (int(r["sort_order"]) for r in self.loaded_rows), default=0
        ) + 1))

    def on_select(self, _event=None):
        selected = self.tree.selection()
        if not selected:
            if getattr(self, "_multi_selection", False):
                self.new()
            return
        if len(selected) > 1:
            self._multi_selection = True
            self.editing_id = None
            selected_rows = [
                row for row in self.loaded_rows
                if row["task_template_id"] in selected
            ]
            set_form_state(self, False)
            for field, variable in self.vars.items():
                variable.set(selection_display(selected_rows, field))
            self.description.delete("1.0", "end")
            self.description.insert(
                "1.0", selection_display(selected_rows, "description")
            )
            set_form_state(self, True)
            return
        was_multiple = getattr(self, "_multi_selection", False)
        self._multi_selection = False
        set_form_state(self, False)
        if _event is not None and not was_multiple and selected[0] == self.editing_id:
            return
        row = next(
            (r for r in self.loaded_rows
             if r["task_template_id"] == selected[0]), None
        )
        if row is None:
            return
        self.editing_id = row["task_template_id"]
        for field, variable in self.vars.items():
            variable.set(row[field])
        self.description.delete("1.0", "end")
        self.description.insert("1.0", row["description"])
        self.widgets["task_template_id"].configure(state="disabled")

    def revert(self):
        if self.editing_id and self.tree.exists(self.editing_id):
            self.tree.selection_set(self.editing_id)
            self.on_select()
        else:
            self.new()

    def record(self):
        row = {field: var.get().strip() for field, var in self.vars.items()}
        row["description"] = self.description.get("1.0", "end").strip()
        if not re.fullmatch(r"[A-Z][A-Z0-9_]*", row["task_template_id"]):
            raise ValueError("Template ID must be uppercase letters/digits/underscores, starting with a letter.")
        problems = task_schema.errors(row, set(self.choices("priority")))
        if row["capability_id"] and row["capability_id"] not in self.choices("capability_id"):
            problems.append("unknown capability_id")
        if row["target_family_id"] not in self.choices("target_family_id"):
            problems.append("unknown LOCATION station family")
        if problems:
            raise ValueError("; ".join(problems))
        return row

    def candidate_valid(self, candidate):
        with tempfile.TemporaryDirectory(prefix="doda-task-") as temporary:
            root = Path(temporary)
            destination = root / "data_authoring" / "source"
            destination.mkdir(parents=True)
            for filename in FILES:
                shutil.copy2(
                    candidate if filename == "task_templates.csv"
                    else self.app.source_directory / filename,
                    destination / filename
                )
            done = subprocess.run(
                [sys.executable, str(self.app.validator_path),
                 "--root", str(root), "validate"],
                cwd=root, text=True, encoding="utf-8",
                errors="replace", capture_output=True
            )
            return done.returncode == 0, done.stdout + "\n" + done.stderr

    def commit(self, rows):
        path = self.source_path()
        if self.snapshot_hash is None or self.digest(path) != self.snapshot_hash:
            raise ValueError("Task CSV changed externally. Refresh before saving.")
        temporary = path.with_name(f".tasks-{uuid4().hex}.tmp.csv")
        try:
            with temporary.open("w", encoding="utf-8", newline="") as handle:
                writer = csv.DictWriter(
                    handle, fieldnames=FIELDS, lineterminator="\n"
                )
                writer.writeheader()
                writer.writerows(sorted(rows, key=lambda row: (
                    int(row["sort_order"]), row["task_template_id"]
                )))
            valid, output = self.candidate_valid(temporary)
            self.app.log(output)
            if not valid:
                raise ValueError("Candidate validation failed. CSV unchanged; see Activity Log.")
            if self.digest(path) != self.snapshot_hash:
                raise ValueError("Task CSV changed during validation. CSV unchanged.")
            folder = self.backup_directory()
            folder.mkdir(parents=True, exist_ok=True)
            backup = folder / (
                "task_templates_" + datetime.now().strftime("%Y%m%d_%H%M%S_%f")
                + ".csv"
            )
            shutil.copy2(path, backup)
            os.replace(temporary, path)
            self.app.log(f"Task Templates saved. Backup: {backup}")
            self.app.refresh_all(False)
            messagebox.showinfo(
                "Task Templates", f"Candidate validation passed.\nBackup: {backup}",
                parent=self.app
            )
        finally:
            if temporary.exists():
                temporary.unlink()

    def save(self):
        if not require_single(self, "Save Template", messagebox):
            return
        try:
            row = self.record()
            key = row["task_template_id"]
            if self.editing_id and key != self.editing_id:
                raise ValueError("Template ID cannot change after creation.")
            if not self.editing_id and any(
                r["task_template_id"] == key for r in self.loaded_rows
            ):
                raise ValueError("Template ID already exists.")
            rows = [
                dict(r) for r in self.loaded_rows
                if r["task_template_id"] != key
            ] + [row]
            self.commit(rows)
            self.editing_id = key
            self.tree.selection_set(key)
            self.on_select()
        except (OSError, ValueError) as error:
            messagebox.showerror("Task Templates", str(error), parent=self.app)

    def remove(self):
        if not require_single(self, "Remove Template", messagebox):
            return
        if not self.editing_id:
            messagebox.showwarning(
                "Task Templates", "Select a template first.", parent=self.app
            )
            return
        key = self.editing_id
        if not messagebox.askyesno(
            "Remove Template?", f"Remove '{key}'?\nA CSV backup will be created.",
            default=messagebox.NO, icon="warning", parent=self.app
        ):
            return
        try:
            self.commit([
                dict(row) for row in self.loaded_rows
                if row["task_template_id"] != key
            ])
            self.new()
        except (OSError, ValueError) as error:
            messagebox.showerror("Task Templates", str(error), parent=self.app)

    def open_backups(self):
        self.backup_directory().mkdir(parents=True, exist_ok=True)
        self.app.open_folder(self.backup_directory())

    app.task_templates_editor = TaskTemplatesEditor(app)

@dataclass(frozen=True)
class FieldSpec:
    name: str
    label: str
    kind: str = "text"            # text | int | bool | choice | ref
    required: bool = True
    choices: tuple[str, ...] = ()
    ref_table: str = ""           # for kind == "ref": table whose key supplies choices
    multiline: bool = False


@dataclass(frozen=True)
class TableSpec:
    table: str
    filename: str
    key: str
    fields: tuple[FieldSpec, ...] = field(default_factory=tuple)

    @property
    def columns(self) -> list[str]:
        return [f.name for f in self.fields]

    @property
    def key_columns(self) -> tuple[str, ...]:
        """Composite keys are written as 'a+b'."""
        return tuple(self.key.split("+"))

def page_for_tab_text(text: str) -> PageSpec | None:
    """Longest matching prefix wins, so 'Facility Capabilities' beats 'Facilities'."""
    best: PageSpec | None = None
    for page in PAGES:
        if text.startswith(page.tab_text_prefix):
            if best is None or len(page.tab_text_prefix) > len(best.tab_text_prefix):
                best = page
    return best

    unknown = [key for key in parse_selection(raw) if key not in keys]
    return ";".join(
        [key for key in keys if key in checked] + unknown
    )


def evidence_choices(source_directory):
    path = Path(source_directory) / "resources.csv"
    with path.open(encoding="utf-8-sig", newline="") as handle:
        rows = list(csv.DictReader(handle))
    result = []
    for row in rows:
        rid = (row.get("resource_id") or "").strip()
        if not rid.startswith("EVIDENCE_"):
            continue
        if (row.get("resource_kind") or "").strip().upper() != "STOCKPILE":
            continue
        label = (row.get("display_name") or rid).strip()
        if (row.get("active") or "").strip().upper() != "TRUE":
            label += " (inactive)"
        result.append((rid, label))
    return result


class MultiSelectDropdown(ttk.Menubutton):
    def __init__(self, parent, variable):
        self.variable = variable
        self.summary = tk.StringVar(master=parent)
        super().__init__(parent, textvariable=self.summary, width=32)
        self.options = []
        self.flags = {}
        self.menu = tk.Menu(
            self, tearoff=False, background="#1f2933",
            foreground="#e6edf3", activebackground="#264f78",
            activeforeground="#ffffff",
        )
        self.configure(menu=self.menu)
        self.trace_id = self.variable.trace_add("write", self._sync)
        self.set_options([])

    def set_options(self, options):
        self.options = list(options)
        self.flags = {}
        self.menu.delete(0, "end")
        for key, label in self.options:
            flag = tk.BooleanVar(master=self)
            self.flags[key] = flag
            self.menu.add_checkbutton(
                label=label, variable=flag, command=self._commit,
            )
        self.menu.add_separator()
        self.menu.add_command(
            label="Clear selection", command=lambda: self.variable.set("")
        )
        self._sync()

    def _sync(self, *args):
        selected = parse_selection(self.variable.get())
        labels = dict(self.options)
        for key, flag in self.flags.items():
            flag.set(key in selected)
        self.summary.set(
            "; ".join(labels.get(key, key) for key in selected)
            if selected else "Select evidence types..."
        )

    def _commit(self):
        checked = {key for key, flag in self.flags.items() if flag.get()}
        self.variable.set(merged_selection(
            self.variable.get(), self.options, checked
        ))

    def destroy(self):
        try:
            self.variable.trace_remove("write", self.trace_id)
        except tk.TclError:
            pass
        super().destroy()
