"""Generate a read-only ZScript catalog from validated Workbench exports.

Option B bridge (stopgap): display/lookup data only, no simulation rules.
Inputs : generated/data/network_transfer_rules.json, generated/data/equipment_catalog.json,
         generated/data/department_content.json (facilities, capabilities, links, tracks,
         qualifications, staffing defaults, facility staffing overrides, and task templates)
Output : DoDA/Generated/WorkbenchCatalog.zs (plus expected-dump text for comparison)

ZScript features used (local docs, ZDoom_Documentation):
  03.02 Constant Definitions (int const), switch statement, 05.03.02 Console.PrintF,
  05.04.02 EventHandler / 05.04.08 NetworkProcess, 05.04.01 ConsoleEvent.Name.
String static const arrays are avoided because local docs only show int arrays.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import sys
from pathlib import Path

GENERATOR_VERSION = "0.4.0"
NETWORK_JSON = Path("generated/data/network_transfer_rules.json")
EQUIPMENT_JSON = Path("generated/data/equipment_catalog.json")
DEPARTMENT_JSON = Path("generated/data/department_content.json")
OUTPUT_ZS = Path("DoDA/Generated/WorkbenchCatalog.zs")
EXPECTED_DUMP = Path("generated/data/workbench_catalog_expected_dump.txt")
INCLUDE_LINE = '#include "DoDA/Generated/WorkbenchCatalog.zs"'
DUMP_EVENT = "DoDA_CatalogDump"


def find_root(start: Path) -> Path:
    for p in [start, *start.parents]:
        if (p / "zscript.txt").exists() and (p / "data_authoring").exists():
            return p
    raise SystemExit("ERROR: doda-core root not found")


def zs_str(value) -> str:
    s = "" if value is None else str(value)
    s = s.replace("\\", "\\\\").replace('"', '\\"').replace("\r", " ").replace("\n", " ")
    return '"' + s + '"'


def _active(rows):
    return [r for r in rows if r.get("active", True)]


def _sorted(rows, key):
    return sorted(rows, key=lambda r: (r.get("sortOrder", 0), r[key]))


def string_lookup(name: str, values: list[str]) -> str:
    lines = [f"    static String {name}(int i)", "    {", "        switch (i)", "        {"]
    for i, v in enumerate(values):
        lines.append(f"        case {i}: return {zs_str(v)};")
    lines += ["        default: return \"\";", "        }", "    }", ""]
    return "\n".join(lines)


def int_lookup(name: str, values: list[int]) -> str:
    """None -> -1 (not applicable); out-of-range index -> -2."""
    lines = [f"    static int {name}(int i)", "    {", "        switch (i)", "        {"]
    for i, v in enumerate(values):
        lines.append(f"        case {i}: return {-1 if v is None else int(v)};")
    lines += ["        default: return -2;", "        }", "    }", ""]
    return "\n".join(lines)


def find_fn(name: str, count_const: str, id_fn: str) -> str:
    return "\n".join([
        f"    static int {name}(String id)",
        "    {",
        f"        for (int i = 0; i < {count_const}; i++)",
        "        {",
        f"            if ({id_fn}(i) == id) return i;",
        "        }",
        "        return -1;",
        "    }",
        "",
    ])


def load(root: Path):
    net_p, eq_p = root / NETWORK_JSON, root / EQUIPMENT_JSON
    for p in (net_p, eq_p):
        if not p.exists():
            raise ValueError(f"Missing export {p}; run Export All first")
    net, eq = json.loads(net_p.read_bytes()), json.loads(eq_p.read_bytes())
    p = root / DEPARTMENT_JSON
    if not p.exists():
        raise ValueError(f"Missing export {p}; run Export All first")
    dept = json.loads(p.read_bytes())
    return net, eq, content_hash(net), content_hash(eq), dept


def department_catalog(dept):
    """Project only this slice. Reject dangling active links before any write."""
    tables = dept["tables"]
    facilities = _sorted(_active(tables["facilities"]), "facility_id")
    capabilities = _sorted(_active(tables["capabilities"]), "capability_id")
    tracks = {r["track_id"]: r for r in _active(tables["department_tracks"])}
    links = sorted(_active(tables["facility_capabilities"]),
                   key=lambda r: (r["facility_id"], r["sort_order"], r["capability_id"]))
    # Department CSVs use snake_case sort order.
    capabilities.sort(key=lambda r: (r["sort_order"], r["capability_id"]))
    facility_ids = {r["facility_id"] for r in facilities}
    capability_ids = {r["capability_id"] for r in capabilities}
    seen = set()
    for row in facilities:
        if row["track_id"] not in tracks:
            raise ValueError(f"Unknown active track: {row['track_id']}")
    for row in links:
        pair = (row["facility_id"], row["capability_id"])
        if pair in seen or pair[0] not in facility_ids or pair[1] not in capability_ids:
            raise ValueError(f"Invalid/duplicate active facility-capability link: {pair}")
        seen.add(pair)
    rules, effective, qualifications = staffing_catalog(tables, capabilities, links)
    templates = task_template_catalog(tables, capability_ids, seen)
    projected = dict(facilities=facilities, capabilities=capabilities, links=links,
                     tracks=sorted(tracks.values(), key=lambda r: r["track_id"]),
                     staffing_defaults=rules, effective_staffing=effective,
                     task_templates=templates, qualifications=sorted(qualifications.values(),
                                           key=lambda r: r["qualification_id"]))
    return facilities, capabilities, links, tracks, content_hash(projected), rules, effective, qualifications, templates


def task_template_catalog(tables, capability_ids, facility_pairs):
    """Authored templates only. Effort, deadline, and auto-allocation stay metadata."""
    priorities = {r["enum_value"] for r in _active(tables["enum_catalog"])
                  if r["enum_group"] == "TASK_PRIORITY"}
    installed = {pair[1] for pair in facility_pairs}
    templates = sorted(_active(tables["task_templates"]),
                       key=lambda r: (r["sort_order"], r["task_template_id"]))
    seen = set()
    for row in templates:
        key = row["task_template_id"]
        if not key or key in seen:
            raise ValueError(f"Invalid/duplicate active task-template ID: {key}")
        seen.add(key)
        if row["capability_id"] not in capability_ids or row["capability_id"] not in installed:
            raise ValueError(f"Task template references unavailable authored capability: {key}")
        if row["priority"] not in priorities:
            raise ValueError(f"Unknown task priority: {key}")
        for field in ("default_target_effort_minutes", "default_deadline_minutes"):
            if type(row[field]) is not int or row[field] <= 0:
                raise ValueError(f"Task-template {field} must be a positive integer: {key}")
        if type(row["auto_allocate"]) is not bool:
            raise ValueError(f"Task-template auto_allocate must be boolean: {key}")
    return templates


def task_template_lookups(templates):
    parts = []
    for suffix, key in (("Id", "task_template_id"), ("DisplayName", "display_name"),
                        ("Description", "description"), ("CapabilityId", "capability_id"),
                        ("Priority", "priority")):
        parts.append(string_lookup("TaskTemplate" + suffix, [r[key] for r in templates]))
    for suffix, key in (("TargetEffortMinutes", "default_target_effort_minutes"),
                        ("DeadlineMinutes", "default_deadline_minutes")):
        parts.append(int_lookup("TaskTemplate" + suffix, [r[key] for r in templates]))
    parts.append(int_lookup("TaskTemplateAutoAllocate", [int(r["auto_allocate"]) for r in templates]))
    parts.append(find_fn("FindTaskTemplate", "TASK_TEMPLATE_COUNT", "TaskTemplateId"))
    return "\n".join(parts)


def staffing_catalog(tables, capabilities, links):
    """Resolve authored ranges only, by exact composite keys. Never sum people.

    An active override replaces all three counts for one qualification row.
    Missing/inactive overrides inherit the complete global row. Provenance
    remains OVERRIDE even when its values equal the global baseline.
    """
    qualifications = {}
    for row in _active(tables["qualification_catalog"]):
        key = row["qualification_id"]
        if key in qualifications:
            raise ValueError(f"Duplicate active qualification: {key}")
        qualifications[key] = row
    capability_ids = {r["capability_id"] for r in capabilities}
    capability_names = {r["capability_id"]: r["display_name"] for r in capabilities}
    pair_ids = {(r["facility_id"], r["capability_id"]) for r in links}
    rules = sorted(_active(tables["capability_staffing_rules"]),
                   key=lambda r: (r["capability_id"], r["sort_order"], r["qualification_id"]))
    by_key = {}

    def check_range(row):
        values = [row[k] for k in ("min_contributors", "preferred_contributors", "max_contributors")]
        if any(type(v) is not int for v in values) or not 0 <= values[0] <= values[1] <= values[2]:
            raise ValueError(f"Invalid authored staffing range: {row}")

    for row in rules:
        key = (row["capability_id"], row["qualification_id"])
        if key in by_key or key[0] not in capability_ids or key[1] not in qualifications:
            raise ValueError(f"Invalid/duplicate active staffing default: {key}")
        check_range(row)
        by_key[key] = row
    for capability_id in capability_ids:
        if not any(key[0] == capability_id for key in by_key):
            raise ValueError(f"Capability has no active staffing defaults: {capability_id}")
    overrides = {}
    for row in _active(tables["facility_capability_staffing_overrides"]):
        key = (row["facility_id"], row["capability_id"], row["qualification_id"])
        if key in overrides or key[:2] not in pair_ids or key[1:] not in by_key:
            raise ValueError(f"Invalid/duplicate active staffing override: {key}")
        check_range(row)
        overrides[key] = row
    effective = []
    for link in links:
        for rule in rules:
            if rule["capability_id"] != link["capability_id"]:
                continue
            key = (link["facility_id"], rule["capability_id"], rule["qualification_id"])
            selected = overrides.get(key, rule)
            row = dict(facility_id=key[0], capability_id=key[1], qualification_id=key[2],
                       capability_label=link["display_name_override"] or capability_names[key[1]],
                       provenance="OVERRIDE" if key in overrides else "INHERITED")
            for field in ("min_contributors", "preferred_contributors", "max_contributors"):
                row["global_" + field] = rule[field]
                row[field] = selected[field]
            effective.append(row)
    return rules, effective, qualifications


def staffing_lookups(rules, effective, qualifications):
    parts = []
    for prefix, rows in (("StaffingDefault", rules), ("FacilityStaffing", effective)):
        columns = [("CapabilityId", "capability_id"), ("QualificationId", "qualification_id")]
        if prefix == "FacilityStaffing":
            columns += [("FacilityId", "facility_id"), ("Source", "provenance"),
                        ("CapabilityName", "capability_label")]
        for suffix, key in columns:
            parts.append(string_lookup(prefix + suffix, [r[key] for r in rows]))
        parts.append(string_lookup(prefix + "QualificationName", [
            qualifications[r["qualification_id"]]["display_name"] for r in rows]))
        for suffix, key in (("Min", "min_contributors"), ("Preferred", "preferred_contributors"),
                            ("Max", "max_contributors")):
            parts.append(int_lookup(prefix + suffix, [r[key] for r in rows]))
            if prefix == "FacilityStaffing":
                parts.append(int_lookup(prefix + "Global" + suffix, [r["global_" + key] for r in rows]))
    return "\n".join(parts)


def department_lookups(facilities, capabilities, links, tracks):
    parts = []
    def strings(prefix, rows, columns):
        for suffix, key in columns:
            parts.append(string_lookup(prefix + suffix, [r[key] for r in rows]))
    strings("Facility", facilities, [
        ("Id", "facility_id"), ("DisplayName", "display_name"),
        ("Description", "description"), ("TrackId", "track_id"),
        ("ZoneId", "hq_zone_id"), ("InitialTier", "initial_tier_id")])
    parts.append(string_lookup("FacilityTrackName", [
        tracks[r["track_id"]]["display_name"] for r in facilities]))
    parts.append(int_lookup("FacilityCapacity", [r["personnel_capacity"] for r in facilities]))
    parts.append(find_fn("FindFacility", "FACILITY_COUNT", "FacilityId"))
    strings("Capability", capabilities, [
        ("Id", "capability_id"), ("DisplayName", "display_name"),
        ("Description", "description"), ("StandingLabel", "standing_operation_label")])
    parts.append(find_fn("FindCapability", "CAPABILITY_COUNT", "CapabilityId"))
    strings("FacilityCapability", links, [
        ("FacilityId", "facility_id"), ("CapabilityId", "capability_id"),
        ("InitialTier", "initial_tier_id")])
    names = {r["capability_id"]: r["display_name"] for r in capabilities}
    parts.append(string_lookup("FacilityCapabilityDisplayName", [
        r["display_name_override"] or names[r["capability_id"]] for r in links]))
    parts.append(int_lookup("FacilityCapabilityCapacity", [
        r["max_concurrent_personnel"] for r in links]))
    return "\n".join(parts)


def content_hash(doc: dict) -> str:
    """Stable fingerprint: ignores export timestamps so unchanged data keeps the same hash."""
    stable = {k: v for k, v in doc.items() if k != "generatedAtUtc"}
    return hashlib.sha256(json.dumps(stable, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def build(net, eq, net_hash, eq_hash, dept):
    tiers = _sorted(_active(net["tiers"]), "tierId")
    hw = _sorted(_active(net["hardwareModels"]), "modelId")

        string_lookup("FamilyId", [f["familyId"] for f in fams]),
        string_lookup("FamilyDisplayName", [f["displayName"] for f in fams]),
        string_lookup("FamilyOwnership", [f["ownership"] for f in fams]),

        string_lookup("FamilyConnectionKind", [f["connectionKind"] for f in fams]),

        '        Console.PrintF("DODA_CATALOG END");',
        "    }",
        "}",
        "",
    ]
    dump = [f"DODA_CATALOG BEGIN gen={GENERATOR_VERSION} net={net_hash[:16]} eq={eq_hash[:16]} dept={dept_hash[:16]}",
            f"DODA_CATALOG COUNTS tiers={len(tiers)} hardware={len(hw)} families={len(fams)} "
            f"models={len(models)} limit={int(eq['totalPersonalDeviceLimit'])}"]
    dump += [f"DODA_CATALOG TIER {t['tierId']}" for t in tiers]

    dump += [f"DODA_CATALOG MODEL {m['modelId']} {m['familyId']}" for m in models]

    dump += [f"DODA_CATALOG FACILITY {r['facility_id']}" for r in facilities]
    dump += [f"DODA_CATALOG CAPABILITY {r['capability_id']}" for r in capabilities]
    dump += [f"DODA_CATALOG LINK {r['facility_id']} {r['capability_id']}" for r in links]
    dump += [f"DODA_CATALOG STAFFING defaults={len(rules)} effective={len(effective)} overrides={override_count}"]

             f"{r['min_contributors']}/{r['preferred_contributors']}/{r['max_contributors']}" for r in rules]

             f"global={r['global_min_contributors']}/{r['global_preferred_contributors']}/{r['global_max_contributors']} "
             f"effective={r['min_contributors']}/{r['preferred_contributors']}/{r['max_contributors']} "
             f"source={r['provenance']}" for r in effective]
    dump += [f"DODA_CATALOG TASK_TEMPLATES count={len(templates)}"]
    dump += [f"DODA_CATALOG TASK_TEMPLATE {r['task_template_id']} {r['capability_id']} "
             f"priority={r['priority']} effort={r['default_target_effort_minutes']} "
             f"deadline={r['default_deadline_minutes']} auto={int(r['auto_allocate'])}" for r in templates]
    dump.append("DODA_CATALOG END")
    return "\n".join(out), "\n".join(dump) + "\n"


def generate(root: Path) -> tuple[Path, str, bool]:
    """Write catalog + expected dump. Returns (path, sha256, include_present)."""
    zs, dump = build(*load(root))
    target = root / OUTPUT_ZS
    included = INCLUDE_LINE in (root / "zscript.txt").read_text(encoding="utf-8", errors="replace")
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(zs, encoding="utf-8", newline="\n")
    (root / EXPECTED_DUMP).write_text(dump, encoding="utf-8", newline="\n")
    return target, hashlib.sha256(zs.encode()).hexdigest(), included


def is_current(root: Path) -> bool:
    zs, _ = build(*load(root))
    target = root / OUTPUT_ZS
    return target.exists() and target.read_text(encoding="utf-8") == zs


def main(argv=None) -> int:
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--root", type=Path, default=None)
    ap.add_argument("--check", action="store_true", help="fail if generated file is stale; write nothing")
    a = ap.parse_args(argv)
    root = a.root or find_root(Path(__file__).resolve().parent)
    if a.check:
        ok = is_current(root)
        print("RESULT: " + ("PASS (catalog current)" if ok else "FAIL (catalog stale or missing)"))
        return 0 if ok else 1
    target, sha, included = generate(root)
    print(f"Wrote {target} sha256 {sha}")
    print(f"Wrote {root / EXPECTED_DUMP}")
    print("zscript.txt include present: " + ("yes" if included else "NO - add manually"))
    print("RESULT: PASS")
    return 0


if __name__ == "__main__":
    sys.exit(main())

def find_project_root(start: Path) -> Path:
    current = start.resolve()

    for candidate in (current, *current.parents):
        if (
            (candidate / "data_authoring").is_dir()
            and (candidate / "tools").is_dir()
            and (candidate / "DoDA").is_dir()
        ):
            return candidate

    raise FileNotFoundError(
        "Unable to locate DoDA project root. Expected a folder containing "
        "'data_authoring', 'tools', and 'DoDA'. Use --root to provide it explicitly."
    )


def load_csv(
    path: Path,
    required_columns: list[str],
    table_name: str,
    result: ValidationResult,
) -> list[dict[str, str]]:
    if not path.is_file():
        result.error(f"[{table_name}] Missing required source file: {path}")
        return []

    try:
        with path.open("r", encoding="utf-8-sig", newline="") as source_file:
            reader = csv.DictReader(source_file)

            if reader.fieldnames is None:
                result.error(f"[{table_name}] File has no CSV header row: {path.name}")
                return []

            actual_columns = [column.strip() for column in reader.fieldnames]
            missing_columns = [
                column for column in required_columns if column not in actual_columns
            ]

            if missing_columns:
                result.error(
                    f"[{table_name}] Missing required columns in {path.name}: "
                    + ", ".join(missing_columns)
                )
                return []

            rows: list[dict[str, str]] = []

            for line_number, raw_row in enumerate(reader, start=2):
                row = {
                    key.strip(): (value or "").strip()
                    for key, value in raw_row.items()
                    if key is not None
                }

                if not any(row.values()):
                    continue

                row["_line_number"] = str(line_number)
                rows.append(row)

            result.stats[table_name] = len(rows)

            if not rows:
                result.warning(f"[{table_name}] Source file contains no data rows: {path.name}")

            return rows

    except UnicodeDecodeError as error:
        result.error(f"[{table_name}] Could not read UTF-8 CSV {path.name}: {error}")
        return []
    except csv.Error as error:
        result.error(f"[{table_name}] CSV parsing error in {path.name}: {error}")
        return []


def validate_keys(
    table_name: str,
    rows: list[dict[str, str]],
    key_columns: list[str],
    result: ValidationResult,
) -> None:
    seen_keys: dict[tuple[str, ...], int] = {}

    for row in rows:
        line_number = row["_line_number"]
        key = tuple(row.get(column, "").strip() for column in key_columns)

        if any(not value for value in key):
            result.error(
                f"[{table_name}] Line {line_number}: blank key value in "
                + ", ".join(key_columns)
            )
            continue

        if key in seen_keys:
            original_line = seen_keys[key]
            result.error(
                f"[{table_name}] Line {line_number}: duplicate key "
                f"{' / '.join(key)}; first defined on line {original_line}"
            )
        else:
            seen_keys[key] = int(line_number)


def validate_active_values(
    table_name: str,
    rows: list[dict[str, str]],
    result: ValidationResult,
) -> None:
    if not rows or "active" not in rows[0]:
        return

    for row in rows:
        value = row.get("active", "").upper()

        if value not in {"TRUE", "FALSE"}:
            result.error(
                f"[{table_name}] Line {row['_line_number']}: active must be TRUE "
                f"or FALSE, found '{row.get('active', '')}'"
            )


def parse_int(
    table_name: str,
    row: dict[str, str],
    column: str,
    result: ValidationResult,
) -> int | None:
    value = row.get(column, "")

    try:
        return int(value)
    except ValueError:
        result.error(
            f"[{table_name}] Line {row['_line_number']}: {column} must be an "
            f"integer, found '{value}'"
        )
        return None


def parse_bool(
    table_name: str,
    row: dict[str, str],
    column: str,
    result: ValidationResult,
) -> bool | None:
    value = row.get(column, "").upper()

    if value == "TRUE":
        return True

    if value == "FALSE":
        return False

    result.error(
        f"[{table_name}] Line {row['_line_number']}: {column} must be TRUE "
        f"or FALSE, found '{row.get(column, '')}'"
    )
    return None

def build_id_set(
    rows: list[dict[str, str]],
    column: str,
    active_only: bool = False,
) -> set[str]:
    return {
        row[column]
        for row in rows
        if row.get(column, "").strip()
        and (not active_only or row.get("active", "").upper() == "TRUE")
    }


def build_row_index(
    rows: list[dict[str, str]],
    key_columns: list[str],
    active_only: bool = False,
) -> dict[tuple[str, ...], dict[str, str]]:
    index: dict[tuple[str, ...], dict[str, str]] = {}

    for row in rows:
        if active_only and row.get("active", "").upper() != "TRUE":
            continue

        key = tuple(row.get(column, "") for column in key_columns)

        if all(key):
            index[key] = row

    return index


def build_enum_index(
    rows: list[dict[str, str]],
    active_only: bool = False,
) -> dict[str, set[str]]:
    enum_values: dict[str, set[str]] = defaultdict(set)

    for row in rows:
        if active_only and row.get("active", "").upper() != "TRUE":
            continue

        enum_values[row["enum_group"]].add(row["enum_value"])

    return dict(enum_values)


def validate_metrics(
    metrics: list[dict[str, str]],
    enum_values: dict[str, set[str]],
    result: ValidationResult,
) -> dict[str, dict[str, str]]:
    metric_by_id: dict[str, dict[str, str]] = {}

    allowed_polarity = enum_values.get("METRIC_POLARITY", set())
    allowed_display_modes = enum_values.get("METRIC_DISPLAY_MODE", set())

    for row in metrics:
        metric_id = row["metric_id"]
        metric_by_id[metric_id] = row

        minimum = parse_int("metrics", row, "internal_min", result)
        maximum = parse_int("metrics", row, "internal_max", result)
        default_value = parse_int("metrics", row, "default_value", result)

        if minimum is not None and maximum is not None and minimum > maximum:
            result.error(
                f"[metrics] Line {row['_line_number']}: internal_min ({minimum}) "
                f"cannot exceed internal_max ({maximum}) for {metric_id}"
            )

        if (
            minimum is not None
            and maximum is not None
            and default_value is not None
            and not minimum <= default_value <= maximum
        ):
            result.error(
                f"[metrics] Line {row['_line_number']}: default_value "
                f"({default_value}) is outside [{minimum}, {maximum}] for {metric_id}"
            )

        polarity = row["polarity"]
        if polarity not in allowed_polarity:
            result.error(
                f"[metrics] Line {row['_line_number']}: polarity '{polarity}' "
                "does not exist in active enum_catalog METRIC_POLARITY"
            )

        display_mode = row["ui_display_mode"]
        if display_mode not in allowed_display_modes:
            result.error(
                f"[metrics] Line {row['_line_number']}: ui_display_mode "
                f"'{display_mode}' does not exist in active enum_catalog "
                "METRIC_DISPLAY_MODE"
            )

    return metric_by_id


def validate_metric_tiers(
    tiers: list[dict[str, str]],
    metrics_by_id: dict[str, dict[str, str]],
    flavor_text_ids: set[str],
    result: ValidationResult,
) -> None:
    tiers_by_metric: dict[str, list[dict[str, str]]] = defaultdict(list)

    for row in tiers:
        metric_id = row["metric_id"]

        if metric_id not in metrics_by_id:
            result.error(
                f"[metric_tiers] Line {row['_line_number']}: metric_id '{metric_id}' "
                "does not exist in metrics.csv"
            )

        flavor_text_id = row["flavor_text_id"]
        if flavor_text_id not in flavor_text_ids:
            result.error(
                f"[metric_tiers] Line {row['_line_number']}: flavor_text_id "
                f"'{flavor_text_id}' does not exist in flavor_text.csv"
            )

        minimum = parse_int("metric_tiers", row, "min_inclusive", result)
        maximum = parse_int("metric_tiers", row, "max_inclusive", result)

        if minimum is not None and maximum is not None and minimum > maximum:
            result.error(
                f"[metric_tiers] Line {row['_line_number']}: min_inclusive "
                f"({minimum}) cannot exceed max_inclusive ({maximum})"
            )

        tiers_by_metric[metric_id].append(row)

    validate_sort_order("metric_tiers", tiers, result)

    for metric_id, metric_row in metrics_by_id.items():
        metric_tiers = tiers_by_metric.get(metric_id, [])

        if not metric_tiers:
            result.error(f"[metric_tiers] No tiers exist for metric '{metric_id}'")
            continue

        parsed_tiers: list[tuple[int, int, int, dict[str, str]]] = []

        for row in metric_tiers:
            minimum = parse_int("metric_tiers", row, "min_inclusive", result)
            maximum = parse_int("metric_tiers", row, "max_inclusive", result)
            sort_order = parse_int("metric_tiers", row, "sort_order", result)

            if minimum is None or maximum is None or sort_order is None:
                continue

            parsed_tiers.append((minimum, maximum, sort_order, row))

        parsed_tiers.sort(key=lambda item: (item[0], item[1], item[2]))

        metric_minimum = parse_int("metrics", metric_row, "internal_min", result)
        metric_maximum = parse_int("metrics", metric_row, "internal_max", result)

        if metric_minimum is None or metric_maximum is None or not parsed_tiers:
            continue

        first_minimum = parsed_tiers[0][0]
        if first_minimum != metric_minimum:
            result.error(
                f"[metric_tiers] Metric '{metric_id}' begins at {first_minimum}, "
                f"but metrics.csv defines internal_min {metric_minimum}"
            )

        expected_next_minimum = metric_minimum

        for minimum, maximum, _, row in parsed_tiers:
            if minimum < expected_next_minimum:
                result.error(
                    f"[metric_tiers] Line {row['_line_number']}: tier range "
                    f"[{minimum}, {maximum}] overlaps or starts before the "
                    f"expected next value {expected_next_minimum} for {metric_id}"
                )
            elif minimum > expected_next_minimum:
                result.error(
                    f"[metric_tiers] Line {row['_line_number']}: tier range "
                    f"[{minimum}, {maximum}] leaves a gap after "
                    f"{expected_next_minimum - 1} for {metric_id}"
                )

            expected_next_minimum = max(expected_next_minimum, maximum + 1)

        final_maximum = parsed_tiers[-1][1]
        if final_maximum != metric_maximum:
            result.error(
                f"[metric_tiers] Metric '{metric_id}' ends at {final_maximum}, "
                f"but metrics.csv defines internal_max {metric_maximum}"
            )


def validate_facilities(
    facilities: list[dict[str, str]],
    track_ids: set[str],
    enum_values: dict[str, set[str]],
    result: ValidationResult,
) -> dict[str, dict[str, str]]:
    facility_by_id: dict[str, dict[str, str]] = {}
    allowed_tiers = enum_values.get("FACILITY_TIER", set())

    for row in facilities:
        facility_id = row["facility_id"]
        facility_by_id[facility_id] = row

        track_id = row["track_id"]
        tier_id = row["initial_tier_id"]
        personnel_capacity = parse_int("facilities", row, "personnel_capacity", result)

        if track_id not in track_ids:
            result.error(
                f"[facilities] Line {row['_line_number']}: track_id '{track_id}' "
                "does not exist in active department_tracks.csv"
            )

        if tier_id not in allowed_tiers:
            result.error(
                f"[facilities] Line {row['_line_number']}: initial_tier_id "
                f"'{tier_id}' does not exist in active enum_catalog FACILITY_TIER"
            )

        if personnel_capacity is not None and personnel_capacity < 1:
            result.error(
                f"[facilities] Line {row['_line_number']}: personnel_capacity "
                "must be at least 1"
            )

    return facility_by_id


def validate_catalog_sort_orders(
    table_name: str,
    rows: list[dict[str, str]],
    result: ValidationResult,
) -> None:
    minimum_allowed = 0 if table_name == "enum_catalog" else 1

    for row in rows:
        value = parse_int(table_name, row, "sort_order", result)

        if value is not None and value < minimum_allowed:
            result.error(
                f"[{table_name}] Line {row['_line_number']}: sort_order must be "
                f"at least {minimum_allowed}"
            )


def validate_facility_capabilities(
    rows: list[dict[str, str]],
    active_facility_ids: set[str],
    active_capability_ids: set[str],
    facility_by_id: dict[str, dict[str, str]],
    enum_values: dict[str, set[str]],
    result: ValidationResult,
) -> dict[tuple[str, str], dict[str, str]]:
    allowed_tiers = enum_values.get("FACILITY_TIER", set())
    active_pairs: dict[tuple[str, str], dict[str, str]] = {}

    for row in rows:
        facility_id = row["facility_id"]
        capability_id = row["capability_id"]
        maximum = parse_int(
            "facility_capabilities",
            row,
            "max_concurrent_personnel",
            result,
        )

        if facility_id not in active_facility_ids:
            result.error(
                f"[facility_capabilities] Line {row['_line_number']}: facility_id "
                f"'{facility_id}' does not exist in active facilities.csv"
            )

        if capability_id not in active_capability_ids:
            result.error(
                f"[facility_capabilities] Line {row['_line_number']}: capability_id "
                f"'{capability_id}' does not exist in active capabilities.csv"
            )

        if row["initial_tier_id"] not in allowed_tiers:
            result.error(
                f"[facility_capabilities] Line {row['_line_number']}: "
                f"initial_tier_id '{row['initial_tier_id']}' does not exist in "
                "active enum_catalog FACILITY_TIER"
            )

        if maximum is not None and maximum < 1:
            result.error(
                f"[facility_capabilities] Line {row['_line_number']}: "
                "max_concurrent_personnel must be at least 1"
            )

        parent = facility_by_id.get(facility_id)
        if parent is not None and maximum is not None:
            capacity = parse_int("facilities", parent, "personnel_capacity", result)
            if capacity is not None and maximum > capacity:
                result.error(
                    f"[facility_capabilities] Line {row['_line_number']}: "
                    f"max_concurrent_personnel ({maximum}) exceeds facility "
                    f"personnel_capacity ({capacity}) for '{facility_id}'"
                )

        if row.get("active", "").upper() == "TRUE":
            active_pairs[(facility_id, capability_id)] = row

    validate_sort_order("facility_capabilities", rows, result)
    return active_pairs


def validate_staffing_range(
    table_name: str,
    row: dict[str, str],
    result: ValidationResult,
) -> None:
    minimum = parse_int(table_name, row, "min_contributors", result)
    preferred = parse_int(table_name, row, "preferred_contributors", result)
    maximum = parse_int(table_name, row, "max_contributors", result)

    if (
        minimum is not None
        and preferred is not None
        and maximum is not None
        and not 0 <= minimum <= preferred <= maximum
    ):
        result.error(
            f"[{table_name}] Line {row['_line_number']}: contributor counts "
            "must satisfy 0 <= min_contributors <= preferred_contributors "
            "<= max_contributors"
        )


def validate_capability_staffing_rules(
    rows: list[dict[str, str]],
    active_capability_ids: set[str],
    active_qualification_ids: set[str],
    result: ValidationResult,
) -> dict[tuple[str, str], dict[str, str]]:
    active_rules: dict[tuple[str, str], dict[str, str]] = {}

    for row in rows:
        capability_id = row["capability_id"]
        qualification_id = row["qualification_id"]

        if capability_id not in active_capability_ids:
            result.error(
                f"[capability_staffing_rules] Line {row['_line_number']}: "
                f"capability_id '{capability_id}' does not exist in active "
                "capabilities.csv"
            )

        if qualification_id not in active_qualification_ids:
            result.error(
                f"[capability_staffing_rules] Line {row['_line_number']}: "
                f"qualification_id '{qualification_id}' does not exist in active "
                "qualification_catalog.csv"
            )

        validate_staffing_range("capability_staffing_rules", row, result)

        if row.get("active", "").upper() == "TRUE":
            active_rules[(capability_id, qualification_id)] = row

    validate_sort_order("capability_staffing_rules", rows, result)

    active_rules_by_capability: dict[str, int] = defaultdict(int)
    for capability_id, _ in active_rules:
        active_rules_by_capability[capability_id] += 1

    for capability_id in active_capability_ids:
        if active_rules_by_capability.get(capability_id, 0) == 0:
            result.error(
                f"[capability_staffing_rules] No active staffing rule exists "
                f"for active capability '{capability_id}'"
            )

    return active_rules


def validate_facility_capability_staffing_overrides(
    rows: list[dict[str, str]],
    active_facility_capabilities: dict[tuple[str, str], dict[str, str]],
    active_qualification_ids: set[str],
    active_global_rules: dict[tuple[str, str], dict[str, str]],
    result: ValidationResult,
) -> None:
    for row in rows:
        facility_id = row["facility_id"]
        capability_id = row["capability_id"]
        qualification_id = row["qualification_id"]

        if (facility_id, capability_id) not in active_facility_capabilities:
            result.error(
                f"[facility_capability_staffing_overrides] Line "
                f"{row['_line_number']}: facility_id/capability_id pair "
                f"'{facility_id}/{capability_id}' does not exist in active "
                "facility_capabilities.csv"
            )

        if qualification_id not in active_qualification_ids:
            result.error(
                f"[facility_capability_staffing_overrides] Line "
                f"{row['_line_number']}: qualification_id '{qualification_id}' "
                "does not exist in active qualification_catalog.csv"
            )

        if (capability_id, qualification_id) not in active_global_rules:
            result.error(
                f"[facility_capability_staffing_overrides] Line "
                f"{row['_line_number']}: no active global staffing rule exists "
                f"for '{capability_id}/{qualification_id}'"
            )

        validate_staffing_range(
            "facility_capability_staffing_overrides",
            row,
            result,
        )

    validate_sort_order(
        "facility_capability_staffing_overrides",
        rows,
        result,
    )


def validate_task_templates(
    rows: list[dict[str, str]],
    active_capability_ids: set[str],
    active_facility_capabilities: dict[tuple[str, str], dict[str, str]],
    enum_values: dict[str, set[str]],
    result: ValidationResult,
) -> None:
    allowed_priorities = enum_values.get("TASK_PRIORITY", set())
    installed_capability_ids = {
        capability_id
        for _, capability_id in active_facility_capabilities
    }

    for row in rows:
        capability_id = row["capability_id"]
        priority = row["priority"]

        if capability_id not in active_capability_ids:
            result.error(
                f"[task_templates] Line {row['_line_number']}: capability_id "
                f"'{capability_id}' does not exist in active capabilities.csv"
            )

        if priority not in allowed_priorities:
            result.error(
                f"[task_templates] Line {row['_line_number']}: priority "
                f"'{priority}' does not exist in active enum_catalog TASK_PRIORITY"
            )

        target_effort = parse_int(
            "task_templates",
            row,
            "default_target_effort_minutes",
            result,
        )
        deadline = parse_int(
            "task_templates",
            row,
            "default_deadline_minutes",
            result,
        )

        if target_effort is not None and target_effort <= 0:
            result.error(
                f"[task_templates] Line {row['_line_number']}: "
                "default_target_effort_minutes must be greater than 0"
            )

        if deadline is not None and deadline <= 0:
            result.error(
                f"[task_templates] Line {row['_line_number']}: "
                "default_deadline_minutes must be greater than 0"
            )

        parse_bool("task_templates", row, "auto_allocate", result)

        if (
            row.get("active", "").upper() == "TRUE"
            and capability_id in active_capability_ids
            and capability_id not in installed_capability_ids
        ):
            result.error(
                f"[task_templates] Line {row['_line_number']}: active capability "
                f"'{capability_id}' has no active Facility Capability installation"
            )

    validate_sort_order("task_templates", rows, result)


def warn_on_facility_capacity_overcommit(
    facilities: list[dict[str, str]],
    facility_capabilities: list[dict[str, str]],
    result: ValidationResult,
) -> None:
    capacity_by_facility: dict[str, int] = {}

    for row in facilities:
        if row.get("active", "").upper() != "TRUE":
            continue

        capacity = parse_int("facilities", row, "personnel_capacity", result)
        if capacity is not None:
            capacity_by_facility[row["facility_id"]] = capacity

    concurrent_total: dict[str, int] = defaultdict(int)

    for row in facility_capabilities:
        if row.get("active", "").upper() != "TRUE":
            continue

        maximum = parse_int(
            "facility_capabilities",
            row,
            "max_concurrent_personnel",
            result,
        )
        if maximum is not None:
            concurrent_total[row["facility_id"]] += maximum

    for facility_id, total in sorted(concurrent_total.items()):
        capacity = capacity_by_facility.get(facility_id)
        if capacity is not None and total > capacity:
            result.warning(
                f"[facility_capabilities] Active Facility Capability maximum "
                f"concurrency totals {total} for '{facility_id}', exceeding "
                f"facility personnel_capacity {capacity}. This is permitted as "
                "a strategic allocation tradeoff."
            )


def validate_project(project_root: Path) -> ValidationResult:
    result = ValidationResult()
    source_directory = project_root / "data_authoring" / "source"
    loaded_tables: dict[str, list[dict[str, str]]] = {}

    for table_name, specification in TABLE_SPECS.items():
        path = source_directory / specification["filename"]
        rows = load_csv(
            path,
            specification["required_columns"],
            table_name,
            result,
        )
        loaded_tables[table_name] = rows
        validate_keys(table_name, rows, specification["key_columns"], result)
        validate_active_values(table_name, rows, result)

    for table_name in (
        "enum_catalog",
        "department_tracks",
        "qualification_catalog",
        "capabilities",
    ):
        validate_catalog_sort_orders(table_name, loaded_tables[table_name], result)

    enum_values = build_enum_index(
        loaded_tables["enum_catalog"],
        active_only=True,
    )
    track_ids = build_id_set(
        loaded_tables["department_tracks"],
        "track_id",
        active_only=True,
    )
    flavor_text_ids = build_id_set(
        loaded_tables["flavor_text"],
        "flavor_text_id",
        active_only=True,
    )
    active_facility_ids = build_id_set(
        loaded_tables["facilities"],
        "facility_id",
        active_only=True,
    )
    active_qualification_ids = build_id_set(
        loaded_tables["qualification_catalog"],
        "qualification_id",
        active_only=True,
    )
    active_capability_ids = build_id_set(
        loaded_tables["capabilities"],
        "capability_id",
        active_only=True,
    )

    metrics_by_id = validate_metrics(
        loaded_tables["metrics"],
        enum_values,
        result,
    )
    validate_metric_tiers(
        loaded_tables["metric_tiers"],
        metrics_by_id,
        flavor_text_ids,
        result,
    )
    facility_by_id = validate_facilities(
        loaded_tables["facilities"],
        track_ids,
        enum_values,
        result,
    )
    active_facility_capabilities = validate_facility_capabilities(
        loaded_tables["facility_capabilities"],
        active_facility_ids,
        active_capability_ids,
        facility_by_id,
        enum_values,
        result,
    )
    active_global_rules = validate_capability_staffing_rules(
        loaded_tables["capability_staffing_rules"],
        active_capability_ids,
        active_qualification_ids,
        result,
    )
    validate_facility_capability_staffing_overrides(
        loaded_tables["facility_capability_staffing_overrides"],
        active_facility_capabilities,
        active_qualification_ids,
        active_global_rules,
        result,
    )
    validate_task_templates(
        loaded_tables["task_templates"],
        active_capability_ids,
        active_facility_capabilities,
        enum_values,
        result,
    )
    warn_on_facility_capacity_overcommit(
        loaded_tables["facilities"],
        loaded_tables["facility_capabilities"],
        result,
    )

    return result


def write_report(project_root: Path, result: ValidationResult) -> Path:
    report_directory = project_root / "generated" / "reports"
    report_directory.mkdir(parents=True, exist_ok=True)

    report_path = report_directory / "content_validation_report.json"

    report = {
        "toolVersion": TOOL_VERSION,
        "generatedAtUtc": datetime.now(timezone.utc).isoformat(),
        "projectRoot": str(project_root),
        "valid": result.valid,
        "errorCount": len(result.errors),
        "warningCount": len(result.warnings),
        "stats": result.stats,
        "errors": result.errors,
        "warnings": result.warnings,
    }

    report_path.write_text(
        json.dumps(report, indent=2) + "\n",
        encoding="utf-8",
    )

    return report_path


def print_result(project_root: Path, result: ValidationResult, report_path: Path) -> None:
    print()
    print("=" * 72)
    print("DoDA Content Workbench Validation")
    print("=" * 72)
    print(f"Project root: {project_root}")
    print()

    print("Loaded rows:")
    for table_name in sorted(result.stats):
        print(f"  {table_name}: {result.stats[table_name]}")

    print()

    if result.warnings:
        print(f"Warnings ({len(result.warnings)}):")
        for warning in result.warnings:
            print(f"  WARNING: {warning}")
        print()

    if result.errors:
        print(f"Validation failed with {len(result.errors)} error(s):")
        for error in result.errors:
            print(f"  ERROR: {error}")
    else:
        print("Validation passed with zero errors.")

    print()
    print(f"Report written to: {report_path}")
    print("=" * 72)


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="doda_content_workbench",
        description="Validate DoDA authored content.",
    )

    parser.add_argument(
        "--root",
        type=Path,
        default=None,
        help="Path to the doda-core project root. Defaults to auto-detection.",
    )

    subparsers = parser.add_subparsers(dest="command", required=True)

    subparsers.add_parser(
        "validate",
        help="Validate authored CSV source data and write a JSON report.",
    )

    return parser


def main() -> int:
    parser = build_parser()
    arguments = parser.parse_args()

    try:
        project_root = (
            arguments.root.resolve()
            if arguments.root is not None
            else find_project_root(Path.cwd())
        )
    except FileNotFoundError as error:
        parser.error(str(error))
        return 2

    if arguments.command == "validate":
        result = validate_project(project_root)
        report_path = write_report(project_root, result)
        print_result(project_root, result, report_path)
        return 0 if result.valid else 1

    parser.error(f"Unsupported command: {arguments.command}")
    return 2


if __name__ == "__main__":
    sys.exit(main())

def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()

    with path.open("rb") as source_file:
        for chunk in iter(lambda: source_file.read(65536), b""):
            digest.update(chunk)

    return digest.hexdigest()

def normalize_value(table_name: str, column_name: str, value: str) -> Any:
    normalized = value.strip()

    if column_name in INTEGER_COLUMNS.get(table_name, set()):
        return int(normalized)

    if column_name in BOOLEAN_COLUMNS.get(table_name, set()):
        return normalized.upper() == "TRUE"

    return normalized


def load_table(
    project_root: Path,
    table_name: str,
    filename: str,
) -> list[dict[str, Any]]:
    path = project_root / "data_authoring" / "source" / filename

    with path.open("r", encoding="utf-8-sig", newline="") as source_file:
        reader = csv.DictReader(source_file)
        rows: list[dict[str, Any]] = []

        for raw_row in reader:
            if raw_row is None:
                continue

            if not any((value or "").strip() for value in raw_row.values()):
                continue

            row: dict[str, Any] = {}

            for column_name, raw_value in raw_row.items():
                if column_name is None:
                    continue

                row[column_name.strip()] = normalize_value(
                    table_name,
                    column_name.strip(),
                    raw_value or "",
                )

            rows.append(row)

        return rows


def escape_language_text(value: str) -> str:
    return (
        value.replace("\\", "\\\\")
        .replace('"', '\\"')
        .replace("\r\n", "\\n")
        .replace("\n", "\\n")
        .replace("\r", "\\n")
    )


def language_key_from_flavor_id(flavor_text_id: str) -> str:
    safe_key = re.sub(r"[^A-Za-z0-9_]", "_", flavor_text_id.upper())
    return f"DODA_{safe_key}"


def build_language_fragment(flavor_rows: list[dict[str, Any]]) -> str:
    lines = [
        "// GENERATED FILE - DO NOT EDIT BY HAND",
        "// Source: data_authoring/source/flavor_text.csv",
        "// Generated by DoDA Content Workbench.",
        "",
        "[default]",
        "",
    ]

    for row in sorted(flavor_rows, key=lambda item: item["flavor_text_id"]):
        base_key = language_key_from_flavor_id(str(row["flavor_text_id"]))
        title = escape_language_text(str(row["title"]))
        body = escape_language_text(str(row["body"]))

        lines.append(f'{base_key}_TITLE = "{title}";')
        lines.append(f'{base_key}_BODY = "{body}";')
        lines.append("")

    return "\n".join(lines).rstrip() + "\n"


def write_json(path: Path, payload: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps(payload, indent=2, ensure_ascii=False) + "\n",
        encoding="utf-8",
    )


def write_text(path: Path, content: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(content, encoding="utf-8")


def build_pk3(
    pk3_path: Path,
    department_json_path: Path,
    language_path: Path,
    manifest_path: Path,
) -> list[str]:
    pk3_path.parent.mkdir(parents=True, exist_ok=True)

    archive_entries = {
        "data/department_content.json": department_json_path,
        "language/DODA_GENERATED_LANGUAGE.txt": language_path,
        "manifest/content_manifest.json": manifest_path,
    }

    package_readme = (
        "DoDA Generated Content Overlay\n"
        "==============================\n"
        "\n"
        "This PK3 was generated by DoDA Content Workbench.\n"
        "\n"
        "It is currently a data-packaging proof. The archive contains generated\n"
        "department data, a future LANGUAGE fragment, and a build manifest.\n"
        "It does not yet add active runtime behavior because doda-core has not\n"
        "yet been given a native data loader or static include seam for these\n"
        "generated resources.\n"
    )

    temporary_path = pk3_path.with_suffix(".pk3.tmp")

    if temporary_path.exists():
        temporary_path.unlink()

    with zipfile.ZipFile(
        temporary_path,
        mode="w",
        compression=zipfile.ZIP_DEFLATED,
        compresslevel=9,
    ) as archive:
        archive.writestr("README.txt", package_readme)

        for archive_name, source_path in archive_entries.items():
            archive.write(source_path, archive_name)

    if pk3_path.exists():
        pk3_path.unlink()

    temporary_path.replace(pk3_path)

    with zipfile.ZipFile(pk3_path, mode="r") as archive:
        return sorted(archive.namelist())


def export_content(project_root: Path) -> tuple[dict[str, Path], list[str]]:
    run_validator(project_root)

    print()
    print("Loading validated authoring CSV files...")

    tables: dict[str, list[dict[str, Any]]] = {}

    for table_name, filename in TABLE_FILES.items():
        table_rows = load_table(project_root, table_name, filename)
        tables[table_name] = table_rows
        print(f"  {table_name}: {len(table_rows)} row(s)")

    generated_at = datetime.now(timezone.utc).isoformat()

    department_payload = {
        "schemaVersion": SCHEMA_VERSION,
        "toolVersion": TOOL_VERSION,
        "generatedAtUtc": generated_at,
        "tables": tables,
    }

    generated_directory = project_root / "generated"
    department_json_path = generated_directory / "data" / "department_content.json"
    language_path = (
        generated_directory
        / "language"
        / "DODA_GENERATED_LANGUAGE.txt"
    )
    manifest_path = (
        generated_directory
        / "manifest"
        / "content_manifest.json"
    )
    pk3_path = project_root / "build" / "doda-generated-content.pk3"

    write_json(department_json_path, department_payload)

    language_fragment = build_language_fragment(tables["flavor_text"])
    write_text(language_path, language_fragment)

    source_hashes = {
        filename: sha256_file(
            project_root / "data_authoring" / "source" / filename
        )
        for filename in TABLE_FILES.values()
    }

    output_hashes = {
        "generated/data/department_content.json": sha256_file(department_json_path),
        "generated/language/DODA_GENERATED_LANGUAGE.txt": sha256_file(language_path),
    }

    manifest_payload = {
        "schemaVersion": SCHEMA_VERSION,
        "toolVersion": TOOL_VERSION,
        "generatedAtUtc": generated_at,
        "projectRoot": str(project_root),
        "sourceFiles": source_hashes,
        "outputFiles": output_hashes,
        "tableRowCounts": {
            table_name: len(rows)
            for table_name, rows in tables.items()
        },
        "runtimeStatus": {
            "nativeLoaderAvailable": False,
            "mapinfoIntegrationEnabled": False,
            "languageIntegrationEnabled": False,
            "zscriptIntegrationEnabled": False,
            "notes": [
                "This is a generated content overlay proof.",
                "The PK3 is not yet active runtime data.",
                "Do not manually edit generated output files.",
            ],
        },
    }

    write_json(manifest_path, manifest_payload)

    archive_entries = build_pk3(
        pk3_path,
        department_json_path,
        language_path,
        manifest_path,
    )

    outputs = {
        "department_json": department_json_path,
        "language_fragment": language_path,
        "manifest": manifest_path,
        "pk3_overlay": pk3_path,
    }

    return outputs, archive_entries

def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="doda_content_export",
        description="Export validated DoDA authoring content and build a PK3 overlay.",
    )

    parser.add_argument(
        "--root",
        type=Path,
        default=None,
        help="Path to doda-core. Defaults to automatic project-root discovery.",
    )

    return parser


def main() -> int:
    parser = build_parser()
    arguments = parser.parse_args()

    try:
        project_root = (
            arguments.root.resolve()
            if arguments.root is not None
            else find_project_root(Path.cwd())
        )

        outputs, archive_entries = export_content(project_root)
        print_export_result(project_root, outputs, archive_entries)
        return 0

    except (
        FileNotFoundError,
        RuntimeError,
        ValueError,
        OSError,
        zipfile.BadZipFile,
    ) as error:
        print()
        print(f"EXPORT FAILED: {error}", file=sys.stderr)
        return 1


if __name__ == "__main__":
    sys.exit(main())

def enable_windows_dark_title_bar(window: tk.Tk) -> None:
    """Request a dark native Windows title bar for the top-level app frame."""

    if sys.platform != "win32":
        return

    try:
        import ctypes

        window.update_idletasks()

        user32 = ctypes.windll.user32
        dwm_api = ctypes.windll.dwmapi.DwmSetWindowAttribute

        user32.GetAncestor.argtypes = [
            ctypes.c_void_p,
            ctypes.c_uint,
        ]
        user32.GetAncestor.restype = ctypes.c_void_p

        dwm_api.argtypes = [
            ctypes.c_void_p,
            ctypes.c_uint,
            ctypes.c_void_p,
            ctypes.c_uint,
        ]
        dwm_api.restype = ctypes.c_long

        # Tk exposes a client-area handle. DWM title-bar settings must target
        # the top-level frame window instead.
        GA_ROOT = 2
        client_handle = window.winfo_id()
        top_level_handle = user32.GetAncestor(
            ctypes.c_void_p(client_handle),
            GA_ROOT,
        )

        hwnd = top_level_handle or client_handle
        enabled = ctypes.c_int(1)

        # Attribute 20 is used by current Windows 10/11 releases.
        result = dwm_api(
            ctypes.c_void_p(hwnd),
            20,
            ctypes.byref(enabled),
            ctypes.sizeof(enabled),
        )

        if result != 0:
            # Attribute 19 is the Windows 10 1809â€“1909 compatibility value.
            dwm_api(
                ctypes.c_void_p(hwnd),
                19,
                ctypes.byref(enabled),
                ctypes.sizeof(enabled),
            )

    except (AttributeError, OSError, tk.TclError):
        # Unsupported Windows versions retain their standard system title bar.
        pass

def read_csv(path: Path) -> list[dict[str, str]]:
    if not path.is_file():
        return []

    with path.open("r", encoding="utf-8-sig", newline="") as source_file:
        reader = csv.DictReader(source_file)
        rows: list[dict[str, str]] = []

        for raw_row in reader:
            if raw_row is None:
                continue

            row = {
                (key or "").strip(): (value or "").strip()
                for key, value in raw_row.items()
                if key is not None
            }

            if any(row.values()):
                rows.append(row)

        return rows


def truncate_text(value: str, maximum_length: int = 120) -> str:
    normalized = " ".join((value or "").split())

    return normalized[: maximum_length - 3] + "..."

    def open_foims_preview(self) -> None:
        """Open the singleton FOIMS-X preview monitor or raise it if already open."""

        if (
            self.foims_preview is not None and
            self.foims_preview.is_open()
        ):
            self.foims_preview.show()
            self.foims_preview.refresh_preview()
            return

        self.foims_preview = FoimsXPreview(self)
        self.log("Opened FOIMS-X Preview Monitor.")

    def refresh_foims_preview(self) -> None:
        """Refresh an open FOIMS-X monitor without creating a new window."""

        if (
            self.foims_preview is None or
            not self.foims_preview.is_open()
        ):
            self.open_foims_preview()
            return

        self.foims_preview.refresh_preview()
        self.foims_preview.show()
        self.log("Refreshed FOIMS-X Preview Monitor.")

    def toggle_fullscreen(
        self,
        _event: Any | None = None,
    ) -> str:
        """Toggle the main Content Workbench window between normal and full screen."""
        is_fullscreen = bool(self.attributes("-fullscreen"))
        self.attributes("-fullscreen", not is_fullscreen)
        return "break"

    def open_foims_preview_from_shortcut(
        self,
        _event: Any | None = None,
    ) -> str:
        """Open FOIMS-X from Ctrl+O while consuming the Tk key event."""
        self.open_foims_preview()
        return "break"
    def _show_client_menu(
        self,
        event: tk.Event[Any],
        menu: tk.Menu,
    ) -> str:
        """Post a client-area menu directly below the clicked menu button."""

        try:
            menu.tk_popup(
                event.x_root,
                event.y_root + event.widget.winfo_height(),
            )
        finally:
            menu.grab_release()

        return "break"

    def _build_client_menu_bar(self, parent: tk.Widget) -> None:
        """Render a dark, fully functional menu row inside the client area."""

        menu_row = tk.Frame(
            parent,
            background="#161b22",
            highlightthickness=0,
            borderwidth=0,
        )
        menu_row.pack(fill=tk.X, pady=(0, 10))

        menu_items = (
            ("File", self.file_menu),
            ("Actions", self.actions_menu),
            ("Preview", self.preview_menu),
            ("View", self.view_menu),
            ("Folders", self.folders_menu),
            ("Help", self.help_menu),
        )

        for label, menu in menu_items:
            button = tk.Menubutton(
                menu_row,
                text=label,
                menu=menu,
                background="#161b22",
                foreground="#e6edf3",
                activebackground="#264f78",
                activeforeground="#ffffff",
                disabledforeground="#6e7681",
                borderwidth=0,

                anchor=tk.W,
                cursor="hand2",

            )

            button.pack(side=tk.LEFT)

            button.bind(
                "<Button-1>",
                lambda event, target_menu=menu: self._show_client_menu(
                    event,
                    target_menu,

            button.bind(
                "<Return>",

            button.bind(

                    target_menu,
                ),
            )

            button.bind(
                "<Enter>",
                lambda event, target=button: target.configure(
                    background="#264f78",
                    foreground="#ffffff",
                ),
            )

            button.bind(
                "<Leave>",
                lambda event, target=button: target.configure(
                    background="#161b22",
                    foreground="#e6edf3",
                ),
            )
    def select_relative_notebook_tab(self, offset: int) -> str:
        tab_ids = self.notebook.tabs()

        if not tab_ids:
            return "break"

        try:
            current_index = self.notebook.index(self.notebook.select())
        except tk.TclError:
            current_index = 0

        target_index = (current_index + offset) % len(tab_ids)

        self.notebook.select(tab_ids[target_index])
        self.notebook.focus_set()

        return "break"

    def select_previous_notebook_tab(
        self,
        _event: Any = None,
    ) -> str:
        return self.select_relative_notebook_tab(-1)

    def select_next_notebook_tab(
        self,
        _event: Any = None,
    ) -> str:
        return self.select_relative_notebook_tab(1)

    def _build_browser(
        self,
        parent: ttk.Frame,
        title: str,
        columns: tuple[tuple[str, str, int], ...],
        select_handler: Any,
    ) -> tuple[ttk.Treeview, tk.Text]:
        parent.columnconfigure(0, weight=3)
        parent.columnconfigure(1, weight=2)
        parent.rowconfigure(0, weight=1)

        list_frame = ttk.LabelFrame(parent, text=title, padding=8)
        detail_frame = ttk.LabelFrame(parent, text="Record Detail", padding=8)

        list_frame.grid(row=0, column=0, sticky=tk.NSEW, padx=(0, 6))
        detail_frame.grid(row=0, column=1, sticky=tk.NSEW)

        list_frame.columnconfigure(0, weight=1)
        list_frame.rowconfigure(0, weight=1)

        column_ids = tuple(column[0] for column in columns)

        tree = ttk.Treeview(
            list_frame,
            columns=column_ids,
            show="headings",
            selectmode="browse",
        )

        for column_id, heading, width in columns:
            tree.heading(column_id, text=heading)
            tree.column(column_id, width=width, anchor=tk.W)

        scrollbar = ttk.Scrollbar(
            list_frame,
            orient=tk.VERTICAL,
            command=tree.yview,
        )
        tree.configure(yscrollcommand=scrollbar.set)

        tree.grid(row=0, column=0, sticky=tk.NSEW)
        scrollbar.grid(row=0, column=1, sticky=tk.NS)

        detail = tk.Text(
            detail_frame,
            wrap=tk.WORD,
            borderwidth=0,
            highlightthickness=0,
            background="#0d1117",
            foreground="#e6edf3",
            insertbackground="#e6edf3",
            font=("Consolas", 10),
        )

        detail_scrollbar = ttk.Scrollbar(
            detail_frame,
            orient=tk.VERTICAL,
            command=detail.yview,
        )
        detail.configure(yscrollcommand=detail_scrollbar.set)

        detail.pack(side=tk.LEFT, fill=tk.BOTH, expand=True)
        detail_scrollbar.pack(side=tk.RIGHT, fill=tk.Y)

        detail.configure(state=tk.DISABLED)

        tree.bind("<<TreeviewSelect>>", select_handler)

        return tree, detail

    def set_detail(self, detail_widget: tk.Text, record: dict[str, str]) -> None:
        lines: list[str] = []

        for key, value in record.items():
            lines.append(f"{key}:")
            lines.append(f"  {value}")
            lines.append("")

        detail_widget.configure(state=tk.NORMAL)
        detail_widget.delete("1.0", tk.END)
        detail_widget.insert("1.0", "\n".join(lines).rstrip() + "\n")
        detail_widget.configure(state=tk.DISABLED)

    def refresh_all(self, show_message: bool = False) -> None:
        try:
            self.tables = {
                table_name: read_csv(self.source_directory / filename)
                for table_name, filename in SOURCE_FILES.items()

                    result.errors += attribute_result.errors

    def run_command(
        self,
        label: str,
        command: list[str],
        success_message: str,
        announce_success: bool = True,
    ) -> bool:
        self.config(cursor="watch")
        self.update_idletasks()

        self.log(f"Starting {label.lower()}: {' '.join(command)}")

        try:
            completed = subprocess.run(
                command,
                cwd=self.project_root,
                text=True,
                encoding="utf-8",
                errors="replace",
                capture_output=True,
            )

            combined_output = (
                completed.stdout.rstrip()
                + ("\n" if completed.stdout and completed.stderr else "")
                + completed.stderr.rstrip()
            ).strip()

            if combined_output:
                self.log(f"{label} output:")
                for line in combined_output.splitlines():
                    self.log(f"  {line}")

            self.log(f"{label} process exit code: {completed.returncode}")

            self.refresh_all(show_message=False)

            if completed.returncode == 0:
                self.status_text.set(success_message)
                if announce_success:
                    messagebox.showinfo(APP_TITLE, success_message)
                return True
            else:
                self.status_text.set(f"{label} failed.")
                messagebox.showerror(
                    APP_TITLE,
                    f"{label} failed with exit code {completed.returncode}.\n\n"
                    "Review the Activity Log tab for complete output.",
                )
                return False

        except OSError as error:
            self.status_text.set(f"{label} could not start.")
            self.log(f"{label} launch error: {error}")
            messagebox.showerror(
                APP_TITLE,
                f"Could not start {label.lower()}.\n\n{error}",
            )
            return False
        finally:
            self.config(cursor="")
            self.update_idletasks()

@dataclass(frozen=True)
class FieldSpec:
    name: str
    label: str
    kind: str = "text"            # text | int | bool | choice | ref
    required: bool = True
    choices: tuple[str, ...] = ()
    ref_table: str = ""           # for kind == "ref": table whose key supplies choices
    multiline: bool = False


@dataclass(frozen=True)
class TableSpec:
    table: str
    filename: str
    key: str
    fields: tuple[FieldSpec, ...] = field(default_factory=tuple)

    @property
    def columns(self) -> list[str]:
        return [f.name for f in self.fields]

    @property
    def key_columns(self) -> tuple[str, ...]:
        """Composite keys are written as 'a+b'."""
        return tuple(self.key.split("+"))

def page_for_tab_text(text: str) -> PageSpec | None:
    """Longest matching prefix wins, so 'Facility Capabilities' beats 'Facilities'."""
    best: PageSpec | None = None
    for page in PAGES:
        if text.startswith(page.tab_text_prefix):
            if best is None or len(page.tab_text_prefix) > len(best.tab_text_prefix):
                best = page
    return best

"""
DoDA Content Workbench - grouped, resizable, collapsible navigation sidebar.

Builds a grouped tree from the notebook's actual tabs (so no editor is
reinstalled or rebuilt) using doda_page_registry for grouping and labels.
Tabs not in the registry appear under "Other". The notebook tab strip is
hidden while the sidebar is shown; View > Navigation can restore it.
Switching pages never destroys page widgets, so unsaved forms are preserved.
"""

from typing import Any

class GroupedNavigation:
    def __init__(self, app: Any) -> None:
        self.app = app
        self.nb: ttk.Notebook = app.notebook
        self.width = DEFAULT_WIDTH
        self.visible = True
        self.tab_strip_hidden = False
        self.item_to_tab: dict[str, str] = {}
        self.tab_to_item: dict[str, str] = {}
        self.groups: dict[str, list[tuple[str, str]]] = {}
        self.subgroup_of: dict[str, str] = {}

        style = ttk.Style(app)
        self._orig_layout = style.layout("TNotebook.Tab")
        style.layout("Grouped.TNotebook.Tab", [])
        style.configure("Grouped.TNotebook", tabmargins=0)

        self.container = ttk.Frame(self.nb.master, width=self.width)
        self.container.pack_propagate(False)
        self.container.pack(side=tk.LEFT, fill=tk.Y, before=self.nb, padx=(0, 0))
        self.handle = tk.Frame(self.nb.master, width=5, cursor="sb_h_double_arrow", background="#30363d")
        self.handle.pack(side=tk.LEFT, fill=tk.Y, before=self.nb, padx=(2, 8))
        self.handle.bind("<B1-Motion>", self._drag)
        self.handle.bind("<Enter>", lambda _e: self.handle.configure(background="#58a6ff"))
        self.handle.bind("<Leave>", lambda _e: self.handle.configure(background="#30363d"))

        bar = ttk.Frame(self.container)
        bar.pack(fill=tk.X, pady=(0, 4))
        ttk.Label(bar, text="Navigation", style="Section.TLabel").pack(side=tk.LEFT)
        ttk.Button(bar, text="+", width=2, command=lambda: self.set_all_open(True)).pack(side=tk.RIGHT)
        ttk.Button(bar, text="âˆ’", width=2, command=lambda: self.set_all_open(False)).pack(side=tk.RIGHT, padx=(0, 2))

        self.tree = ttk.Treeview(self.container, show="tree", selectmode="browse")
        self.tree_xsb = ttk.Scrollbar(self.container, orient=tk.HORIZONTAL, command=self.tree.xview)
        self.tree.configure(xscrollcommand=self.tree_xsb.set)
        self.tree_xsb.pack(side=tk.BOTTOM, fill=tk.X)
        self.tree.pack(fill=tk.BOTH, expand=True)
        self.tree.bind("<<TreeviewSelect>>", self._on_select)
        self.nb.bind("<<NotebookTabChanged>>", self._sync_from_notebook, add="+")

        self.rebuild()
        self.set_tab_strip_hidden(True)

    # ------------------------------------------------------------------
    def rebuild(self) -> None:
        self.tree.delete(*self.tree.get_children())
        self.item_to_tab.clear()
        self.tab_to_item.clear()
        self.groups = {}
        self.subgroup_of = {}
        for tab_id in self.nb.tabs():
            text = self.nb.tab(tab_id, "text")
            page = page_for_tab_text(text.replace(" - ", " â€” "))
            group = page.group if page else "Other"
            label = page.label if page else text
            self.groups.setdefault(group, []).append((label, tab_id))
            self.subgroup_of[tab_id] = page.subgroup if page else ""
        order = [g for g in GROUP_ORDER if g in self.groups] + (["Other"] if "Other" in self.groups else [])
        for g in order:
            gid = self.tree.insert("", tk.END, text=g, open=True)
            sub_nodes: dict[str, str] = {}
            for label, tab_id in self.groups[g]:
                parent = gid
                sg = self.subgroup_of.get(tab_id, "")
                if sg:
                    if sg not in sub_nodes:
                        sub_nodes[sg] = self.tree.insert(gid, tk.END, text=sg, open=False)
                    parent = sub_nodes[sg]
                iid = self.tree.insert(parent, tk.END, text="  " + label)
                self.item_to_tab[iid] = tab_id
                self.tab_to_item[tab_id] = iid
        self._fit_columns()
        self._sync_from_notebook()

    def _on_select(self, _e: Any = None) -> None:
        sel = self.tree.selection()
        if not sel:
            return
        tab = self.item_to_tab.get(sel[0])
        if tab is None:      # group row: toggle open/closed
            self.tree.item(sel[0], open=not self.tree.item(sel[0], "open"))
            return
        if self.nb.select() != tab:
            self.nb.select(tab)

    def _sync_from_notebook(self, _e: Any = None) -> None:
        item = self.tab_to_item.get(self.nb.select())
        if item and self.tree.selection() != (item,):
            parent = self.tree.parent(item)
            while parent:
                self.tree.item(parent, open=True)
                parent = self.tree.parent(parent)
            self.tree.selection_set(item)
            self.tree.see(item)

    def _drag(self, event: Any) -> None:
        new = self.container.winfo_width() + event.x
        self.width = max(MIN_WIDTH, min(MAX_WIDTH, new))
        self.container.configure(width=self.width)

    # ---- View-menu actions ------------------------------------------
    def set_all_open(self, is_open: bool) -> None:
        for g in self.tree.get_children():
            self._open_recursive(g, is_open)

    def _open_recursive(self, item: str, is_open: bool) -> None:
        self.tree.item(item, open=is_open)
        for child in self.tree.get_children(item):
            self._open_recursive(child, is_open)

    def _fit_columns(self) -> None:
        import tkinter.font as tkfont
        try:
            desc = ttk.Style(self.app).lookup("Treeview", "font") or "TkDefaultFont"
            font = tkfont.Font(root=self.app, font=desc)
        except tk.TclError:
            font = tkfont.nametofont("TkDefaultFont")
        widest = 160

        def walk(item: str, depth: int) -> None:
            nonlocal widest
            widest = max(widest, font.measure(self.tree.item(item, "text")) + 20 * (depth + 1) + 24)
            for child in self.tree.get_children(item):
                walk(child, depth + 1)

        for g in self.tree.get_children():
            walk(g, 0)
        self.tree.column("#0", width=widest, minwidth=widest, stretch=False)

        if self.panel_vars["nav"].get() != self.visible:

    def set_editor_panel(self, which: str) -> None:
        other = "record" if which == "list" else "list"
        want = self.panel_vars[which].get()
        if not want and not self.panel_vars[other].get():
            self.panel_vars[which].set(True)
            return
        self.app.panel_state[which] = want
        for ed in getattr(self.app, "network_editors", []):
            fn = getattr(ed, "set_panel_visible", None)
            if fn:
                fn(which, want)

    def toggle_sidebar(self) -> None:
        if self.visible:
            self.container.pack_forget()
            self.handle.pack_forget()
            self.set_tab_strip_hidden(False)
        else:
            self.container.pack(side=tk.LEFT, fill=tk.Y, before=self.nb)
            self.handle.pack(side=tk.LEFT, fill=tk.Y, before=self.nb, padx=(2, 8))
            self.set_tab_strip_hidden(True)
        self.visible = not self.visible
        pv = getattr(self, "panel_vars", None)
        if pv:
            pv["nav"].set(self.visible)

    def set_tab_strip_hidden(self, hidden: bool) -> None:
        # The native notebook strip squeezes tabs and cannot scroll, so it stays
        # hidden; a scrollable tab bar is shown instead when the sidebar is hidden.
        self.nb.configure(style="Grouped.TNotebook")
        self.tab_strip_hidden = hidden
        if hidden:
            if getattr(self, "tabbar", None) is not None:
                self.tabbar.pack_forget()
        else:
            self._show_tabbar()

    def _show_tabbar(self) -> None:
        if getattr(self, "tabbar", None) is None:
            self._build_tabbar()
        self._fill_tabbar()
        self.tabbar.pack(side=tk.TOP, fill=tk.X, before=self.nb, pady=(0, 6))
        self.tabbar.after(60, self._reveal_tab)

    def _build_tabbar(self) -> None:
        bar = ttk.Frame(self.nb.master)
        canvas = tk.Canvas(bar, height=34, highlightthickness=0, borderwidth=0,
                           background=TAB_SEL_BG, xscrollincrement=40)
        sb = ttk.Scrollbar(bar, orient=tk.HORIZONTAL, command=canvas.xview)
        canvas.configure(xscrollcommand=sb.set)
        canvas.pack(side=tk.TOP, fill=tk.X)
        sb.pack(side=tk.TOP, fill=tk.X)
        inner = tk.Frame(canvas, background=TAB_SEL_BG)
        canvas.create_window((0, 0), window=inner, anchor="nw")
        inner.bind("<Configure>", lambda _e: canvas.configure(scrollregion=canvas.bbox("all")))
        for w in (canvas, inner, sb):
            w.bind("<MouseWheel>", self._tab_wheel)
        self.tabbar, self.tab_canvas, self.tab_inner = bar, canvas, inner
        self.tab_labels = {}
        self.nb.bind("<<NotebookTabChanged>>", self._tabbar_highlight, add="+")

    def _fill_tabbar(self) -> None:
        label_of = {tab: label for items in self.groups.values() for label, tab in items}
        for w in self.tab_inner.winfo_children():
            w.destroy()
        self.tab_labels = {}
        for tab_id in self.nb.tabs():
            text = label_of.get(tab_id) or self.nb.tab(tab_id, "text")
            lbl = tk.Label(self.tab_inner, text=text, padx=14, pady=7, cursor="hand2",
                           background=TAB_BG, foreground=TAB_FG)
            lbl.pack(side=tk.LEFT, padx=(0, 2))
            lbl.bind("<Button-1>", lambda _e, t=tab_id: self.nb.select(t))
            lbl.bind("<Enter>", lambda _e, t=tab_id: self._hover_tab(t, True))
            lbl.bind("<Leave>", lambda _e, t=tab_id: self._hover_tab(t, False))
            lbl.bind("<MouseWheel>", self._tab_wheel)
            self.tab_labels[tab_id] = lbl
        self.tab_inner.update_idletasks()
        self.tab_canvas.configure(height=self.tab_inner.winfo_reqheight())
        self._tabbar_highlight()

    def _paint_tab(self, tab_id: str) -> None:
        lbl = self.tab_labels.get(tab_id)
        if lbl is None:
            return
        if self.nb.select() == tab_id:
            lbl.configure(background=TAB_SEL_BG, foreground=TAB_SEL_FG)
        else:
            lbl.configure(background=TAB_BG, foreground=TAB_FG)

    def _hover_tab(self, tab_id: str, over: bool) -> None:
        lbl = self.tab_labels.get(tab_id)
        if lbl is not None and over and self.nb.select() != tab_id:
            lbl.configure(background=TAB_HOVER)
        else:
            self._paint_tab(tab_id)

    def _tabbar_highlight(self, _e: Any = None) -> None:
        if getattr(self, "tabbar", None) is None or not self.tab_labels:
            return
        for tab_id in self.tab_labels:
            self._paint_tab(tab_id)
        self._reveal_tab()

    def _reveal_tab(self) -> None:
        lbl = self.tab_labels.get(self.nb.select())
        if lbl is None:
            return
        self.tab_canvas.update_idletasks()
        total = max(1, self.tab_inner.winfo_reqwidth())
        view = self.tab_canvas.winfo_width()
        left = self.tab_canvas.canvasx(0)
        x0, x1 = lbl.winfo_x(), lbl.winfo_x() + lbl.winfo_width()
        if x0 < left:
            self.tab_canvas.xview_moveto(max(0.0, (x0 - 8) / total))
        elif x1 > left + view:
            self.tab_canvas.xview_moveto(min(1.0, (x1 + 8 - view) / total))

    def _tab_wheel(self, event: Any) -> str:
        self.tab_canvas.xview_scroll(-3 if event.delta > 0 else 3, "units")
        return "break"

    def go(self, tab_id: str) -> None:
        self.nb.select(tab_id)


def install_grouped_navigation(app: Any) -> GroupedNavigation:
    nav = GroupedNavigation(app)
    app.navigation = nav

    menu = app.view_menu
    opts = {k: menu.cget(k) for k in ("background", "foreground", "activebackground", "activeforeground")}
    nav_menu = tk.Menu(menu, tearoff=False, borderwidth=0, relief=tk.FLAT, **opts)
    nav_menu.add_command(label="Show/Hide Navigation Sidebar", command=nav.toggle_sidebar, accelerator="Ctrl+B")
    nav_menu.add_command(label="Expand All Groups", command=lambda: nav.set_all_open(True))
    nav_menu.add_command(label="Collapse All Groups", command=lambda: nav.set_all_open(False))
    nav_menu.add_separator()
    ordered = [g for g in GROUP_ORDER if g in nav.groups] + [g for g in nav.groups if g not in GROUP_ORDER]
    for group in ordered:
        pages = nav.groups[group]
        sub = tk.Menu(nav_menu, tearoff=False, borderwidth=0, relief=tk.FLAT, **opts)
        for label, tab_id in pages:
            sub.add_command(label=label, command=lambda t=tab_id: nav.go(t))
        nav_menu.add_cascade(label=group, menu=sub)
    menu.add_separator()
    panel_menu = tk.Menu(menu, tearoff=False, borderwidth=0, relief=tk.FLAT,
                         selectcolor=opts["foreground"], **opts)
    app.panel_state = {"list": True, "record": True}
    nav.panel_vars = {k: tk.BooleanVar(master=app, value=True) for k in ("nav", "list", "record")}
    panel_menu.add_checkbutton(label="Panel 1: Navigation", variable=nav.panel_vars["nav"],
                               command=nav.set_nav_panel)
    panel_menu.add_checkbutton(label="Panel 2: List", variable=nav.panel_vars["list"],
                               command=lambda: nav.set_editor_panel("list"))
    panel_menu.add_checkbutton(label="Panel 3: Record", variable=nav.panel_vars["record"],
                               command=lambda: nav.set_editor_panel("record"))
    menu.add_cascade(label="Show / Hide Panel", menu=panel_menu)
    menu.add_cascade(label="Navigation", menu=nav_menu)
    app.bind_all("<Control-b>", lambda _e: (nav.toggle_sidebar(), "break")[1], add="+")
    return nav

from copy import deepcopy
from doda_foims_x_facility_catalog import FacilityCatalog, order, status

def fields(row, identity):
    lines = [
        row.get("display_name") or row.get(identity, ""),
        "ID: " + row.get(identity, ""),
        "Authoring status: " + status(row),
        "CATALOG ONLY - not live personnel or task state."
    ]
    for key, value in row.items():
        if key not in (identity, "display_name", "active", "sort_order"):
            lines.append(
                key.replace("_", " ") + ": "
                + (str(value).strip() or "UNCONFIGURED")
            )
    return lines

def unique(tables, table, key, value):
    rows = [r for r in tables.get(table, []) if r.get(key) == value]
    return rows[0] if len(rows) == 1 else None

def reference_label(tables, table, key, value):
    row = unique(tables, table, key, value)
    if row is None:
        return (value or "BLANK") + " [MISSING OR DUPLICATE]"
    return (row.get("display_name") or value) + " / " + value + " [" + status(row) + "]"

def staffing_line(row):
    counts = [
        str(row.get(k, "")).strip() or "?"
        for k in (
            "min_contributors", "preferred_contributors", "max_contributors"
        )
    ]
    return "Contributors min/preferred/max: " + "/".join(counts) + " [" + status(row) + "]"

class ReferenceCatalog(FacilityCatalog):
    table = ""
    identity = ""

    def facilities(self):
        rows = sorted(
            self.tables.get(self.table, []),
            key=lambda row: order(row, self.identity)
        )
        result = []
        for row in rows:
            item = deepcopy(row)
            item["facility_id"] = item.get(self.identity, "")
            result.append(item)
        return result

    def selected(self, identity):
        return unique(self.tables, self.table, self.identity, identity)

class TaskCatalog(ReferenceCatalog):
    table = "task_templates"
    identity = "task_template_id"

    def describe(self, identity):
        row = self.selected(identity)
        if row is None:
            return ["Task template missing or duplicated; run Workbench validation."]
        lines = fields(row, self.identity)
        lines += [
            "Target effort is work effort, not an estimated completion time.",
            "Auto allocate is an authored flag; this browser does not execute it.",
            "", "CAPABILITY"
        ]
        cid = row.get("capability_id", "")
        lines.append(reference_label(
            self.tables, "capabilities", "capability_id", cid
        ))
        capability = unique(self.tables, "capabilities", "capability_id", cid)
        if capability:
            lines.append(capability.get("description") or "No capability description authored.")

        lines += ["", "QUALIFICATIONS VIA GLOBAL STAFFING RULES"]
        rules = sorted(
            [r for r in self.tables.get("capability_staffing_rules", [])
             if r.get("capability_id") == cid],
            key=lambda r: order(r, "qualification_id")
        )
        if not rules:
            lines.append("No staffing rules authored for this capability.")
        for rule in rules:
            qid = rule.get("qualification_id", "")
            lines.append(reference_label(
                self.tables, "qualification_catalog", "qualification_id", qid
            ))
            lines.append(staffing_line(rule))

        lines += ["", "AUTHORED FACILITY CAPABILITY LINKS"]
        links = sorted(
            [r for r in self.tables.get("facility_capabilities", [])
             if r.get("capability_id") == cid],
            key=lambda r: order(r, "facility_id")
        )
        if not links:
            lines.append("No facility links authored for this capability.")
        for link in links:
            fid = link.get("facility_id", "")
            lines.append(reference_label(
                self.tables, "facilities", "facility_id", fid
            ))
            lines.append("Capability link: " + status(link))
        lines.append("Links do not prove construction, staffing or task readiness.")
        return lines

class QualificationCatalog(ReferenceCatalog):
    table = "qualification_catalog"
    identity = "qualification_id"

    def describe(self, identity):
        row = self.selected(identity)
        if row is None:
            return ["Qualification missing or duplicated; run Workbench validation."]
        lines = fields(row, self.identity)
        lines += ["", "GLOBAL CAPABILITY STAFFING REFERENCES"]
        rules = sorted(
            [r for r in self.tables.get("capability_staffing_rules", [])
             if r.get("qualification_id") == identity],
            key=lambda r: order(r, "capability_id")
        )
        if not rules:
            lines.append("No global staffing references authored.")
        capability_ids = set()
        for rule in rules:
            cid = rule.get("capability_id", "")
            capability_ids.add(cid)
            lines.append(reference_label(
                self.tables, "capabilities", "capability_id", cid
            ))
            lines.append(staffing_line(rule))

        lines += ["", "FACILITY-SPECIFIC STAFFING OVERRIDES"]
        overrides = sorted(
            [r for r in self.tables.get("facility_capability_staffing_overrides", [])
             if r.get("qualification_id") == identity],
            key=lambda r: (r.get("facility_id", ""), order(r, "capability_id"))
        )
        if not overrides:
            lines.append("No facility-specific overrides authored.")
        for override in overrides:
            fid, cid = override.get("facility_id", ""), override.get("capability_id", "")
            capability_ids.add(cid)
            lines.append(reference_label(
                self.tables, "facilities", "facility_id", fid
            ))
            lines.append(reference_label(
                self.tables, "capabilities", "capability_id", cid
            ))
            lines.append(staffing_line(override))

        lines += ["", "RELATED TASKS VIA CAPABILITY STAFFING"]
        tasks = sorted(
            [r for r in self.tables.get("task_templates", [])
             if r.get("capability_id") in capability_ids],
            key=lambda r: order(r, "task_template_id")
        )
        if not tasks:
            lines.append("No related task templates authored.")
        for task in tasks:
            lines.append(
                (task.get("display_name") or task.get("task_template_id", ""))
                + " / " + task.get("task_template_id", "")
                + " [" + status(task) + "]"
            )
        lines.append("Indirect references are not a live eligibility check.")
        return lines

# 3. Your First Guided Workflow

## 3.1 Goal and precautions

This exercise changes one display name on an existing task template. It teaches selection, form editing, reversion, saving, validation, and export without adding new dependency relationships.

> Recommended practice: use a separate complete project copy for this exercise. A successful save changes the CSV under the displayed project root. Export overwrites generated outputs at that root. Do not experiment in production data unless you intend to make those changes.

Before proceeding, confirm:

- The correct project root is displayed.
- The project contains SIGNAL_REVIEW, or you have selected another existing task.
- No other application is editing the same CSVs.
- You have recorded the original display name.

For the supplied snapshot, the original name is Priority Signal Review. The temporary exercise name is Priority Signal Review - Training.

## 3.2 Validate the starting data

1. Choose Actions > Run Validation.
2. Read the completion notification and process output in Activity Log.
3. If validation fails, stop the exercise and resolve the reported source-data errors first.
4. Review warnings separately from errors. A valid result means no errors; warnings may still be present.

The validation report is written to:

```text
generated/reports/content_validation_report.json
```

Choose Folders > Open Validation Reports Folder to locate it.

Validation checks saved source data. It does not test arbitrary unsaved text currently typed into a form.

## 3.3 Make a small edit

1. Open the Task Templates editor.
2. Select SIGNAL_REVIEW.
3. Find Display Name.
4. Replace Priority Signal Review with Priority Signal Review - Training.
5. Leave the ID, capability, priority, timing fields, sort order, and checkboxes unchanged.

At this point, the change exists only in the form. You have not completed a save.

## 3.4 Revert an unsaved edit

1. Click Revert Form.
2. Confirm Display Name returns to the original value.
3. Enter Priority Signal Review - Training again.

Revert Form reloads the selected record from the editor's loaded rows. It does not retrieve a historical backup and does not undo a completed save.

## 3.5 Save and verify

1. Click Save Template.
2. Wait for validation and the save result.
3. Confirm the success message reports candidate validation passed and provides a backup path.
4. Confirm the edited name appears in the record list.
5. Select another record, then select SIGNAL_REVIEW again.
6. Confirm the edited name is still populated.
7. Choose Actions > Run Validation to check the saved project again.

### What happens during a Task Templates save

| Stage | What the editor does |
|---|---|
| Form checks | Checks required text, ID format, selected choices, and positive numeric fields |
| Duplicate/identity checks | Rejects a new duplicate ID or a changed existing ID |
| Initial change guard | Checks that the source CSV matches the loaded snapshot |
| Candidate creation | Writes sorted task rows to a temporary CSV |
| Project validation | Validates a temporary project containing all twelve tables |
| Second change guard | Checks that the task CSV did not change during validation |
| Backup | Copies the previous task CSV to its backup directory |
| Replacement | Replaces the task CSV with the validated candidate |
| Refresh | Calls the application refresh and displays success |

Task backups are stored under:

```text
data_authoring/backups/task_templates/
```

Click Open Backups to open that directory. Backup names include a timestamp. Opening the directory does not restore a backup.

A form-validation or candidate-validation rejection does not mean the proposed values were saved. Read the message and Activity Log before retrying.

## 3.6 Export saved content

1. Confirm the intended changes have been saved.
2. Choose Actions > Export Generated Content.
3. Wait for the exporter to complete.
4. Read the export process output and completion result.
5. If validation fails, correct the CSV source and retry. The exporter runs validation before generating outputs.

Export reads CSV files, not unsaved forms. Saving does not automatically make the generated JSON or PK3 current.

The exporter is not a general rollback transaction for all generated files. This introductory procedure does not promise that an export failure automatically restores every prior output. Keep backups where preservation of a prior build matters.

## 3.7 Locate the outputs

Use Folders > Open Generated Output Folder and Folders > Open Build Folder.

| Output | Path relative to the project root | Purpose |
|---|---|---|
| Department JSON | generated/data/department_content.json | Normalized records from all twelve tables |
| LANGUAGE fragment | generated/language/DODA_GENERATED_LANGUAGE.txt | Generated title/body language entries from flavor text |
| Manifest | generated/manifest/content_manifest.json | Versions, timestamp, row counts, hashes, and runtime status |
| PK3 overlay | build/doda-generated-content.pk3 | Packaged generated content |

The JSON and manifest use schema version 2 and exporter tool version 0.1.1. The GUI's application version and validator version are different identifiers; do not expect them all to match.

The PK3 contains:

```text
README.txt
data/department_content.json
language/DODA_GENERATED_LANGUAGE.txt
manifest/content_manifest.json
```

Do not manually change these files to fix authored content. Correct the CSV source through the editor and export again.

### CSV and JSON example

The following is a partial CSV extract, not a complete importable task record:

```csv
task_template_id,default_target_effort_minutes,default_deadline_minutes,auto_allocate
SIGNAL_REVIEW,240,120,TRUE
```

```json

  "task_template_id": "SIGNAL_REVIEW",
  "default_target_effort_minutes": 240,

```

## 3.8 Restore the original example value

If you want the project to finish with its original display name:

1. Select SIGNAL_REVIEW.
2. Change Display Name back to the original value you recorded.
3. Click Save Template and confirm success.
4. Run Actions > Run Validation.
5. Run Actions > Export Generated Content again if you exported the training name earlier.

This restores the selected field value through another validated save. It does not restore the old file byte-for-byte: save ordering, backup history, and export timestamps can differ.

## 3.9 Troubleshooting

| Message or symptom | Meaning | Next action |
|---|---|---|
| Template ID already exists | A new record uses an existing ID | Edit the existing record or choose a genuinely new ID |
| Template ID cannot change after creation | Existing record identity was changed | Retain its original ID |
| Select a valid capability_id or priority | The value is not an allowed active choice | Choose a valid dropdown entry |
| A numeric field must be at least 1 | The field is nonpositive or not a usable integer | Enter a positive whole number |
| Candidate validation failed | The proposed project violates a validator rule | Read Activity Log and fix the identified field or dependency |
| Task CSV changed externally | The loaded snapshot no longer matches the file | Record your unsaved edits, refresh source data, reselect the record, and reapply only the intended changes |
| Task CSV changed during validation | The file changed while the candidate was checked | Stop concurrent editing; refresh and review before retrying |
| Export validation fails | Saved source data is invalid | Fix source data, not generated files |
| Export succeeds but game behavior is unchanged | Runtime integration is not enabled by this exporter | Treat the package as a data-only overlay proof |

Refresh can repopulate editor controls. Copy any unsaved text you want to retain before refreshing; do not assume it will be preserved.

- I validated saved source data.
- I exported and located the generated outputs.
- I understand that export does not enable runtime integration.
- I restored the original display name if this was only a training exercise.

## 4.2 Common controls and save behavior

The six reviewed editors use a record list and a form. Selecting a record loads its saved values into the form. Key fields are disabled for existing records.

| Editor | Prepare new form | Save | Request deletion |
|---|---|---|---|
| Department Tracks | Add Track | Save Track | Remove Track |
| Metrics | Add Metric | Save Metric | Remove Metric |
| Metric Tiers | Add Tier | Save Tier | Remove Tier |
| Facilities | Add Facility | Save Facility | Remove Facility |
| Flavor Text | Add Flavor Text | Save Flavor Text | Remove Flavor Text |
| Enum Catalog | Add Enum Value | Save Enum Value | Remove Enum Value |

Add prepares a new-record form; it does not immediately add a saved row. All six provide Revert Form and Open Backups. Revert reloads the selected record or resets a new form. It does not undo a completed save.

### Save sequence

1. The editor checks the form fields.
2. It builds a candidate CSV containing the proposed change.
3. It validates a complete temporary project using all twelve tables.
4. If candidate validation fails, it leaves the real source CSV unchanged.
5. If validation passes, it creates a timestamped backup and replaces the source CSV.
6. It refreshes the application and runs post-save validation.
7. It reports success or a post-save warning.

A post-save validation warning differs from candidate rejection: the CSV has already been replaced. Stop additional edits and inspect Activity Log and the reported backup. Do not assume an automatic rollback.

These six implementations do not contain Task Templates' source-hash checks before and after validation. Avoid external edits and multiple Workbench sessions; do not assume identical concurrent-change protection across all editors.

Existing-record keys cannot be renamed through these forms. Change a display label or name when only the readable wording needs to change.

### ID format

For the IDs checked by these forms, begin with an uppercase letter and use uppercase letters, digits, and underscores thereafter.

| Valid illustrative IDs | Invalid illustrative IDs | Reason |
|---|---|---|
| TRAINING_TRACK | training_track | Lowercase letters |
| TRAINING_2 | 2_TRAINING | Starts with a digit |
| FLV_TRAINING_NOTE | TRAINING-NOTE | Hyphen |

This pattern does not automatically apply to every text field. Owner System, Category, descriptive text, and facility linkage fields have their own checks.

### Creation and range-change limitation

Every metric must have complete tier coverage. Consequently, Add Metric followed by Save Metric can be rejected because the new metric has no tiers. Adding its tiers first can also be rejected because the metric does not yet exist.

The reviewed forms save one table at a time; they do not expose a coordinated metric-plus-tiers transaction. Do not present sequential creation as a guaranteed working workflow. Creating a new metric and its full tier set requires a coordinated authoring procedure outside these individual save operations, to be documented and verified separately.

Changing an existing metric's bounds can similarly fail while its current tier coverage still matches the old bounds. A valid-looking form can therefore be blocked by cross-table validation.

### Removal limitation

Remove Metric asks for confirmation and validates the candidate project. Existing tier references prevent removal. Removing tiers first is not a general solution because missing tier coverage is itself invalid. Treat metric retirement as a coordinated data change, not a beginner deletion exercise.

### Understand inclusive ranges

Both endpoints belong to a tier. In the supplied PREPAREDNESS snapshot:

| Tier | Minimum | Maximum | Sort order |
|---|---:|---:|---:|
| FRAGILE | 0 | 200 | 1 |
| OPERATIONAL | 201 | 500 | 2 |
| HARDENED | 501 | 800 | 3 |
| OVEREXTENDED | 801 | 1000 | 4 |

Partial explanatory CSV extract, not a complete importable tier file:


|---|---|

| Start OPERATIONAL at 202 | Leaves 201 uncovered |

The editor handles CSV quoting when saving. You do not need to add CSV quotation marks around text in the form. Commas or line breaks in Body should be entered as ordinary text; generated LANGUAGE output escapes special characters.

The record list shortens long titles for display. Select the row to read the full title in the form rather than assuming the stored text was truncated.

Referenced flavor text cannot be deleted while metric tiers still use its ID. Reassign a tier to another valid flavor record and save that change before attempting removal.

The current exporter iterates all flavor rows when building its LANGUAGE fragment; it does not filter them by Active. Do not use deactivation as a guarantee that text disappears from exported artifacts.

## 4.8 Enum catalog administration

Enum Catalog defines group/value pairs used for controlled choices. It is an administrative editor, not a mechanism for implementing a new game feature.

Tab label verified in its source: Enum Catalog â€” Admin.

Source CSV: data_authoring/source/enum_catalog.csv

Record key: enum_group + enum_value

| Form field | CSV field | Entry rule or purpose |
|---|---|---|
| Enum Group | enum_group | Uppercase identifier; existing suggestion or new text |
| Enum Value | enum_value | Uppercase identifier; existing suggestion or new text |
| Display Label | display_label | Required readable label |
| Description | description | Required explanatory text |
| Sort Order | sort_order | Integer of at least 0; unlike most other ordered tables, zero is allowed |
| Active | active | Checkbox |

The filter above the record list also reads Enum Group. That filter changes which records are displayed; it does not edit their group or deactivate hidden records. Choose All Groups to see the whole list.

Group and Value are disabled for an existing record. Add Enum Value prepares a new form; both identifiers are editable for new records.

### Inspect task priorities safely

1. Open Enum Catalog â€” Admin.
2. Set the list filter to TASK_PRIORITY.
3. Select a priority row.
4. Read its identifiers, label, description, sort order, and Active setting.
5. Do not change the Group or Value of an existing row.
6. Set the filter back to All Groups when finished.

Adding a group that no application component consumes will not create a new dropdown elsewhere. Adding a value may be insufficient where an editor separately restricts supported choices, as Metrics currently does.

### Administrative edit procedure

1. Validate the current project.
2. Filter to the intended group and select the exact row.
3. Record its original label and description.
4. Change only Display Label or Description for a first exercise.
5. Click Save Enum Value.
6. Read the save and post-save validation result.
7. Inspect the relevant dependent editor choices before using the changed catalog in further work.

Deletion and deactivation are subject to full candidate validation. A value still used by facilities, metrics, or tasks may be required to remain active.

### Dependent-choice refresh caveat

The uploaded module defines refresh_dependents_after_catalog_save() with a call to itself instead of a dependent-refresh operation. If invoked, this is self-recursive and does not implement the intended refresh. The reviewed save method calls the application's refresh_all(), not that helper directly, so this finding alone does not prove catalog saves fail.

If choices appear stale, preserve unsaved form text, use Actions > Refresh Source Data, then inspect again. Capture any logged errors. Do not repeatedly save or assume a new catalog value is immediately supported everywhere.

## 4.9 Dependency planning

| Proposed change | Check first | Why |
|---|---|---|
| Remove or deactivate a track | Facilities using track_id | Facilities require active tracks |
| Change metric bounds | Every tier for that metric | Tier coverage must match both endpoints |
| Add or remove a metric | Metric and complete tier set together | One-table saves can be blocked by circular validation requirements |
| Move a tier boundary | Neighboring intervals | The intermediate state may contain a gap or overlap |
| Remove flavor text | Metric tiers using flavor_text_id | Referenced text must exist |
| Lower facility capacity | Installed capability concurrency | Each installation limit must fit |
| Remove a facility | Installations and staffing overrides | Child references can block deletion |
| Remove or deactivate an enum | Fields consuming that group/value | Referenced controlled values can be required active |

Not every valid final design can be reached through a sequence of individually valid one-record saves. Stop and plan coordinated changes rather than trying to force an invalid intermediate state through the editor.

## 4.10 Backup and troubleshooting reference

All paths below are relative to the project root:

| Editor | Backup directory |
|---|---|
| Department Tracks | data_authoring/backups/department_tracks |
| Metrics | data_authoring/backups/metrics |
| Metric Tiers | data_authoring/backups/metric_tiers |
| Facilities | data_authoring/backups/facilities |
| Flavor Text | data_authoring/backups/flavor_text |
| Enum Catalog | data_authoring/backups/enum_catalog |

Successful removals generally use filenames containing before_remove. Backups copy an entire table CSV, not just the selected row. Restoring one can affect unrelated records saved afterward.

| Symptom | Response |
|---|---|
| Add was clicked but no row exists | Fill the new form and click its Save button |
| Duplicate key error | Select the existing record or choose a genuinely new key |
| Candidate validation fails | Read Activity Log; another table may be responsible |
| Post-save validation fails | Stop editing; the source may already be changed; retain the backup path |
| Numeric range edit fails | Review metric bounds and complete tier coverage |
| Remove fails | Inspect dependencies and required coverage |
| Tier controls overlap | Report the layout concern; do not rely on inaccessible controls |
| Enum changes do not appear | Preserve unsaved text, refresh, and inspect supported choices |
| A linkage ID saves but has no game effect | A nonempty identifier is not proof of runtime integration |

Open Backups only opens the directory. Backup restoration will be documented as a separate, deliberate recovery operation after that procedure is reviewed.

## 4.11 Practice checklist

In a separate practice project:

1. Create TRAINING_TRACK and validate.
2. Create TRAINING_ROOM referring to that track and validate.
3. Create the standalone FLV_TRAINING_NOTE and validate.
4. Inspect PREPAREDNESS and its four tiers without changing their ranges.
5. Inspect TASK_PRIORITY through the Enum Catalog filter without editing.
6. Remove the unreferenced training note if no longer needed.
7. Remove TRAINING_ROOM before removing TRAINING_TRACK.
8. Validate after cleanup.
9. Export only if you need generated outputs updated to the practice project's final saved state.

If removal is blocked, read the exact validation errors instead of deleting additional records indiscriminately.
