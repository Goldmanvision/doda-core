@dataclass(frozen=True)
class FieldSpec:
    name: str
    label: str
    kind: str = "text"            # text | int | bool | choice | ref
    required: bool = True
    choices: tuple[str, ...] = ()
    ref_table: str = ""           # for kind == "ref": table whose key supplies choices
    multiline: bool = False


@dataclass(frozen=True)
class TableSpec:
    table: str
    filename: str
    key: str
    fields: tuple[FieldSpec, ...] = field(default_factory=tuple)

    @property
    def columns(self) -> list[str]:
        return [f.name for f in self.fields]

    @property
    def key_columns(self) -> tuple[str, ...]:
        """Composite keys are written as 'a+b'."""
        return tuple(self.key.split("+"))

def page_for_tab_text(text: str) -> PageSpec | None:
    """Longest matching prefix wins, so 'Facility Capabilities' beats 'Facilities'."""
    best: PageSpec | None = None
    for page in PAGES:
        if text.startswith(page.tab_text_prefix):
            if best is None or len(page.tab_text_prefix) > len(best.tab_text_prefix):
                best = page
    return best

def _sha(path: Path) -> str | None:
    return hashlib.sha256(path.read_bytes()).hexdigest() if path.is_file() else None


def read_rows(path: Path, spec: TableSpec) -> list[dict[str, str]]:
    if not path.is_file():
        return []
    with path.open("r", encoding="utf-8-sig", newline="") as fh:
        reader = csv.DictReader(fh)
        rows = []
        for raw in reader:
            row = {c: (raw.get(c) or "").strip() for c in spec.columns}
            if any(row.values()):
                rows.append(row)
    return rows


def write_rows(path: Path, spec: TableSpec, rows: list[dict[str, str]]) -> None:
    tmp = path.with_suffix(path.suffix + ".tmp")
    with tmp.open("w", encoding="utf-8", newline="") as fh:
        w = csv.DictWriter(fh, fieldnames=spec.columns, lineterminator="\r\n")
        w.writeheader()
        for r in rows:
            w.writerow({c: r.get(c, "") for c in spec.columns})
    tmp.replace(path)


def validate_candidate(project_root: Path, table: str, rows: list[dict[str, str]]) -> network_rules.Result:
    """Validate the candidate table together with the other current network table."""
    src = project_root / "data_authoring" / "source"
    tmp_root = Path(tempfile.gettempdir()) / f"doda_net_candidate_{datetime.now():%Y%m%d%H%M%S%f}"
    tmp_src = tmp_root / "data_authoring" / "source"
    tmp_src.mkdir(parents=True)
    (tmp_root / "tools").mkdir()
    try:
        for other in TABLE_METADATA.values():
            if other.table == table:
                write_rows(tmp_src / other.filename, other, rows)
            elif (src / other.filename).is_file():
                shutil.copy2(src / other.filename, tmp_src / other.filename)
        track_path = src / "department_tracks.csv"
        if track_path.is_file():
            shutil.copy2(track_path, tmp_src / track_path.name)
        res = network_rules.Result()
        network_rules.validate_contract(res)
        network_rules.validate_network(tmp_root, res)
        equipment_rules.validate_equipment(tmp_root, res)
        care_rules.validate_care(tmp_root, res)
        condition_rules.validate_conditions(tmp_root, res)
        attribute_rules.validate_attributes(tmp_root, res)
        personal_care_rules.validate_personal_care(tmp_root, res)
        time_rules.validate_time_settings(tmp_root, res)
        rc_rules.validate_recreation_content(tmp_root, res)
        sobriety_rules.validate_sobriety(tmp_root, res)
        territory_rules.validate_territory(tmp_root, res)
        weapon_rules.validate_weapons(tmp_root, res)
        hostile_rules.validate_hostiles(tmp_root, res)
        progression_identity_rules.validate_identities(tmp_root, res)
        authored_mission_rules.validate_authored_missions(tmp_root, res)
        return res
    finally:
        shutil.rmtree(tmp_root, ignore_errors=True)



def reference_keys(source_directory, field):
    if field.kind == "track_ref":
        path = source_directory / "department_tracks.csv"
        if not path.is_file():
            return []
        with path.open(encoding="utf-8-sig", newline="") as fh:
            keys = [(row.get("track_id") or "").strip() for row in csv.DictReader(fh)]
        return list(dict.fromkeys(key for key in keys if key))
    ref = TABLE_METADATA[field.ref_table]
    return [row[ref.key] for row in read_rows(source_directory / ref.filename, ref)]

class NetworkTableEditor:
    def __init__(self, app: Any, tab: ttk.Frame, table: str, title: str) -> None:
        self.app = app
        self.tab = tab
        self.spec = TABLE_METADATA[table]
        self.title = title
        self.rows: list[dict[str, str]] = []
        self.loaded_sha: str | None = None
        self.dirty = False
        self.selected_index: int | None = None
        self.vars: dict[str, tk.StringVar] = {}
        self.widgets: dict[str, Any] = {}
        self._build()

    # ---- paths -------------------------------------------------------
    @property
    def path(self) -> Path:
        return self.app.source_directory / self.spec.filename

    @property
    def backup_dir(self) -> Path:
        return self.app.project_root / "data_authoring" / "backups" / self.spec.table

    # ---- UI ----------------------------------------------------------
    def _build(self) -> None:
        tab = self.tab
        tab.columnconfigure(0, weight=3)
        tab.columnconfigure(1, weight=2, minsize=430)
        tab.rowconfigure(1, weight=1)

        self.status = tk.StringVar(value="")
        ttk.Label(tab, text=self.title, style="Section.TLabel").grid(row=0, column=0, sticky=tk.W)
        ttk.Label(tab, textvariable=self.status, foreground="#9da7b3").grid(row=0, column=1, sticky=tk.E)

        pane = tk.PanedWindow(tab, orient=tk.HORIZONTAL, sashwidth=6, sashrelief=tk.FLAT,
                              background=SASH_IDLE, borderwidth=0, opaqueresize=True)
        pane.grid(row=1, column=0, columnspan=2, sticky=tk.NSEW)
        self.pane = pane
        pane.bind("<Motion>", self._sash_hover)
        pane.bind("<Leave>", lambda _e: pane.configure(background=SASH_IDLE))
        pane.bind("<Map>", lambda _e: pane.after(60, self._place_sash))

        left = ttk.Frame(pane)
        pane.add(left, minsize=320, stretch="always")
        left.rowconfigure(0, weight=1)
        left.columnconfigure(0, weight=1)
        self._list_frame = left
        cols = [f.name for f in self.spec.fields if not f.multiline]
        self.tree = ttk.Treeview(left, columns=cols, show="headings", selectmode="browse")
        for f in self.spec.fields:
            if f.multiline:
                continue
            self.tree.heading(f.name, text=f.label)
            self.tree.column(f.name, width=110, minwidth=60, stretch=False, anchor=tk.W)
        sb = ttk.Scrollbar(left, orient=tk.VERTICAL, command=self.tree.yview)
        self.tree.configure(yscrollcommand=sb.set)
        xsb = ttk.Scrollbar(left, orient=tk.HORIZONTAL, command=self.tree.xview)
        self.tree.configure(xscrollcommand=xsb.set)
        xsb.grid(row=1, column=0, sticky=tk.EW)
        self.tree.grid(row=0, column=0, sticky=tk.NSEW)
        sb.grid(row=0, column=1, sticky=tk.NS)
        self.tree.bind("<<TreeviewSelect>>", self._on_select)

        form = ttk.LabelFrame(pane, text="Record", padding=8)
        pane.add(form, minsize=400, stretch="never")
        form.columnconfigure(0, weight=1)
        form.rowconfigure(0, weight=1)
        self._form_frame = form

        bg = ttk.Style().lookup("TFrame", "background") or "#0d1117"
        canvas = tk.Canvas(form, highlightthickness=0, borderwidth=0, background=bg,
                           yscrollincrement=24, xscrollincrement=24)
        vbar = ttk.Scrollbar(form, orient=tk.VERTICAL, command=canvas.yview)
        hbar = ttk.Scrollbar(form, orient=tk.HORIZONTAL, command=canvas.xview)
        canvas.configure(yscrollcommand=vbar.set, xscrollcommand=hbar.set)
        canvas.grid(row=0, column=0, sticky=tk.NSEW)
        vbar.grid(row=0, column=1, sticky=tk.NS)
        hbar.grid(row=1, column=0, sticky=tk.EW)
        fields = ttk.Frame(canvas)
        fields.columnconfigure(1, weight=1)
        win = canvas.create_window((0, 0), window=fields, anchor="nw")

        def fit(_e: Any = None) -> None:
            canvas.itemconfigure(win, width=max(canvas.winfo_width(), fields.winfo_reqwidth()))
            canvas.configure(scrollregion=canvas.bbox("all"))

        fields.bind("<Configure>", fit)
        canvas.bind("<Configure>", fit)

        for i, f in enumerate(self.spec.fields):
            ttk.Label(fields, text=f.label + ("" if f.required else " (optional)")).grid(
                row=i, column=0, sticky=tk.W, pady=2, padx=(0, 6))
            var = tk.StringVar()
            self.vars[f.name] = var
            if f.kind == "bool":
                w = ttk.Combobox(fields, textvariable=var, values=("true", "false"), state="readonly")
            elif f.kind in ("choice", "ref", "track_ref"):
                # Optional dropdowns get a blank entry so a value can be cleared.
                values = tuple(f.choices) if f.required else ("",) + tuple(f.choices)
                w = ttk.Combobox(fields, textvariable=var, values=values, state="readonly")
            else:
                w = ttk.Entry(fields, textvariable=var)
            w.grid(row=i, column=1, sticky=tk.EW, pady=2)
            self.widgets[f.name] = w

        def on_wheel(event: Any) -> str:
            step = -2 if event.delta > 0 else 2
            if event.state & 0x0001:
                canvas.xview_scroll(step, "units")
            elif fields.winfo_reqheight() > canvas.winfo_height():
                canvas.yview_scroll(step, "units")
            return "break"

        def reveal(event: Any) -> None:
            w = event.widget
            total = fields.winfo_reqheight()
            view = canvas.winfo_height()
            if total <= view or total <= 0:
                return
            top, bottom = w.winfo_y(), w.winfo_y() + w.winfo_height()
            y0 = canvas.canvasy(0)
            if top < y0:
                canvas.yview_moveto(max(0.0, (top - 4) / total))
            elif bottom > y0 + view:
                canvas.yview_moveto(min(1.0, (bottom + 4 - view) / total))

        for w in [canvas, fields] + list(fields.winfo_children()):
            w.bind("<MouseWheel>", on_wheel)
        for w in self.widgets.values():
            w.bind("<FocusIn>", reveal, add="+")

        btns = ttk.Frame(form)
        btns.grid(row=2, column=0, columnspan=2, sticky=tk.EW, pady=(10, 0))
        for text, cmd in (("New", self.new_record), ("Apply to List", self.apply_record),
                          ("Delete", self.delete_record)):
            ttk.Button(btns, text=text, command=cmd).pack(side=tk.LEFT, padx=(0, 4))
        btns2 = ttk.Frame(form)
        btns2.grid(row=3, column=0, columnspan=2, sticky=tk.EW, pady=(6, 0))
        ttk.Button(btns2, text="Save to Source", command=self.save).pack(side=tk.LEFT, padx=(0, 4))
        ttk.Button(btns2, text="Revert", command=self.revert).pack(side=tk.LEFT, padx=(0, 4))
        ttk.Button(btns2, text="Validate Draft", command=self.validate_draft).pack(side=tk.LEFT)
        btns3 = ttk.Frame(form)
        btns3.grid(row=4, column=0, columnspan=2, sticky=tk.EW, pady=(6, 0))
        ttk.Button(btns3, text="Copy to CSV", command=self.copy_csv).pack(side=tk.LEFT, padx=(0, 4))
        ttk.Button(btns3, text="Paste CSV", command=self.paste_csv).pack(side=tk.LEFT)

        state = getattr(self.app, "panel_state", None) or {}
        for which in ("list", "record"):
            if not state.get(which, True):
                self.set_panel_visible(which, False)

    # ---- panel show / hide (View > Show / Hide Panel) ----------------
    def _pane_paths(self) -> list:
        return [str(p) for p in self.pane.panes()]

    def set_panel_visible(self, which: str, show: bool) -> None:
        frame = self._list_frame if which == "list" else self._form_frame
        present = str(frame) in self._pane_paths()
        if show == present:
            return
        try:
            if not show:
                if len(self._pane_paths()) < 2:
                    return
                self.pane.forget(frame)
            elif which == "list":
                paths = self._pane_paths()
                if paths:
                    self.pane.add(frame, before=paths[0], minsize=320, stretch="always")
                else:
                    self.pane.add(frame, minsize=320, stretch="always")
            else:
                self.pane.add(frame, minsize=400, stretch="never")
            paths = self._pane_paths()
            if len(paths) == 1:
                self.pane.paneconfigure(paths[0], stretch="always")
            else:
                self.pane.paneconfigure(self._list_frame, stretch="always")
                self.pane.paneconfigure(self._form_frame, stretch="never")
                self._sash_set = False
                self.pane.after(60, self._place_sash)
        except tk.TclError:
            pass

    # ---- CSV copy / paste --------------------------------------------
    def copy_csv(self) -> None:
        import io
        names = [f.name for f in self.spec.fields]
        buf = io.StringIO()
        w = csv.writer(buf, lineterminator="\r\n")
        w.writerow(names)
        w.writerow([self.vars[n].get().strip() for n in names])
        self.tab.clipboard_clear()
        self.tab.clipboard_append(buf.getvalue())
        self.tab.update()
        self.app.log(f"[{self.spec.table}] Copied current record to clipboard as CSV.")
        self._set_status("Copied current record to clipboard as CSV.")

    def paste_csv(self) -> None:
        import io
        try:
            text = self.tab.clipboard_get()
        except tk.TclError:
            messagebox.showwarning(self.title, "The clipboard is empty or does not contain text.")
            return
        try:
            rows = [r for r in csv.reader(io.StringIO(text.lstrip("\ufeff")))
                    if any(c.strip() for c in r)]
        except csv.Error as exc:
            messagebox.showwarning(self.title, f"Clipboard text is not valid CSV: {exc}")
            return
        if len(rows) < 2:
            messagebox.showwarning(self.title, "Expected a header row followed by one data row.")
            return
        header = [h.strip() for h in rows[0]]
        names = {f.name for f in self.spec.fields}
        pasted = {h: v.strip() for h, v in zip(header, rows[1]) if h in names}
        if not pasted:
            messagebox.showwarning(self.title, "No column in the clipboard CSV matches this table.")
            return
        for f in self.spec.fields:
            blank = "true" if f.kind == "bool" else ""
            self.vars[f.name].set(pasted.get(f.name, blank))
        notes = []
        unknown = [h for h in header if h and h not in names]
        if unknown:
            notes.append("ignored columns: " + ", ".join(unknown))
        if len(rows) > 2:
            notes.append(f"used the first of {len(rows) - 1} data rows")
        self._set_status(f"Pasted {len(pasted)} field(s) into the form; not applied yet."
                         + (" (" + "; ".join(notes) + ")" if notes else ""))

    # ---- data --------------------------------------------------------
    def _sash_hover(self, event: Any) -> None:
        over = bool(self.pane.identify(event.x, event.y))
        self.pane.configure(background=SASH_ACTIVE if over else SASH_IDLE)

    def _place_sash(self) -> None:
        if getattr(self, "_sash_set", False):
            return
        try:
            width = self.pane.winfo_width()
            if width > 700:
                self.pane.sash_place(0, max(320, width - 450), 0)
                self._sash_set = True
        except tk.TclError:
            pass

    def _ref_choices(self) -> None:
        for f in self.spec.fields:
            if f.kind in ("ref", "track_ref"):
                keys = reference_keys(self.app.source_directory, f)
                if not f.required:
                    keys = [""] + keys
                self.widgets[f.name].configure(values=keys)

    def load(self) -> None:
        self.rows = read_rows(self.path, self.spec)
        self.loaded_sha = _sha(self.path)
        self.dirty = False
        self.selected_index = None
        self._ref_choices()
        self._render()
        self._clear_form()

    def update(self) -> None:
        """Called by the main refresh. Never discards unsaved drafts silently."""
        if self.dirty:
            self._ref_choices()
            self._set_status("Unsaved draft kept; source refresh skipped for this table.")
            return
        self.load()

    def _render(self) -> None:
        self.tree.delete(*self.tree.get_children())
        for i, r in enumerate(self.rows):
            self.tree.insert("", tk.END, iid=str(i),
                             values=[r.get(f.name, "") for f in self.spec.fields if not f.multiline])
        n = len(self.rows)
        self._set_status(f"{n} row{'s' if n != 1 else ''}" + (" - UNSAVED CHANGES" if self.dirty else ""))

    def _set_status(self, text: str) -> None:
        self.status.set(text)

    def _clear_form(self) -> None:
        for f in self.spec.fields:
            self.vars[f.name].set("true" if f.kind == "bool" else "")

    def _on_select(self, _e: Any = None) -> None:
        sel = self.tree.selection()
        if not sel:
            return
        self.selected_index = int(sel[0])
        row = self.rows[self.selected_index]
        for f in self.spec.fields:
            self.vars[f.name].set(row.get(f.name, ""))

    def new_record(self) -> None:
        self.tree.selection_remove(self.tree.selection())
        self.selected_index = None
        self._clear_form()
        next_sort = max([int(r["sort_order"]) for r in self.rows if r.get("sort_order", "").isdigit()] or [0]) + 10
        self.vars["sort_order"].set(str(next_sort))

    def _form_row(self) -> dict[str, str] | None:
        row = {f.name: self.vars[f.name].get().strip() for f in self.spec.fields}
        missing = [f.label for f in self.spec.fields if f.required and not row[f.name]]
        if missing:
            messagebox.showwarning(self.title, "Required fields are blank:\n" + "\n".join(missing))
            return None
        for f in self.spec.fields:
            if f.kind == "int" and (f.required or row[f.name]):
                try:
                    int(row[f.name])
                except ValueError:
                    messagebox.showwarning(self.title, f"{f.label} must be a whole number.")
                    return None
        key = tuple(row[k] for k in self.spec.key_columns)
        for i, r in enumerate(self.rows):
            if tuple(r[k] for k in self.spec.key_columns) == key and i != self.selected_index:
                messagebox.showwarning(self.title, f"{' / '.join(key)} already exists.")
                return None
        return row

    def apply_record(self) -> None:
        row = self._form_row()
        if row is None:
            return
        if self.selected_index is None:
            self.rows.append(row)
            self.selected_index = len(self.rows) - 1
        else:
            self.rows[self.selected_index] = row
        self.dirty = True
        self._render()
        self.tree.selection_set(str(self.selected_index))

    def delete_record(self) -> None:
        if self.selected_index is None:
            return
        key = " / ".join(self.rows[self.selected_index][k] for k in self.spec.key_columns)
        if not messagebox.askyesno(self.title, f"Remove {key} from the draft? (Not saved until Save to Source.)"):
            return
        del self.rows[self.selected_index]
        self.selected_index = None
        self.dirty = True
        self._render()
        self._clear_form()

    def revert(self) -> None:
        if self.dirty and not messagebox.askyesno(self.title, "Discard unsaved changes to this table?"):
            return
        self.load()

    def validate_draft(self) -> bool:
        res = validate_candidate(self.app.project_root, self.spec.table, self.rows)
        self._report(res, "Draft validation")
        return res.valid

    def _report(self, res: network_rules.Result, label: str) -> None:
        for line in res.errors:
            self.app.log(f"[{self.spec.table}] ERROR: {line}")
        for line in res.warnings:
            self.app.log(f"[{self.spec.table}] WARNING: {line}")
        if res.errors:
            messagebox.showerror(self.title, f"{label} failed:\n\n" + "\n".join(res.errors[:12]))
        elif res.warnings:
            messagebox.showwarning(self.title, f"{label} passed with warnings:\n\n" + "\n".join(res.warnings[:12]))
        else:
            self._set_status(f"{label} passed.")

    def save(self) -> None:
        if _sha(self.path) != self.loaded_sha:
            messagebox.showerror(self.title, f"{self.spec.filename} changed on disk since it was loaded.\n"
                                             "Revert to reload it, then reapply your edits.")
            return
        res = validate_candidate(self.app.project_root, self.spec.table, self.rows)
        if not res.valid:
            self._report(res, "Save blocked; validation")
            return
        if self.path.is_file():
            self.backup_dir.mkdir(parents=True, exist_ok=True)
            backup = self.backup_dir / f"{self.path.stem}_{datetime.now():%Y%m%d_%H%M%S_%f}.csv"
            shutil.copy2(self.path, backup)
            self.app.log(f"Backup written: {backup}")
        write_rows(self.path, self.spec, self.rows)
        self.app.log(f"Saved {self.spec.filename} ({len(self.rows)} rows).")
        if res.warnings:
            self._report(res, "Saved")
        self.load()
        for other in getattr(self.app, "network_editors", []):
            if other is not self and not other.dirty:
                other.load()


class TransferRulesView:
    def __init__(self, app: Any, tab: ttk.Frame) -> None:
        self.app = app
        tab.columnconfigure(0, weight=1)
        tab.rowconfigure(1, weight=1)
        ttk.Label(tab, text="Transfer and Interruption Rules (read-only; settled design decisions)",
                  style="Section.TLabel").grid(row=0, column=0, sticky=tk.W, pady=(0, 6))
        self.text = tk.Text(tab, wrap=tk.WORD, background="#0d1117", foreground="#e6edf3",
                            insertbackground="#e6edf3", relief=tk.FLAT, font=("Consolas", 10))
        sb = ttk.Scrollbar(tab, orient=tk.VERTICAL, command=self.text.yview)
        self.text.configure(yscrollcommand=sb.set)
        self.text.grid(row=1, column=0, sticky=tk.NSEW)
        sb.grid(row=1, column=1, sticky=tk.NS)
        self.update()

    def update(self) -> None:
        c = network_rules.build_contract()
        lines = [f"Contract version {c['contractVersion']}   Runtime implemented: {c['runtimeImplemented']}", ""]
        lines.append("TRANSFER STATES")
        for s in c["transferStates"]:
            lines.append(f"  {s['state']:<20} {'(terminal) ' if s['terminal'] else ''}{s['description']}")
        lines += ["", "TRANSITIONS"]
        for t in c["transferTransitions"]:
            lines.append(f"  {t['from']:<20} -> {t['to']:<20} on {t['trigger']:<18} {t['notes']}")
        lines += ["", "RULE FLAGS"]
        lines += [f"  {k:<42} {v}" for k, v in c["ruleFlags"].items()]
        lines += ["", "ROLE PERMISSIONS", json.dumps(c["rolePermissions"], indent=2)]
        lines += ["", "TASK AND UPDATE INTERRUPTION", json.dumps(c["taskInterruption"], indent=2)]
        lines += ["", "PERMANENT UNAVAILABILITY CAUSES", "  " + ", ".join(c["permanentUnavailabilityCauses"])]
        lines += ["", "OPEN ITEMS"] + [f"  {k}: {v}" for k, v in c["openItems"].items()]
        self.text.configure(state=tk.NORMAL)
        self.text.delete("1.0", tk.END)
        self.text.insert("1.0", "\n".join(lines))
        self.text.configure(state=tk.DISABLED)


def network_editors_dirty(app: Any) -> list[str]:
    return [e.title for e in getattr(app, "network_editors", []) if e.dirty]

def ordinary_lead_cost(base_cost, growth_percent_scaled, prior_purchases):
    values = (base_cost, growth_percent_scaled, prior_purchases)
    if any(value is None for value in values):
        return None
    if any(type(value) is not int for value in values):
        raise ValueError("Lead pricing inputs must be whole integers")
    full = 100 * SCALE
    if base_cost < 1 or not 0 < growth_percent_scaled <= full or prior_purchases < 0:
        raise ValueError("Base cost and growth must be positive, growth at most 100 percent, purchases nonnegative")
    numerator = base_cost * (full + growth_percent_scaled) ** prior_purchases
    denominator = full ** prior_purchases
    return (2 * numerator + denominator) // (2 * denominator)

def _opt_int(raw, ctx, field, res, minimum):
    if not raw:
        return None
    try:
        n = int(raw)
    except ValueError:
        res.errors.append(f"{ctx}: {field} must be a whole number")
        return None
    if n < minimum:
        res.errors.append(f"{ctx}: {field} must be at least {minimum}")
        return None
    return n


def _percent(raw, ctx, res):
    try:
        d = Decimal(raw)
    except InvalidOperation:
        res.errors.append(f"{ctx}: value must be a number")
        return None
    if not d.is_finite() or not 0 < d <= 100:
        res.errors.append(f"{ctx}: value must be above 0 and at most 100 (percent)")
        return None
    exponent = d.normalize().as_tuple().exponent
    if isinstance(exponent, int) and exponent < -MAX_DECIMAL_PLACES:
        res.errors.append(f"{ctx}: value may have at most {MAX_DECIMAL_PLACES} decimal places")
        return None
    return d

def _validate_greater_territory_tags(src, res, greater_ids):
    import re as _re
    tags_path = src / TAGS_FILE
    links_path = src / TAG_LINKS_FILE
    if not tags_path.is_file():

        elif name.casefold() in names:
            res.errors.append(f"{ctx}: duplicate display_name '{name}'")
        else:
            names.add(name.casefold())
        tags.append({"tagId": tid, "category": category, "displayName": name,
                     "sourceReference": r["source_reference"] or None, "description": r["description"],
                     "sortOrder": net._int(r["sort_order"], ctx, "sort_order", res),
                     "active": net._bool(r["active"], ctx, res)})
    link_rows = (net.read_table(links_path, TAG_LINK_COLUMNS, "greater_territory_tag_links", res)
                 if links_path.is_file() else []) or []
    link_ids, pairs, links = set(), set(), []
    for r in link_rows:
        ctx = f"[greater_territory_tag_links] Line {r['_line']}"
        lid = r["link_id"]
        if not lid or lid in link_ids:
            res.errors.append(f"{ctx}: blank or duplicate link_id '{lid}'")
            continue
        link_ids.add(lid)
        gid, tid = r["greater_territory_id"], r["tag_id"]
        if greater_ids is not None and gid not in greater_ids:
            res.errors.append(f"{ctx}: greater_territory_id '{gid}' is not a known greater_territory_id")
        if tid not in seen:
            res.errors.append(f"{ctx}: tag_id '{tid}' is not a known tag_id")
        if (gid, tid) in pairs:
            res.errors.append(f"{ctx}: duplicate link for '{gid}' and '{tid}'")

        for gid in sorted(greater_ids - linked):
            res.warnings.append(f"[greater_territory_tag_links] Greater Territory '{gid}' has no tags yet")
    tags.sort(key=lambda x: (x["sortOrder"] or 0, x["tagId"]))
    links.sort(key=lambda x: (x["sortOrder"] or 0, x["linkId"]))
    return tags, links

def component_score(value, floor, cap, scale="LINEAR"):
    """Return a 0 to 100 component score, or None when an input is undecided."""
    if value is None or floor is None or cap is None:
        return None
    if cap <= floor:
        raise ValueError("cap must be above floor")
    if scale not in SCALES:
        raise ValueError("scale must be LINEAR or LOG10")
    if scale == "LOG10" and floor < 1:
        raise ValueError("LOG10 floor must be at least 1")
    if value <= floor:
        return 0.0
    if value >= cap:
        return 100.0
    if scale == "LOG10":
        return 100.0 * (math.log10(value) - math.log10(floor)) / (math.log10(cap) - math.log10(floor))
    return 100.0 * (value - floor) / (cap - floor)


def combined_score(scores, weights):
    """Weighted average. A strong component can make up for a weak one."""
    total = 0.0
    weight_sum = 0
    for key, score in scores.items():
        weight = weights.get(key)
        if score is None or not weight:
            continue
        total += score * weight
        weight_sum += weight
    return total / weight_sum if weight_sum else None


def level_for_score(score, levels):
    """Highest level whose min_score is at or below the score, else None."""
    if score is None:
        return None
    best = None
    for level in sorted((lv for lv in levels if lv.get("minScore") is not None), key=lambda lv: lv["minScore"]):
        if level["minScore"] <= score:
            best = level
    return best


def territories_granted(levels, highest_level_id):
    """Permanent territories: sum of grants for every level up to the highest one reached."""
    if highest_level_id is None:
        return 0
    total = 0
    for level in sorted(levels, key=lambda lv: (lv.get("sortOrder") or 0, lv["levelId"])):
        total += level.get("grantTerritories") or 0
        if level["levelId"] == highest_level_id:
            return total
    raise ValueError(f"unknown level '{highest_level_id}'")


def recurring_budget(levels, current_level_id):
    """Recurring budget per cycle follows the CURRENT level, so it can drop."""
    if current_level_id is None:
        return 0
    for level in levels:
        if level["levelId"] == current_level_id:
            return level.get("grantFunding")
    raise ValueError(f"unknown level '{current_level_id}'")

def _decimal(value: str, ctx: str, field: str, res: net.Result) -> Decimal | None:
    if not value:
        res.errors.append(f"{ctx}: {field} is required")
        return None
    try:
        d = Decimal(value)
    except InvalidOperation:
        res.errors.append(f"{ctx}: {field} must be a number")
        return None
    if not d.is_finite():
        res.errors.append(f"{ctx}: {field} must be a finite number")
        return None
    if d <= 0:
        res.errors.append(f"{ctx}: {field} must be above 0")
        return None
    exponent = d.normalize().as_tuple().exponent
    if isinstance(exponent, int) and exponent < -MAX_DECIMAL_PLACES:
        res.errors.append(f"{ctx}: {field} may have at most {MAX_DECIMAL_PLACES} decimal places")
        return None
    return d


def _plain(d: Decimal) -> str:
    return format(d.normalize(), "f")


def _scaled(d: Decimal) -> int:
    return int((d * SCALE).to_integral_value())

        return _plain(d), _scaled(d)

    speed_rows = []
    for s in speeds:
        m = s["_d"]
        if m is None:
            continue
        tics, tics_s = pair(tics1x / m)
        sps, sps_s = pair(Decimal(60) * sim * m * tps / tics1x)
        speed_rows.append({"settingId": s["settingId"], "multiplier": _plain(m),
                           "doomTicsPerSimTick": tics, "doomTicsPerSimTickScaled": tics_s,
                           "strategicSecondsPerRealSecond": sps,
                           "strategicSecondsPerRealSecondScaled": sps_s})
    out_tics, out_tics_s = pair(tps * 60 * sim / outside)
    out_real, out_real_s = pair(Decimal(60) * sim / outside)
    return {"foimsxSpeeds": speed_rows,
            "outsideFoimsx": {"strategicSecondsPerRealSecond": _plain(outside),
                              "doomTicsPerSimTick": out_tics, "doomTicsPerSimTickScaled": out_tics_s,
                              "realSecondsPerSimTick": out_real, "realSecondsPerSimTickScaled": out_real_s}}


def validate_time_settings(root: Path, res: net.Result) -> dict[str, Any]:

    rows = net.read_table(src / SETTINGS_FILE, SETTING_COLUMNS, "time_settings", res) or []
    settings: dict[str, dict[str, Any]] = {}
    for r in rows:
        ctx = f"[time_settings] Line {r['_line']}"
        sid = r["setting_id"]
        if not sid or sid in settings:
            res.errors.append(f"{ctx}: blank or duplicate setting_id '{sid}'")
            continue
        if not r["display_name"]:
            res.errors.append(f"{ctx}: display_name is required")
        category = r["category"].upper()
        if category not in CATEGORIES:
            res.errors.append(f"{ctx}: category must be one of {', '.join(CATEGORIES)}")
        unit = r["unit"].upper()
        if unit not in UNITS:
            res.errors.append(f"{ctx}: unit must be one of {', '.join(UNITS)}")
        value = _decimal(r["value"], ctx, "value", res)
        editable = net._bool(r["editable"], ctx, res)
        if category == "ENGINE" and editable:
            res.errors.append(f"{ctx}: engine constants cannot be editable")
        settings[sid] = {"settingId": sid, "displayName": r["display_name"], "category": category,
                         "unit": unit, "editable": editable,
                         "sourceReference": r["source_reference"] or None,
                         "description": r["description"],
                         "sortOrder": net._int(r["sort_order"], ctx, "sort_order", res),
                         "active": net._bool(r["active"], ctx, res), "_d": value, "_ctx": ctx}
    for sid in REQUIRED_IDS:
        if sid not in settings:
            res.errors.append(f"[time_settings] Required setting '{sid}' is missing")
    sim = settings.get(SIM_TICK_ID)
    if sim and sim["_d"] is not None and sim["_d"] != sim["_d"].to_integral_value():
        res.errors.append(f"{sim['_ctx']}: {SIM_TICK_ID} must be a whole number of minutes")
    eng = settings.get(ENGINE_TIC_ID)
    if eng and eng["_d"] is not None and eng["_d"] != EXPECTED_TICRATE:
        res.warnings.append(f"{eng['_ctx']}: {ENGINE_TIC_ID} is not {EXPECTED_TICRATE} (the engine TICRATE)")
    speeds = sorted((s for s in settings.values() if s["category"] == "FOIMSX_SPEED"),
                    key=lambda s: (s["sortOrder"] or 0, s["settingId"]))
    if len(speeds) != EXPECTED_SPEED_COUNT:
        res.warnings.append(f"[time_settings] Expected {EXPECTED_SPEED_COUNT} FOIMS-X speed settings, found {len(speeds)}")
    for a, b in zip(speeds, speeds[1:]):
        if a["_d"] is not None and b["_d"] is not None and not a["_d"] < b["_d"]:
            res.errors.append(f"{b['_ctx']}: speed multipliers must ascend ({b['settingId']} is not above {a['settingId']})")
    derived = _derived(settings, speeds)
    out_settings = []
    for s in sorted(settings.values(), key=lambda x: (x["sortOrder"] or 0, x["settingId"])):
        item = {k: v for k, v in s.items() if k not in ("_d", "_ctx")}
        d = s["_d"]
        item["value"] = _plain(d) if d is not None else None
        item["valueScaled"] = _scaled(d) if d is not None else None
        out_settings.append(item)
    return {"settings": out_settings, "derived": derived}

r"""
DoDA Content Workbench - Personnel condition catalog (authoring slice).

Authored definitions only: which per-agent condition stats exist, their
ranges, and the severity bands that name each range. Current values belong to
the native simulation and campaign/save state, never to this catalog.

Settled decisions encoded here (2026-10-07):
  * PERS-03: The catalog holds per-agent stats only. Department scores stay in
    metrics.csv; world-pressure stats wait for their own slice.
  * PERS-04: Health is an at-a-glance summary, not an exact measure. It averages
    the contributing condition values (NEGATIVE stats are inverted first) and
    shows a colour-coded "Condition: ..." label. Health bands are therefore
    presentation only: no assignment effects and no forced off duty.
  * PERS-05: A band may suggest a care service where one exists; other actions
    stay as text until personal-care services are authored.
  * PERS-06: Any agent stat may count toward the Health average, chosen per stat
    with a formula: NORMALIZED (position within the stat's min-max range) or
    PERCENT_OF_REFERENCE (value as a percent of another stat, e.g. Current HP
    of Base HP). NEGATIVE stats are inverted before averaging. Loyalty does not
    count.
  * PERS-07: Death happens when Current HP drops to zero or below (killed in
    action): from NPC damage, or from severe illness when Nutrition, Stress, or
    Fatigue drive HP very low. That simulation is not implemented yet.
  * PERS-10: Each stat says how its value starts when an agent or analyst is
    created: DEFAULT (default_value), RANDOM_IN_RANGE (a roll within the stat's
    minimum-maximum), or COPY_OF_STAT (another stat's starting value). Base HP
    rolls 40-120; Current HP starts equal to Base HP.
  * PERS-11 (deferred): HP drains on a tick, like Fatigue and Nutrition; more
    severe conditions drain HP faster.
  * PERS-12: Dead means dead. The Dead label comes from the death state, never
    from the Health average (band_trigger DEATH_STATE).

Source files:
  data_authoring/source/condition_stats.csv
  data_authoring/source/condition_bands.csv
  (condition_bands.suggested_care_service_id references care_services.csv)
"""

def validate_conditions(root: Path, res: net.Result) -> dict[str, Any]:
    src = root / "data_authoring" / "source"
    stat_rows = net.read_table(src / STATS_FILE, STAT_COLUMNS, "condition_stats", res)
    if stat_rows is None:
        return {"stats": [], "bands": []}
    stats: dict[str, dict[str, Any]] = {}
    for r in stat_rows:
        ctx = f"[condition_stats] Line {r['_line']}"
        sid = r["stat_id"]
        if not sid or sid in stats:
            res.errors.append(f"{ctx}: blank or duplicate stat_id '{sid}'")
            continue
        if not r["display_name"]:
            res.errors.append(f"{ctx}: display_name is required")
        for field, allowed in (("stat_category", STAT_CATEGORIES), ("direction", DIRECTIONS),
                               ("storage_model", STORAGE_MODELS), ("player_visibility", VISIBILITIES)):
            if r[field].upper() not in allowed:
                res.errors.append(f"{ctx}: {field} must be one of {', '.join(allowed)}")
        storage, derivation = r["storage_model"].upper(), r["derivation"].upper()
        if storage == "DERIVED" and derivation not in DERIVATIONS:
            res.errors.append(f"{ctx}: DERIVED stats need a derivation ({', '.join(DERIVATIONS)})")
        if storage == "DIRECT" and derivation:
            res.errors.append(f"{ctx}: DIRECT stats have no derivation")
        lo = _opt_int(r["minimum_value"], ctx, "minimum_value", res)
        hi = _opt_int(r["maximum_value"], ctx, "maximum_value", res)
        default = _opt_int(r["default_value"], ctx, "default_value", res)
        if lo is None:
            res.errors.append(f"{ctx}: minimum_value is required")
        if lo is not None and hi is not None and lo >= hi:
            res.errors.append(f"{ctx}: minimum_value must be below maximum_value")
        if default is not None and ((lo is not None and default < lo) or (hi is not None and default > hi)):
            res.errors.append(f"{ctx}: default_value is outside the stat range")
        creation, source = r["creation_rule"].upper(), r["creation_source_stat_id"]
        if storage == "DERIVED" and creation:
            res.errors.append(f"{ctx}: DERIVED stats are calculated, so they have no creation_rule")
        if storage == "DIRECT":
            if creation not in CREATION_RULES:
                res.errors.append(f"{ctx}: creation_rule must be one of {', '.join(CREATION_RULES)}")
            if creation == "DEFAULT" and default is None:
                res.errors.append(f"{ctx}: DEFAULT needs a default_value")
            if creation == "RANDOM_IN_RANGE" and (lo is None or hi is None):
                res.errors.append(f"{ctx}: RANDOM_IN_RANGE needs a known minimum and maximum")
            if creation in ("RANDOM_IN_RANGE", "COPY_OF_STAT") and default is not None:
                res.errors.append(f"{ctx}: default_value is unused by {creation}; leave it blank")
            if creation == "COPY_OF_STAT" and not source:
                res.errors.append(f"{ctx}: COPY_OF_STAT needs creation_source_stat_id")
        if source and creation != "COPY_OF_STAT":
            res.errors.append(f"{ctx}: creation_source_stat_id is only used by COPY_OF_STAT")
        contributes = net._bool(r["contributes_to_health"], ctx, res)
        formula, ref = r["health_formula"].upper(), r["health_reference_stat_id"]
        if contributes:
            if formula not in HEALTH_FORMULAS:
                res.errors.append(f"{ctx}: a Health contributor needs a health_formula "
                                  f"({', '.join(HEALTH_FORMULAS)})")
            if formula == "NORMALIZED" and (hi is None or lo is None):
                res.errors.append(f"{ctx}: NORMALIZED needs a known minimum and maximum")
            if formula == "PERCENT_OF_REFERENCE" and not ref:
                res.errors.append(f"{ctx}: PERCENT_OF_REFERENCE needs health_reference_stat_id")
            if formula != "PERCENT_OF_REFERENCE" and ref:
                res.errors.append(f"{ctx}: health_reference_stat_id is only used by PERCENT_OF_REFERENCE")
        elif formula or ref:
            res.errors.append(f"{ctx}: health_formula and reference are only set when the stat counts toward Health")
        if contributes and derivation == "AVERAGE_OF_CONTRIBUTORS":
            res.errors.append(f"{ctx}: the Health summary cannot contribute to itself")
        stats[sid] = {"statId": sid, "displayName": r["display_name"],
                      "statCategory": r["stat_category"].upper(), "minimumValue": lo, "maximumValue": hi,
                      "defaultValue": default, "creationRule": creation or None,
                      "creationSourceStatId": source or None, "direction": r["direction"].upper(), "storageModel": storage,
                      "derivation": derivation or None, "contributesToHealth": contributes,
                      "healthFormula": formula or None, "healthReferenceStatId": ref or None,
                      "playerVisibility": r["player_visibility"].upper(),
                      "highValueMeaning": r["high_value_meaning"], "lowValueMeaning": r["low_value_meaning"],
                      "sourceReference": r["source_reference"] or None, "description": r["description"],
                      "sortOrder": net._int(r["sort_order"], ctx, "sort_order", res),
                      "active": net._bool(r["active"], ctx, res), "_ctx": ctx}

    for s in stats.values():
        ref = s["healthReferenceStatId"]
        if not s["contributesToHealth"] or s["healthFormula"] != "PERCENT_OF_REFERENCE" or not ref:
            continue
        target = stats.get(ref)
        if target is None:
            res.errors.append(f"{s['_ctx']}: unknown health_reference_stat_id '{ref}'")
        elif ref == s["statId"]:
            res.errors.append(f"{s['_ctx']}: a stat cannot be its own Health reference")
        else:
            if target["minimumValue"] is None or target["minimumValue"] < 1:
                res.errors.append(f"{s['_ctx']}: reference '{ref}' needs a minimum of at least 1 "
                                  "(it is divided by)")
            if s["active"] and not target["active"]:
                res.errors.append(f"{s['_ctx']}: active stat uses inactive Health reference '{ref}'")

    for s in stats.values():
        src_id = s["creationSourceStatId"]
        if s["creationRule"] != "COPY_OF_STAT" or not src_id:
            continue
        origin = stats.get(src_id)
        if origin is None:
            res.errors.append(f"{s['_ctx']}: unknown creation_source_stat_id '{src_id}'")
        elif src_id == s["statId"]:
            res.errors.append(f"{s['_ctx']}: a stat cannot copy its own starting value")
        elif origin["creationRule"] == "COPY_OF_STAT":
            res.errors.append(f"{s['_ctx']}: '{src_id}' is itself a copy; copy from its source instead")
        else:
            olo, ohi = origin["minimumValue"], origin["maximumValue"]
            if (olo is not None and s["minimumValue"] is not None and olo < s["minimumValue"]) or \
               (ohi is not None and s["maximumValue"] is not None and ohi > s["maximumValue"]):
                res.errors.append(f"{s['_ctx']}: range of '{src_id}' does not fit inside this stat's range")
            if s["active"] and not origin["active"]:
                res.errors.append(f"{s['_ctx']}: active stat copies inactive stat '{src_id}'")

    death = stats.get(DEATH_STAT)
    if death is not None and (death["minimumValue"] is None or death["minimumValue"] > 0):
        res.errors.append(f"{death['_ctx']}: {DEATH_STAT} must be able to reach 0 (death rule PERS-07)")

    summaries = [s for s in stats.values() if s["derivation"] == "AVERAGE_OF_CONTRIBUTORS"]
    if len(summaries) > 1:
        res.errors.append("[condition_stats] Only one stat may be the Health summary (AVERAGE_OF_CONTRIBUTORS)")
    for s in summaries:
        if s["active"] and not any(c["contributesToHealth"] and c["active"] for c in stats.values()):
            res.errors.append(f"{s['_ctx']}: active Health summary has no active contributors")

    bands = _validate_bands(src, stats, res)
    clean = lambda d: {k: v for k, v in d.items() if k != "_ctx"}
    return {"stats": sorted((clean(s) for s in stats.values()), key=lambda x: (x["sortOrder"] or 0, x["statId"])),
            "bands": [clean(b) for b in bands]}


def _validate_bands(src: Path, stats: dict[str, dict[str, Any]], res: net.Result) -> list[dict[str, Any]]:
    rows = net.read_table(src / BANDS_FILE, BAND_COLUMNS, "condition_bands", res) or []
    care_ids = _care_ids(src)
    bands: dict[str, dict[str, Any]] = {}
    for r in rows:
        ctx = f"[condition_bands] Line {r['_line']}"
        bid = r["band_id"]
        if not bid or bid in bands:
            res.errors.append(f"{ctx}: blank or duplicate band_id '{bid}'")
            continue
        stat = stats.get(r["stat_id"])
        if stat is None:
            res.errors.append(f"{ctx}: unknown stat_id '{r['stat_id']}'")
        if not r["display_name"]:
            res.errors.append(f"{ctx}: display_name is required")
        severity, effect = r["severity"].upper(), r["assignment_effect"].upper()
        if severity not in SEVERITIES:
            res.errors.append(f"{ctx}: severity must be one of {', '.join(SEVERITIES)}")
        if effect not in ASSIGNMENT_EFFECTS:
            res.errors.append(f"{ctx}: assignment_effect must be one of {', '.join(ASSIGNMENT_EFFECTS)}")
        trigger = r["band_trigger"].upper()
        if trigger not in BAND_TRIGGERS:
            res.errors.append(f"{ctx}: band_trigger must be one of {', '.join(BAND_TRIGGERS)}")
        lo = _opt_int(r["minimum_value"], ctx, "minimum_value", res)
        hi = _opt_int(r["maximum_value"], ctx, "maximum_value", res)
        if trigger == "RANGE" and (lo is None or hi is None):
            res.errors.append(f"{ctx}: RANGE bands need minimum_value and maximum_value")
        if trigger == "DEATH_STATE":
            if lo is not None or hi is not None:
                res.errors.append(f"{ctx}: DEATH_STATE bands have no range; leave minimum/maximum blank")
            if stat is not None and stat["derivation"] != "AVERAGE_OF_CONTRIBUTORS":
                res.errors.append(f"{ctx}: only the Health summary has a DEATH_STATE band (PERS-12)")
        if lo is not None and hi is not None and lo > hi:
            res.errors.append(f"{ctx}: minimum_value is above maximum_value")
        off_duty = net._bool(r["forced_off_duty"], ctx, res)
        active = net._bool(r["active"], ctx, res)
        care_id = r["suggested_care_service_id"]
        if care_id and care_id not in care_ids:
            res.errors.append(f"{ctx}: unknown suggested_care_service_id '{care_id}'")
        if r["ui_color"] and not COLOR_RE.match(r["ui_color"]):
            res.errors.append(f"{ctx}: ui_color must be #RRGGBB or blank")
        if stat is not None:
            if stat["derivation"] == "AVERAGE_OF_CONTRIBUTORS" and (effect != "NONE" or off_duty):
                res.errors.append(f"{ctx}: Health bands are presentation only (PERS-04): "
                                  "assignment_effect must be NONE and forced_off_duty false")
            if active and not stat["active"]:
                res.errors.append(f"{ctx}: active band belongs to inactive stat '{stat['statId']}'")
        bands[bid] = {"bandId": bid, "statId": r["stat_id"], "bandTrigger": trigger, "minimumValue": lo, "maximumValue": hi,
                      "displayName": r["display_name"], "severity": severity, "assignmentEffect": effect,
                      "forcedOffDuty": off_duty, "suggestedAction": r["suggested_action"] or None,
                      "suggestedCareServiceId": care_id or None, "foimsWarning": r["foims_warning"] or None,
                      "uiColor": r["ui_color"] or None, "sourceReference": r["source_reference"] or None,
                      "description": r["description"],
                      "sortOrder": net._int(r["sort_order"], ctx, "sort_order", res), "active": active,
                      "_ctx": ctx}

    for sid in stats:
        if sum(b["statId"] == sid and b["bandTrigger"] == "DEATH_STATE" for b in bands.values()) > 1:
            res.errors.append(f"[condition_bands] {sid}: only one DEATH_STATE band is allowed")

    # Each stat's RANGE bands must tile its range: no gaps, no overlaps.
    for sid, stat in stats.items():
        own = sorted((b for b in bands.values() if b["statId"] == sid and b["bandTrigger"] == "RANGE"
                      and b["minimumValue"] is not None
                      and b["maximumValue"] is not None), key=lambda b: b["minimumValue"])
        if not own:
            continue
        expected = stat["minimumValue"]
        for b in own:
            if expected is not None and b["minimumValue"] > expected:
                res.errors.append(f"[condition_bands] {sid}: gap before {b['bandId']} "
                                  f"({expected}-{b['minimumValue'] - 1} has no band)")
            elif expected is not None and b["minimumValue"] < expected:
                res.errors.append(f"[condition_bands] {sid}: {b['bandId']} overlaps the previous band")
            expected = b["maximumValue"] + 1
        if stat["maximumValue"] is not None:
            if expected - 1 < stat["maximumValue"]:
                res.errors.append(f"[condition_bands] {sid}: no band covers {expected}-{stat['maximumValue']}")
            elif expected - 1 > stat["maximumValue"]:
                res.errors.append(f"[condition_bands] {sid}: bands extend past maximum {stat['maximumValue']}")
    return sorted(bands.values(), key=lambda b: (b["statId"], -1 if b["minimumValue"] is None
                                                 else b["minimumValue"]))

"""Authored missions: plot missions and the Tier 5 capstone chain.
Authoring/export only. Campaign progress and completion are native state."""
from __future__ import annotations

import argparse
import json

    "mission_id", "display_name", "kind", "chain_id", "chain_position", "tier_id",
    "unlocks_tier_id", "territory_id", "map_id", "player_only", "source_reference",
    "description", "sort_order", "active",
]

    "objective_id", "mission_id", "display_name", "required", "source_reference",
    "description", "sort_order", "active",
]

    "sharedTableForPlotMissionsAndCapstone": True,
    "plotMissionsPlayerOnly": True,

    "chainPositionsContiguousFromOne": True,
    "capstoneRequiresRequiredObjectivePerMission": True,
    "tier6UnlockTrigger": "ALL_REQUIREMENTS_MET_AUTOMATIC",
    "oneLeadPurchaseUnlocksWholeChain": "UNDECIDED_LEANING_NO_LEADS_STILL_OPEN_MISSION_AREAS",
    "campaignProgressIsNativeState": True,

OPEN_ITEMS = {
    "CAP-OPEN-01": "Legal review is deferred: the site name is fictional (Serena River Site) for now.",
    "CAP-OPEN-02": "Capstone territory HQ name and the qualification agents need to deploy there.",

def _lookup(src, filename, columns, key, label, res):
    path = src / filename
    if not path.is_file():
        return {}
    scratch = net.Result()
    rows = net.read_table(path, columns, label, scratch) or []
    res.errors.extend(scratch.errors)
    return {row[key]: row for row in rows if row[key]}


def _flag(row, field="active"):
    return row.get(field, "").strip().lower() in ("true", "1", "yes")


def _order(row):
    try:
        return int(row["sort_order"])
    except (KeyError, ValueError):
        return 0


def _capstone_status(missions, objectives):
    caps = [m for m in missions if m["kind"] == "TIER_CAPSTONE"]
    if not caps:
        return {"chainId": None, "missionIds": [], "unlocksTierId": None,
                "configured": False, "missingConfiguration": ["CAPSTONE_MISSION_REFERENCE"]}
    caps.sort(key=lambda m: (m["chainPosition"] or 0, m["missionId"]))
    missing = []
    if any(not m["active"] for m in caps):
        missing.append("CAPSTONE_MISSIONS_INACTIVE")
    if any(not m["mapId"] for m in caps):
        missing.append("CAPSTONE_MAPS")
    if any(not m["territoryId"] for m in caps):
        missing.append("CAPSTONE_TERRITORY")
    covered = {o["missionId"] for o in objectives if o["required"] and o["active"]}
    if any(m["missionId"] not in covered for m in caps):
        missing.append("CAPSTONE_REQUIRED_OBJECTIVES")
    return {"chainId": caps[0]["chainId"], "missionIds": [m["missionId"] for m in caps],
            "unlocksTierId": caps[-1]["unlocksTierId"],
            "configured": not missing, "missingConfiguration": missing}


def validate_authored_missions(root, res):
    src = root / "data_authoring/source"
    tiers = _lookup(src, "mission_tiers.csv", ["tier_id", "active", "sort_order"],
                    "tier_id", "mission_tiers", res)
    territories = _lookup(src, "territories.csv", ["territory_id", "active"],
                          "territory_id", "territories", res)
    maps = _lookup(src, "maps.csv", ["map_id", "active"], "map_id", "maps", res)
    missions, ids = [], set()
    for path, label in ((src / MISSIONS_FILE, "authored_missions"),
                        (src / OBJECTIVES_FILE, "authored_mission_objectives")):
        if not path.is_file():
            res.errors.append(f"[{label}] Required source file missing: {path.name}")
    mission_rows = (net.read_table(src / MISSIONS_FILE, MISSION_COLUMNS, "authored_missions", res)
                    if (src / MISSIONS_FILE).is_file() else []) or []
    objective_rows = (net.read_table(src / OBJECTIVES_FILE, OBJECTIVE_COLUMNS,
                                     "authored_mission_objectives", res)
                      if (src / OBJECTIVES_FILE).is_file() else []) or []

    for row in mission_rows:
        ctx = f"[authored_missions] Line {row['_line']}"
        mid = row["mission_id"]
        if not mid or mid in ids:
            res.errors.append(f"{ctx}: blank or duplicate mission_id '{mid}'")
            continue
        ids.add(mid)
        if not row["display_name"]:
            res.errors.append(f"{ctx}: display_name is required")
        kind = row["kind"].upper()
        if kind not in KINDS:
            res.errors.append(f"{ctx}: kind must be one of {', '.join(KINDS)}")
        if not row["chain_id"]:
            res.errors.append(f"{ctx}: chain_id is required")
        position = net._int(row["chain_position"], ctx, "chain_position", res)
        if position is not None and position < 1:
            res.errors.append(f"{ctx}: chain_position must be at least 1")
        order = net._int(row["sort_order"], ctx, "sort_order", res)
        if order is not None and order < 0:
            res.errors.append(f"{ctx}: sort_order must be nonnegative")
        active = net._bool(row["active"], ctx, res)
        player_only = net._bool(row["player_only"], ctx, res)
        if kind == "PLOT_MISSION" and not player_only:
            res.errors.append(f"{ctx}: plot missions must be player_only")
        tier, unlocks = row["tier_id"], row["unlocks_tier_id"]
        if kind == "TIER_CAPSTONE" and not tier:
            res.errors.append(f"{ctx}: a capstone mission needs tier_id")
        if tier:
            if tier not in tiers:
                res.errors.append(f"{ctx}: unknown tier_id '{tier}'")
            elif active and not _flag(tiers[tier]):
                res.errors.append(f"{ctx}: active mission uses inactive tier '{tier}'")
        if unlocks:
            if kind != "TIER_CAPSTONE":
                res.errors.append(f"{ctx}: only a capstone mission can unlock a tier")
            elif unlocks not in tiers:
                res.errors.append(f"{ctx}: unknown unlocks_tier_id '{unlocks}'")
            elif tier in tiers and _order(tiers[unlocks]) <= _order(tiers[tier]):
                res.errors.append(f"{ctx}: unlocks_tier_id must come after tier_id")
        for field, index, label in (("territory_id", territories, "territory"),
                                    ("map_id", maps, "map")):
            value = row[field]
            if not value:
                continue
            if value not in index:
                res.errors.append(f"{ctx}: unknown {field} '{value}'")
            elif active and not _flag(index[value]):
                res.errors.append(f"{ctx}: active mission uses inactive {label} '{value}'")
        if kind == "TIER_CAPSTONE" and not row["map_id"]:
            res.warnings.append(f"{ctx}: capstone mission '{mid}' has no map yet")
        missions.append({
            "missionId": mid, "displayName": row["display_name"], "kind": kind,
            "chainId": row["chain_id"], "chainPosition": position,
            "tierId": tier or None, "unlocksTierId": unlocks or None,
            "territoryId": row["territory_id"] or None, "mapId": row["map_id"] or None,
            "playerOnly": player_only, "sourceReference": row["source_reference"] or None,
            "description": row["description"], "sortOrder": order, "active": active,
        })

    chains = {}
    for mission in missions:
        chains.setdefault(mission["chainId"], []).append(mission)
    capstone_chains = [cid for cid, items in chains.items()
                       if any(m["kind"] == "TIER_CAPSTONE" for m in items)]
    if len(capstone_chains) > 1:
        res.errors.append("[authored_missions] only one capstone chain is allowed")
    for chain_id, items in chains.items():
        label = f"[authored_missions] Chain '{chain_id}'"
        if len({m["kind"] for m in items}) > 1:
            res.errors.append(f"{label}: missions in one chain must share a kind")
        positions = sorted(m["chainPosition"] for m in items if m["chainPosition"])
        if len(positions) != len(set(positions)):
            res.errors.append(f"{label}: duplicate chain_position")
        elif positions != list(range(1, len(positions) + 1)):
            res.errors.append(f"{label}: chain positions must be contiguous from 1")
        if any(m["kind"] == "TIER_CAPSTONE" for m in items):
            if len({m["tierId"] for m in items}) > 1:
                res.errors.append(f"{label}: capstone missions must share one tier_id")
            final = max((m["chainPosition"] or 0) for m in items)
            with_unlock = [m for m in items if m["unlocksTierId"]]
            if len(with_unlock) != 1 or with_unlock[0]["chainPosition"] != final:
                res.errors.append(f"{label}: exactly the final mission must set unlocks_tier_id")

    mission_index = {m["missionId"]: m for m in missions}
    objectives, objective_ids = [], set()
    for row in objective_rows:
        ctx = f"[authored_mission_objectives] Line {row['_line']}"
        oid = row["objective_id"]
        if not oid or oid in objective_ids:
            res.errors.append(f"{ctx}: blank or duplicate objective_id '{oid}'")
            continue
        objective_ids.add(oid)
        if not row["display_name"]:
            res.errors.append(f"{ctx}: display_name is required")
        active = net._bool(row["active"], ctx, res)
        required = net._bool(row["required"], ctx, res)
        order = net._int(row["sort_order"], ctx, "sort_order", res)
        if order is not None and order < 0:
            res.errors.append(f"{ctx}: sort_order must be nonnegative")
        parent = mission_index.get(row["mission_id"])
        if parent is None:
            res.errors.append(f"{ctx}: unknown mission_id '{row['mission_id']}'")
        elif active and not parent["active"]:
            res.errors.append(f"{ctx}: active objective belongs to inactive mission")
        objectives.append({
            "objectiveId": oid, "missionId": row["mission_id"],
            "displayName": row["display_name"], "required": required,
            "sourceReference": row["source_reference"] or None,
            "description": row["description"], "sortOrder": order, "active": active,
        })
    return {
        "missions": sorted(missions, key=lambda m: (m["chainId"], m["chainPosition"] or 0, m["missionId"])),
        "objectives": sorted(objectives, key=lambda o: (o["missionId"], o["sortOrder"] or 0, o["objectiveId"])),
        "capstoneStatus": _capstone_status(missions, objectives),
    }


def export(root):
    res = net.Result()
    data = validate_authored_missions(root, res)
    if not res.valid:
        return res, None
    out = root / "generated/data/authored_mission_catalog.json"
    out.parent.mkdir(parents=True, exist_ok=True)
    payload = {
        "schema": "doda.authored_mission_catalog", "toolVersion": TOOL_VERSION,
        "generatedAtUtc": datetime.now(timezone.utc).isoformat(),
        **data, "rules": RULES, "openItems": OPEN_ITEMS, "runtimeImplemented": False,
    }
    tmp = out.with_suffix(".json.tmp")
    tmp.write_text(json.dumps(payload, indent=2) + "\n", encoding="utf-8")
    tmp.replace(out)
    return res, out


def main(argv=None):
    parser = argparse.ArgumentParser()
    parser.add_argument("command", choices=["validate", "export"])
    parser.add_argument("--root", type=Path)
    args = parser.parse_args(argv)
    root = args.root.resolve() if args.root else net.find_project_root(Path(__file__).parent)
    out = None
    if args.command == "export":
        res, out = export(root)
    else:
        res = net.Result()
        validate_authored_missions(root, res)
    for message in res.warnings:
        print("WARNING:", message)
    for message in res.errors:
        print("ERROR:", message)
    print("Rows:", res.stats)
    if out:
        print("Exported:", out)
    print("RESULT:", "PASS" if res.valid else "FAIL")
    return 0 if res.valid else 1


if __name__ == "__main__":
    sys.exit(main())
