 150: def load_csv(
 151:     path: Path,
 152:     required_columns: list[str],
 153:     table_name: str,
 154:     result: ValidationResult,
 155: ) -> list[dict[str, str]]:
 156:     if not path.is_file():
 157:         result.error(f"[{table_name}] Missing required source file: {path}")
 158:         return []
 159: 
 160:     try:
 161:         with path.open("r", encoding="utf-8-sig", newline="") as source_file:
 162:             reader = csv.DictReader(source_file)
 163: 
 164:             if reader.fieldnames is None:
 165:                 result.error(f"[{table_name}] File has no CSV header row: {path.name}")
 166:                 return []
 167: 
 168:             actual_columns = [column.strip() for column in reader.fieldnames]
 169:             missing_columns = [
 170:                 column for column in required_columns if column not in actual_columns
 171:             ]
 172: 
 173:             if missing_columns:
 174:                 result.error(
 175:                     f"[{table_name}] Missing required columns in {path.name}: "
 176:                     + ", ".join(missing_columns)
 177:                 )
 178:                 return []
 179: 
 180:             rows: list[dict[str, str]] = []
 181: 
 182:             for line_number, raw_row in enumerate(reader, start=2):
 183:                 row = {
 184:                     key.strip(): (value or "").strip()
 185:                     for key, value in raw_row.items()
 186:                     if key is not None
 187:                 }
 188: 
 189:                 if not any(row.values()):
 190:                     continue
 191: 
 192:                 row["_line_number"] = str(line_number)
 193:                 rows.append(row)
 194: 
 195:             result.stats[table_name] = len(rows)
 196: 
 197:             if not rows:
 198:                 result.warning(f"[{table_name}] Source file contains no data rows: {path.name}")
 199: 
 200:             return rows
 201: 
 202:     except UnicodeDecodeError as error:
 203:         result.error(f"[{table_name}] Could not read UTF-8 CSV {path.name}: {error}")
 204:         return []
 205:     except csv.Error as error:
 206:         result.error(f"[{table_name}] CSV parsing error in {path.name}: {error}")
 207:         return []

 210: def validate_keys(
 211:     table_name: str,
 212:     rows: list[dict[str, str]],
 213:     key_columns: list[str],
 214:     result: ValidationResult,
 215: ) -> None:
 216:     seen_keys: dict[tuple[str, ...], int] = {}
 217: 
 218:     for row in rows:
 219:         line_number = row["_line_number"]
 220:         key = tuple(row.get(column, "").strip() for column in key_columns)
 221: 
 222:         if any(not value for value in key):
 223:             result.error(
 224:                 f"[{table_name}] Line {line_number}: blank key value in "
 225:                 + ", ".join(key_columns)
 226:             )
 227:             continue
 228: 
 229:         if key in seen_keys:
 230:             original_line = seen_keys[key]
 231:             result.error(
 232:                 f"[{table_name}] Line {line_number}: duplicate key "
 233:                 f"{' / '.join(key)}; first defined on line {original_line}"
 234:             )
 235:         else:
 236:             seen_keys[key] = int(line_number)

 239: def validate_active_values(
 240:     table_name: str,
 241:     rows: list[dict[str, str]],
 242:     result: ValidationResult,
 243: ) -> None:
 244:     if not rows or "active" not in rows[0]:
 245:         return
 246: 
 247:     for row in rows:
 248:         value = row.get("active", "").upper()

 251:             result.error(
 252:                 f"[{table_name}] Line {row['_line_number']}: active must be TRUE "
 253:                 f"or FALSE, found '{row.get('active', '')}'"

 270:             f"integer, found '{value}'"

 272:         return None

 283: def validate_metrics(
 284:     metrics: list[dict[str, str]],
 285:     enum_values: dict[str, set[str]],
 286:     result: ValidationResult,
 287: ) -> dict[str, dict[str, str]]:
 288:     metric_by_id: dict[str, dict[str, str]] = {}
 289: 
 290:     allowed_polarity = enum_values.get("METRIC_POLARITY", set())
 291:     allowed_display_modes = enum_values.get("METRIC_DISPLAY_MODE", set())
 292: 
 293:     for row in metrics:
 294:         metric_id = row["metric_id"]
 295:         metric_by_id[metric_id] = row
 296: 
 297:         minimum = parse_int("metrics", row, "internal_min", result)
 298:         maximum = parse_int("metrics", row, "internal_max", result)
 299:         default_value = parse_int("metrics", row, "default_value", result)
 300: 
 301:         if minimum is not None and maximum is not None and minimum > maximum:
 302:             result.error(
 303:                 f"[metrics] Line {row['_line_number']}: internal_min ({minimum}) "
 304:                 f"cannot exceed internal_max ({maximum}) for {metric_id}"
 305:             )
 306: 
 307:         if (
 308:             minimum is not None
 309:             and maximum is not None
 310:             and default_value is not None
 311:             and not minimum <= default_value <= maximum
 312:         ):
 313:             result.error(
 314:                 f"[metrics] Line {row['_line_number']}: default_value "
 315:                 f"({default_value}) is outside [{minimum}, {maximum}] for {metric_id}"
 316:             )
 317: 
 318:         polarity = row["polarity"]
 319:         if polarity not in allowed_polarity:
 320:             result.error(
 321:                 f"[metrics] Line {row['_line_number']}: polarity '{polarity}' "
 322:                 "does not exist in enum_catalog METRIC_POLARITY"
 323:             )
 324: 
 325:         display_mode = row["ui_display_mode"]
 326:         if display_mode not in allowed_display_modes:
 327:             result.error(
 328:                 f"[metrics] Line {row['_line_number']}: ui_display_mode "
 329:                 f"'{display_mode}' does not exist in enum_catalog METRIC_DISPLAY_MODE"
 330:             )
 331: 
 332:     return metric_by_id

 335: def validate_metric_tiers(
 336:     tiers: list[dict[str, str]],
 337:     metrics_by_id: dict[str, dict[str, str]],
 338:     flavor_text_ids: set[str],
 339:     result: ValidationResult,
 340: ) -> None:
 341:     tiers_by_metric: dict[str, list[dict[str, str]]] = defaultdict(list)
 342: 
 343:     for row in tiers:
 344:         metric_id = row["metric_id"]
 345: 
 346:         if metric_id not in metrics_by_id:
 347:             result.error(
 348:                 f"[metric_tiers] Line {row['_line_number']}: metric_id '{metric_id}' "
 349:                 "does not exist in metrics.csv"
 350:             )
 351: 
 352:         flavor_text_id = row["flavor_text_id"]
 353:         if flavor_text_id not in flavor_text_ids:
 354:             result.error(
 355:                 f"[metric_tiers] Line {row['_line_number']}: flavor_text_id "
 356:                 f"'{flavor_text_id}' does not exist in flavor_text.csv"
 357:             )
 358: 
 359:         minimum = parse_int("metric_tiers", row, "min_inclusive", result)
 360:         maximum = parse_int("metric_tiers", row, "max_inclusive", result)
 361:         sort_order = parse_int("metric_tiers", row, "sort_order", result)
 362: 
 363:         if minimum is not None and maximum is not None and minimum > maximum:
 364:             result.error(
 365:                 f"[metric_tiers] Line {row['_line_number']}: min_inclusive "
 366:                 f"({minimum}) cannot exceed max_inclusive ({maximum})"
 367:             )
 368: 
 369:         if sort_order is not None and sort_order < 1:
 370:             result.error(
 371:                 f"[metric_tiers] Line {row['_line_number']}: sort_order must be "
 372:                 "at least 1"
 373:             )
 374: 
 375:         tiers_by_metric[metric_id].append(row)
 376: 
 377:     for metric_id, metric_row in metrics_by_id.items():
 378:         metric_tiers = tiers_by_metric.get(metric_id, [])
 379: 
 380:         if not metric_tiers:
 381:             result.error(
 382:                 f"[metric_tiers] No tiers exist for metric '{metric_id}'"
 383:             )
 384:             continue
 385: 
 386:         parsed_tiers: list[tuple[int, int, int, dict[str, str]]] = []
 387: 
 388:         for row in metric_tiers:
 389:             minimum = parse_int("metric_tiers", row, "min_inclusive", result)
 390:             maximum = parse_int("metric_tiers", row, "max_inclusive", result)
 391:             sort_order = parse_int("metric_tiers", row, "sort_order", result)
 392: 
 393:             if minimum is None or maximum is None or sort_order is None:
 394:                 continue
 395: 
 396:             parsed_tiers.append((minimum, maximum, sort_order, row))
 397: 
 398:         parsed_tiers.sort(key=lambda item: (item[0], item[1], item[2]))
 399: 
 400:         metric_minimum = parse_int("metrics", metric_row, "internal_min", result)
 401:         metric_maximum = parse_int("metrics", metric_row, "internal_max", result)
 402: 
 403:         if metric_minimum is None or metric_maximum is None or not parsed_tiers:
 404:             continue
 405: 
 406:         first_minimum = parsed_tiers[0][0]
 407:         if first_minimum != metric_minimum:
 408:             result.error(
 409:                 f"[metric_tiers] Metric '{metric_id}' begins at {first_minimum}, "
 410:                 f"but metrics.csv defines internal_min {metric_minimum}"
 411:             )
 412: 
 413:         expected_next_minimum = metric_minimum
 414: 
 415:         for minimum, maximum, _, row in parsed_tiers:
 416:             if minimum < expected_next_minimum:
 417:                 result.error(
 418:                     f"[metric_tiers] Line {row['_line_number']}: tier range "
 419:                     f"[{minimum}, {maximum}] overlaps or starts before the "
 420:                     f"expected next value {expected_next_minimum} for {metric_id}"
 421:                 )
 422:             elif minimum > expected_next_minimum:
 423:                 result.error(
 424:                     f"[metric_tiers] Line {row['_line_number']}: tier range "
 425:                     f"[{minimum}, {maximum}] leaves a gap after "
 426:                     f"{expected_next_minimum - 1} for {metric_id}"
 427:                 )
 428: 
 429:             expected_next_minimum = max(expected_next_minimum, maximum + 1)
 430: 
 431:         final_maximum = parsed_tiers[-1][1]
 432:         if final_maximum != metric_maximum:
 433:             result.error(
 434:                 f"[metric_tiers] Metric '{metric_id}' ends at {final_maximum}, "
 435:                 f"but metrics.csv defines internal_max {metric_maximum}"
 436:             )

 465:     enum_values: dict[str, set[str]] = defaultdict(set)

 473: def validate_project(project_root: Path) -> ValidationResult:
 474:     result = ValidationResult()
 475:     source_directory = project_root / "data_authoring" / "source"
 476: 
 477:     loaded_tables: dict[str, list[dict[str, str]]] = {}
 478: 
 479:     for table_name, specification in TABLE_SPECS.items():
 480:         path = source_directory / specification["filename"]
 481:         rows = load_csv(
 482:             path,
 483:             specification["required_columns"],
 484:             table_name,
 485:             result,
 486:         )
 487: 
 488:         loaded_tables[table_name] = rows
 489:         validate_keys(table_name, rows, specification["key_columns"], result)
 490:         validate_active_values(table_name, rows, result)
 491: 
 492:     enum_values = build_enum_index(loaded_tables["enum_catalog"])
 493:     track_ids = build_id_set(loaded_tables["department_tracks"], "track_id")
 494:     flavor_text_ids = build_id_set(loaded_tables["flavor_text"], "flavor_text_id")
 495: 
 496:     metrics_by_id = validate_metrics(
 497:         loaded_tables["metrics"],
 498:         enum_values,
 499:         result,
 500:     )
 501: 
 502:     validate_metric_tiers(
 503:         loaded_tables["metric_tiers"],
 504:         metrics_by_id,
 505:         flavor_text_ids,
 506:         result,
 507:     )
 508: 
 509:     validate_facilities(
 510:         loaded_tables["facilities"],
 511:         track_ids,
 512:         enum_values,
 513:         result,
 514:     )
 515: 
 516:     return result

 600: def main() -> int:
 601:     parser = build_parser()

  81: def sha256_file(path: Path) -> str:
  82:     digest = hashlib.sha256()
  83: 
  84:     with path.open("rb") as source_file:
  85:         for chunk in iter(lambda: source_file.read(65536), b""):
  86:             digest.update(chunk)
  87: 
  88:     return digest.hexdigest()

 122: def normalize_value(table_name: str, column_name: str, value: str) -> Any:
 123:     normalized = value.strip()
 124: 
 125:     if column_name in INTEGER_COLUMNS.get(table_name, set()):
 126:         return int(normalized)
 127: 
 128:     if column_name in BOOLEAN_COLUMNS.get(table_name, set()):
 129:         return normalized.upper() == "TRUE"
 130: 
 131:     return normalized

 134: def load_table(
 135:     project_root: Path,
 136:     table_name: str,
 137:     filename: str,
 138: ) -> list[dict[str, Any]]:
 139:     path = project_root / "data_authoring" / "source" / filename
 140: 
 141:     with path.open("r", encoding="utf-8-sig", newline="") as source_file:
 142:         reader = csv.DictReader(source_file)
 143:         rows: list[dict[str, Any]] = []
 144: 
 145:         for raw_row in reader:
 146:             if raw_row is None:
 147:                 continue
 148: 
 149:             if not any((value or "").strip() for value in raw_row.values()):
 150:                 continue
 151: 
 152:             row: dict[str, Any] = {}
 153: 
 154:             for column_name, raw_value in raw_row.items():
 155:                 if column_name is None:
 156:                     continue
 157: 
 158:                 row[column_name.strip()] = normalize_value(
 159:                     table_name,
 160:                     column_name.strip(),
 161:                     raw_value or "",
 162:                 )
 163: 
 164:             rows.append(row)
 165: 
 166:         return rows

 170:     return (

 179: def language_key_from_flavor_id(flavor_text_id: str) -> str:
 180:     safe_key = re.sub(r"[^A-Za-z0-9_]", "_", flavor_text_id.upper())

 184: def build_language_fragment(flavor_rows: list[dict[str, Any]]) -> str:
 185:     lines = [
 186:         "// GENERATED FILE - DO NOT EDIT BY HAND",
 187:         "// Source: data_authoring/source/flavor_text.csv",
 188:         "// Generated by DoDA Content Workbench.",
 189:         "",
 190:         "[default]",
 191:         "",
 192:     ]
 193: 
 194:     for row in sorted(flavor_rows, key=lambda item: item["flavor_text_id"]):
 195:         base_key = language_key_from_flavor_id(str(row["flavor_text_id"]))
 196:         title = escape_language_text(str(row["title"]))
 197:         body = escape_language_text(str(row["body"]))
 198: 
 199:         lines.append(f'{base_key}_TITLE = "{title}";')
 200:         lines.append(f'{base_key}_BODY = "{body}";')
 201:         lines.append("")
 202: 
 203:     return "\n".join(lines).rstrip() + "\n"

 219: def build_pk3(
 220:     pk3_path: Path,
 221:     department_json_path: Path,
 222:     language_path: Path,
 223:     manifest_path: Path,
 224: ) -> list[str]:
 225:     pk3_path.parent.mkdir(parents=True, exist_ok=True)
 226: 
 227:     archive_entries = {
 228:         "data/department_content.json": department_json_path,
 229:         "language/DODA_GENERATED_LANGUAGE.txt": language_path,
 230:         "manifest/content_manifest.json": manifest_path,
 231:     }
 232: 
 233:     package_readme = (
 234:         "DoDA Generated Content Overlay\n"
 235:         "==============================\n"
 236:         "\n"
 237:         "This PK3 was generated by DoDA Content Workbench.\n"
 238:         "\n"
 239:         "It is currently a data-packaging proof. The archive contains generated\n"
 240:         "department data, a future LANGUAGE fragment, and a build manifest.\n"
 241:         "It does not yet add active runtime behavior because doda-core has not\n"
 242:         "yet been given a native data loader or static include seam for these\n"
 243:         "generated resources.\n"
 244:     )
 245: 
 246:     temporary_path = pk3_path.with_suffix(".pk3.tmp")
 247: 
 248:     if temporary_path.exists():
 249:         temporary_path.unlink()
 250: 
 251:     with zipfile.ZipFile(
 252:         temporary_path,
 253:         mode="w",
 254:         compression=zipfile.ZIP_DEFLATED,
 255:         compresslevel=9,
 256:     ) as archive:
 257:         archive.writestr("README.txt", package_readme)
 258: 
 259:         for archive_name, source_path in archive_entries.items():
 260:             archive.write(source_path, archive_name)
 261: 
 262:     if pk3_path.exists():
 263:         pk3_path.unlink()
 264: 
 265:     temporary_path.replace(pk3_path)
 266: 
 267:     with zipfile.ZipFile(pk3_path, mode="r") as archive:
 268:         return sorted(archive.namelist())

 271: def export_content(project_root: Path) -> tuple[dict[str, Path], list[str]]:

 287:         "schemaVersion": SCHEMA_VERSION,

 335:         "runtimeStatus": {

 337:             "mapinfoIntegrationEnabled": False,
 338:             "languageIntegrationEnabled": False,
 339:             "zscriptIntegrationEnabled": False,
 340:             "notes": [
 341:                 "This is a generated content overlay proof.",
 342:                 "The PK3 is not yet active runtime data.",
 343:                 "Do not manually edit generated output files.",
 344:             ],
 345:         },

 353:         language_path,
 354:         manifest_path,
