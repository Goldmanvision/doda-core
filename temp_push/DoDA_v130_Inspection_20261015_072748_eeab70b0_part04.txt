  454:     def update(self) -> None:
  455:         selected = self.editing_key
  456: 
  457:         self.facility_combo.configure(values=self.facility_ids())
  458:         self.capability_combo.configure(values=self.capability_ids())
  459:         self.tier_combo.configure(values=self.facility_tiers())
  460: 
  461:         for item in self.tree.get_children():
  462:             self.tree.delete(item)
  463: 
  464:         rows = sorted(
  465:             self.rows(),
  466:             key=lambda row: (
  467:                 row.get("facility_id", ""),
  468:                 self.integer_or_default(row.get("sort_order", ""), 999999),
  469:                 row.get("capability_id", ""),
  470:             ),
  471:         )
  472: 
  473:         for row in rows:
  474:             facility_id = row.get("facility_id", "")
  475:             capability_id = row.get("capability_id", "")
  476:             key = f"{facility_id}::{capability_id}"
  477: 
  478:             self.tree.insert(
  479:                 "",
  480:                 tk.END,
  481:                 iid=key,
  482:                 values=(
  483:                     facility_id,
  484:                     capability_id,
  485:                     row.get("display_name_override", ""),
  486:                     row.get("max_concurrent_personnel", ""),
  487:                     row.get("initial_tier_id", ""),
  488:                     row.get("active", ""),
  489:                 ),
  490:             )
  491: 
  492:         if selected:
  493:             key = f"{selected[0]}::{selected[1]}"
  494: 
  495:             if self.tree.exists(key):
  496:                 self.tree.selection_set(key)
  497:                 self.tree.focus(key)
  498: 
  499:         self.update_facility_hint()

  501:     def next_sort_order(self, facility_id: str) -> int:
  502:         values: list[int] = []
  503: 
  504:         for row in self.rows():
  505:             if row.get("facility_id", "") != facility_id:
  506:                 continue
  507: 
  508:             try:
  509:                 values.append(int(row.get("sort_order", "")))
  510:             except ValueError:
  511:                 pass
  512: 
  513:         return max(values, default=0) + 1

  519: 
  520:         if not facility_id:
